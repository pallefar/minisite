**`leanshot/scripts/lighthouse-onboarding.js`** — small node script the Playwright spec invokes:

```javascript
const lighthouse = require('lighthouse');
const chromeLauncher = require('chrome-launcher');

async function main() {
  const url = process.argv[2];
  if (!url) { console.error('usage: lighthouse-onboarding.js <url>'); process.exit(1); }
  const chrome = await chromeLauncher.launch({ chromeFlags: ['--headless=new', '--no-sandbox'] });
  try {
    const result = await lighthouse(url, {
      port: chrome.port,
      output: 'json',
      logLevel: 'error',
      preset: 'mobile',
      throttlingMethod: 'simulate',
      onlyCategories: ['performance', 'accessibility'],
    });
    const lhr = result.lhr;
    const perf = Math.round((lhr.categories.performance.score ?? 0) * 100);
    const a11y = Math.round((lhr.categories.accessibility.score ?? 0) * 100);
    console.log(JSON.stringify({ performance: perf, accessibility: a11y }));
    process.exit(0);
  } finally {
    await chrome.kill();
  }
}
main().catch((err) => { console.error(err); process.exit(2); });
```

import { test, expect } from '@playwright/test';
import { createClient } from '@supabase/supabase-js';

const TEST_SLUG_PREFIX = `p34_activation_${Date.now()}_${Math.random().toString(36).slice(2, 6)}_`;

test.describe('ONBOARD-05 activation event', () => {
  test('first-action tap → record-activation Edge Fn fires → activation_events row written', async ({ page, context }) => {

    const { data: created } = await supabaseAdmin.auth.admin.createUser({ email, email_confirm: true });
    const userId = created.user!.id;
    await supabaseAdmin.from('profiles').upsert({ id: userId, primary_goal: 'lose-weight', completed_onboarding_at: null });

    const { data: linkData } = await supabaseAdmin.auth.admin.generateLink({ type: 'magiclink', email });

    // Network listener for record-activation
    let activationPosted = false;
    await context.route('**/functions/v1/record-activation', async (route) => {
      activationPosted = true;
      await route.continue();
    });

    // Navigate to where FirstActionSurface renders (assume #/onboarding-complete or similar route — adapt to actual nav)

    await page.getByRole('button', { name: /start.*log your first weight/i }).first().click();

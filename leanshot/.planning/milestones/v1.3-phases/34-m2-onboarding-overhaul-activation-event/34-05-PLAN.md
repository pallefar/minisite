**`supabase/migrations/20270706000007_p34_merge_anon_session_rpc.sql`** — SECDEF that operates under advisory lock and returns the winner-row data for the Edge Fn to propagate.

```sql
-- Phase 34 D-07/D-08: merge anonymous_sessions into authenticated user.
-- Race-safe via pg_advisory_xact_lock keyed on user_id.

create or replace function public.merge_anon_session(
  p_user_id    uuid,
  p_cookie_ids text[]
) returns jsonb
language plpgsql
security definer
set search_path = pg_catalog, public, extensions
as $$
declare
  v_winner record;
  v_result jsonb;
begin
  perform pg_advisory_xact_lock(hashtext('merge_anon_session:' || p_user_id::text));

  with eligible as (
    select
      a.id,
      a.cookie_id,
      a.preferences,
      a.draft_entries,
      a.aff_code,
      a.last_activity_at,
      -- D-07: population_score = non-null pref fields + draft count
      (
        coalesce((select count(*) from jsonb_each(a.preferences) where value is not null and value::text <> 'null'), 0)
        + coalesce(jsonb_array_length(a.draft_entries), 0)
      ) as pop_score
    from public.anonymous_sessions a
    where a.cookie_id = any(p_cookie_ids)
      and a.merged_user_id is null
  )
  select id, cookie_id, preferences, draft_entries, aff_code, pop_score
    into v_winner
    from eligible
    order by pop_score desc, last_activity_at desc
    limit 1;

  if v_winner.id is null then
    return jsonb_build_object('winner_id', null);
  end if;

  -- Mark winner + all eligible siblings as merged. Sibling rows stay in the
  -- table until TTL cron sweeps them — preserves audit trail for ~7 days
  -- typical (not 30; 30d TTL applies only to merged_user_id IS NULL rows).
  update public.anonymous_sessions
    set merged_user_id = p_user_id,
        last_activity_at = now()
    where cookie_id = any(p_cookie_ids)
      and merged_user_id is null;

  v_result := jsonb_build_object(
    'winner_id',     v_winner.id,
    'cookie_id',     v_winner.cookie_id,
    'preferences',   v_winner.preferences,
    'draft_entries', v_winner.draft_entries,
    'aff_code',      v_winner.aff_code
  );
  return v_result;
end$$;

revoke all on function public.merge_anon_session(uuid, text[]) from public, anon, authenticated;
grant execute on function public.merge_anon_session(uuid, text[]) to service_role;

comment on function public.merge_anon_session(uuid, text[]) is
  'D-07 richest-data merge under advisory lock. Idempotent within a session — re-runs return {winner_id:null} after first call.';
```

**`supabase/functions/merge-anon-session/`** — full Edge Fn mirroring `plan-personalize/index.ts` for the CORS / admin-singleton / __internal pattern. Differences from create-anon-session (Plan 34-02): this function DOES capture PostHog (alias), so the try/finally + shutdownPostHog wrapping is mandatory.

```typescript
import { aliasServerSide, shutdownPostHog } from '../_shared/posthog-server.ts';

const BodySchema = z.object({
  cookie_ids: z.array(z.string().min(8).max(64)).min(1).max(8),
});

async function handleMergeAnonSession(req: Request): Promise<Response> {
  if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: corsHeaders });
  if (req.method !== 'POST') return jsonError(405, 'method_not_allowed');
  try {
    // 1. Auth
    const auth = req.headers.get('authorization');
    if (!auth) return jsonError(401, 'unauthenticated');
    const admin = getAdmin();
    const { data: userRes, error: authErr } = await admin.auth.getUser(auth.replace(/^Bearer\s+/i, ''));
    if (authErr || !userRes?.user) return jsonError(401, 'unauthenticated');
    const userId = userRes.user.id;

    // 2. Validate body
    const parsed = BodySchema.safeParse(await req.json().catch(() => ({})));
    if (!parsed.success) return jsonError(400, 'invalid_body');

    // 3. Call SECDEF — winner row computed under advisory lock
    const { data: rpcRes, error: rpcErr } = await admin.rpc('merge_anon_session', {
      p_user_id: userId,
      p_cookie_ids: parsed.data.cookie_ids,
    });
    if (rpcErr) return jsonError(500, 'rpc_error');
    const result = rpcRes as { winner_id?: string | null; cookie_id?: string; preferences?: Record<string, unknown>; draft_entries?: unknown[]; aff_code?: string | null };
    if (!result?.winner_id) return jsonResponse(200, { merged: false });

    // 4. Propagate preferences into profiles (D-08a)
    const prefs = result.preferences ?? {};
    const profileUpdates: Record<string, unknown> = {};
    if (typeof prefs.locale === 'string') profileUpdates.locale = prefs.locale;
    if (typeof prefs.units === 'string') profileUpdates.units = prefs.units;
    if (typeof prefs.timezone === 'string') profileUpdates.timezone = prefs.timezone;
    if (typeof prefs.primary_goal === 'string') profileUpdates.primary_goal = prefs.primary_goal;
    if (Object.keys(profileUpdates).length > 0) {
      const { error: profErr } = await admin.from('profiles').update(profileUpdates).eq('id', userId);
      if (profErr) console.warn('[merge-anon-session] profile update failed (non-fatal):', profErr);
    }

    // 5. Affiliate propagation (D-08d) — mirror Phase 19 affiliate-attribute
    if (result.aff_code) {
      const { error: affErr } = await admin
        .from('affiliate_clicks')
        .update({ user_id: userId })
        .eq('aff_code', result.aff_code)
        .is('user_id', null);
      if (affErr) console.warn('[merge-anon-session] affiliate propagation failed (non-fatal):', affErr);
    }

    // 6. PostHog server-side alias (D-08c)
    // Use cookie_id as the anonymous distinct_id surrogate (the client also calls
    // aliasAnonymousToUid with its posthog.get_distinct_id() value; both alias
    // operations are idempotent).
    if (result.cookie_id) {
      aliasServerSide(userId, result.cookie_id);
    }

    return jsonResponse(200, { merged: true, draft_entries: result.draft_entries ?? [] });
  } finally {
    await shutdownPostHog();
  }
}
```

NOTE on draft_entries propagation: the merge does NOT auto-insert draft entries into the user's domain tables (weights, injections, etc.) — that's the browser's job (the dashboard's existing draft-replay flow handles the actual writes after the user reviews them). The Edge Fn just returns the array; the browser hands it to the dashboard.

NOTE on cookie_id vs anonymous distinct_id: in PostHog terms, the anon distinct_id is whatever posthog-js generated when the visitor first hit the site. The browser stores it in localStorage `ph_*_posthog`. The cookie_id we generate in Plan 34-02 is a separate identifier. Per RESEARCH Pattern 4 + Pitfall 7, the canonical alias should be `posthog.alias(supabaseUid, anonDistinctId)`. Since the Edge Fn does NOT have direct access to the browser's posthog distinct_id, two patterns are acceptable:

- **Pattern A (chosen):** Browser sends `{ cookie_ids, anon_distinct_id }` to the Edge Fn; Edge Fn uses `anon_distinct_id` for the server-side alias. This is the right pattern.
- Pattern B: Browser does the alias client-side (existing `aliasAnonymousToUid`); Edge Fn skips server-side alias. Acceptable but loses the adblocker-resistant signal.

Use Pattern A: extend the BodySchema:

```typescript
const BodySchema = z.object({

  anon_distinct_id: z.string().min(8).max(128).optional(),
});
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → merge-anon-session | Authenticated POST; cookie_ids array is untrusted user input (the user could attempt to steal another user's anonymous session). SECDEF blocks this — the merge only succeeds for rows where `merged_user_id IS NULL`; merged rows are skipped silently. |
| Edge Fn → posthog-node alias | posthog-node alias is server-side, key-gated, and idempotent — replaying the alias has no negative effect. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-34-05-01 | Spoofing | Stolen cookie_id | mitigate | Once an anon row is merged, `merged_user_id IS NOT NULL` excludes it from future merges; attacker can't reuse another user's cookie |
| T-34-05-02 | Tampering | Cookie_ids batch injection | mitigate | zod min/max length + array max 8; SECDEF only operates on unmerged rows + advisory lock prevents concurrent steal |
| T-34-05-03 | Information Disclosure | Draft entries leak | mitigate | Per D-10, anonymous_sessions has deny-all RLS for anon role; only the authenticated owner sees their draft entries through this Edge Fn (after auth.getUser validates the JWT) |
| T-34-05-04 | Elevation of Privilege | Replay merge to steal preferences | mitigate | Once merged_user_id is set, subsequent calls return `{ winner_id: null }`; preferences write is idempotent (overwrites with same values) |
| T-34-05-05 | Repudiation | Merge action | accept | merged_user_id + last_activity_at on the merged row provides an audit trail; not surfaced to admin UI until later phase |
| T-34-05-06 | Denial of Service | Large cookie_ids array | mitigate | zod array max 8 caps work per request |

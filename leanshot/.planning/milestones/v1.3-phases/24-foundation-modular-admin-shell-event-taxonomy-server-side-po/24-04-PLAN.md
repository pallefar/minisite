    ```typescript
    // Server-side PostHog helper for Supabase Edge Functions.
    // Phase 24 D-11/D-13. See 24-CONTEXT.md + 24-RESEARCH.md Pattern 3 + PITFALL 1.
    // CRITICAL: every consumer MUST wrap its handler in try/finally and call
    // `await shutdownPostHog()` before returning the Response — otherwise
    // batched events are dropped when the Deno isolate is torn down.

    import { PostHog } from 'npm:posthog-node@5.10.4';

    let _client: PostHog | null = null;
    let _missingKeyWarned = false;

    function getClient(): PostHog | null {
      if (_client) return _client;
      const key = Deno.env.get('POSTHOG_PROJECT_KEY');
      if (!key) {
        if (!_missingKeyWarned) {
          console.warn('[posthog-server] POSTHOG_PROJECT_KEY missing — captureServer is a no-op until set.');
          _missingKeyWarned = true;
        }
        return null;
      }
      const host = Deno.env.get('POSTHOG_HOST') ?? 'https://us.i.posthog.com';
      _client = new PostHog(key, { host });
      return _client;
    }

    export type CaptureArgs = {
      userId: string;           // Supabase auth.users.id — REQUIRED per D-13
      event: string;            // event name from src/lib/analytics/events*.ts
      properties?: Record<string, unknown>;
    };

    export function captureServer(args: CaptureArgs): void {
      if (!args.userId) throw new Error('[posthog-server] userId required (D-13 — always use Supabase auth.users.id as distinct_id)');
      const c = getClient();
      if (!c) return; // vendor-gated no-op
      c.capture({ distinctId: args.userId, event: args.event, properties: args.properties });
    }

    export async function shutdownPostHog(): Promise<void> {
      if (!_client) return;
      try { await _client.shutdown(); } catch (e) { console.error('[posthog-server] shutdown failed', e); }
      _client = null;
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Stripe webhook → posthog-node → PostHog cloud | Untrusted webhook source (verified via signature in existing handler); PostHog ingest is rate-limited but unauthenticated by source |
| Browser anonymous distinct_id → posthog.alias → server-side events | Alias call sets historical anon events under uid; misuse can corrupt analytics for honest users (negligible at v1.3 scale) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-24-05 | Information disclosure (negative — events dropped) | posthog-node not flushed before Edge return | mitigate | try/finally + `await shutdownPostHog()` enforced in every consumer (Task 2 stripe-webhook; future Edge Functions same pattern); Deno test asserts shutdown idempotency |
| T-24-05b | Information disclosure | `stripe_session_id` leaked to PostHog | mitigate | SHA-256 hash only; raw id never sent |
| T-24-04c | Information disclosure | Edge Function imports events.phi.ts and sends PHI to PostHog | mitigate | events.phi.ts is import-allowed in Edge Functions (zone restriction targets only client zones); but each PHI event MUST be reviewed at PR time; Plan 24-08 ROADMAP annotation adds CODEOWNERS on events.phi.ts |
| T-24-12 | Tampering | An attacker setting custom `supabase_uid` in Stripe customer metadata to attribute events to another user | accept (Stripe-side hardening) | Stripe customer metadata is set by our backend code only; webhook signature verification ensures the event came from Stripe |
| T-24-13 | Repudiation | Adblocker user denies receiving a service after server-side payment_completed event recorded | accept | Server-side event is the source of truth; users informed of analytics in privacy policy |
</threat_model>

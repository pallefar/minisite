    Create `scripts/sync-posthog-event-defs.ts`:
    ```typescript
    #!/usr/bin/env node
    import { EVENTS } from '../src/lib/analytics/events';
    import { EVENTS_PHI } from '../src/lib/analytics/events.phi';
    import { zodToJsonSchema } from 'zod-to-json-schema';
    import { readFileSync } from 'node:fs';

    const DRY = process.argv.includes('--dry-run');

    function fail(msg: string): never { console.error(msg); process.exit(1); }

    const projectId = process.env.POSTHOG_PROJECT_ID ?? fail('Error: POSTHOG_PROJECT_ID required');
    const apiKey = process.env.POSTHOG_PROJECT_API_KEY ?? fail('Error: POSTHOG_PROJECT_API_KEY required');
    const host = process.env.POSTHOG_HOST ?? 'https://us.i.posthog.com';

    // TAXO-06 reconciliation guard: refuse to sync if events.ts header lost the reconciliation marker.
    const eventsSrc = readFileSync('src/lib/analytics/events.ts', 'utf8');
    if (!eventsSrc.includes('TAXO-06 reconciliation:')) fail('Error: TAXO-06 reconciliation marker missing from events.ts header — refusing to sync.');

    const allEvents: Array<{ name: string; description: string; phi: boolean; jsonSchema: object }> = [];
    for (const [name, def] of Object.entries(EVENTS))     allEvents.push({ name, description: def.description, phi: def.phi, jsonSchema: zodToJsonSchema(def.payload) });
    for (const [name, def] of Object.entries(EVENTS_PHI)) allEvents.push({ name, description: def.description, phi: def.phi, jsonSchema: zodToJsonSchema(def.payload) });

    async function syncOne(ev: { name: string; description: string; phi: boolean; jsonSchema: object }): Promise<void> {
      const url = `${host}/api/projects/${projectId}/event_definitions/${encodeURIComponent(ev.name)}/`;
      const body = { name: ev.name, description: ev.description, tags: [ev.phi ? 'phi' : 'non-phi'] };
      if (DRY) { console.log(`[DRY] PATCH ${url} body=${JSON.stringify(body)}`); return; }
      const res = await fetch(url, { method: 'PATCH', headers: { Authorization: `Bearer ${apiKey}`, 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
      if (!res.ok) fail(`PATCH ${ev.name} failed: ${res.status} ${await res.text()}`);
      console.log(`[OK] ${ev.name}`);
    }

    (async () => { for (const ev of allEvents) await syncOne(ev); console.log(`Synced ${allEvents.length} event definitions.`); })().catch((e) => fail(String(e)));
    ```

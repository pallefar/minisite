    ```typescript
    fetchUserModerationStatus: async () => {
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) {
        set({ userModerationStatus: null, userModerationExpiresAt: null, userModerationReason: null });
        return;
      }
      const { data, error } = await supabase
        .from('user_moderation_state')
        .select('status, expires_at, reason')
        .eq('user_id', user.id)
        .maybeSingle();
      if (error) {
        // RLS denial or network error — fall back to 'active' (don't block legitimate users on transient errors)
        set({ userModerationStatus: 'active', userModerationExpiresAt: null, userModerationReason: null });
        return;
      }
      set({
        userModerationStatus:    (data?.status as any) ?? 'active',
        userModerationExpiresAt: data?.expires_at ?? null,
        userModerationReason:    data?.reason ?? null,
      });
    },
    ```

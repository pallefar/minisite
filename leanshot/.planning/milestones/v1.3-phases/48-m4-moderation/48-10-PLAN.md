    create or replace function public.triage_report(p_report_id uuid)
    returns void
    language plpgsql
    security definer
    set search_path = public, extensions
    as $fn$
    declare
      v_actor uuid := auth.uid();
      v_before jsonb;
    begin
      if not (public.is_staff() or public.can_moderate_report_org(p_report_id)) then
        raise exception 'forbidden' using errcode = '42501';
      end if;

      select to_jsonb(cr.*) into v_before from public.community_reports cr where cr.id = p_report_id;

      update public.community_reports
        set status     = 'triaged',
            triaged_by = v_actor,
            triaged_at = now()
      where id = p_report_id
        and status = 'open';
      if not found then
        raise exception 'not_open' using errcode = 'P0002';
      end if;

      perform public.log_moderation_action(
        p_action_type => 'report_triaged',
        p_target_type => 'report',
        p_target_id   => p_report_id,
        p_before      => v_before,
        p_after       => jsonb_build_object('status','triaged','triaged_by', v_actor),
        p_reason      => null
      );
    end;
    $fn$;

    revoke all on function public.triage_report(uuid) from public;
    grant execute on function public.triage_report(uuid) to authenticated;
    ```

    /**
     * Phase 48 Plan 10 — ModerationLayout (admin pathname-based module).
     *
     * Pathname-based routing per project convention (CommunityAdminLayout.tsx, ReviewsLayout.tsx).
     * Sub-routes (resolveView regex):
     *   /admin/moderation                 → ReportsQueue (default)
     *   /admin/moderation/auto-flags      → ReportsQueue filtered reporter=NULL
     *   /admin/moderation/banned-words    → BannedWordsEditor
     *   /admin/moderation/bans            → UserBansRoster
     *   /admin/moderation/bans/:userId    → ApplyModerationForm
     *   /admin/moderation/audit-log       → AuditLogViewer
     */
    import { Suspense, lazy, useEffect, useMemo, useState } from 'react';
    import { Skeleton } from '@/components/ui/Skeleton';

    const ReportsQueue        = lazy(() => import('./ReportsQueue').then(m => ({ default: m.ReportsQueue })));
    const BannedWordsEditor   = lazy(() => import('./BannedWordsEditor').then(m => ({ default: m.BannedWordsEditor })));
    const UserBansRoster      = lazy(() => import('./UserBansRoster').then(m => ({ default: m.UserBansRoster })));
    const ApplyModerationForm = lazy(() => import('./ApplyModerationForm').then(m => ({ default: m.ApplyModerationForm })));
    const AuditLogViewer      = lazy(() => import('./AuditLogViewer').then(m => ({ default: m.AuditLogViewer })));

    function resolveView(pathname: string): { view: string; id?: string } {
      const m = pathname.match(/^\/admin\/moderation\/?([^/]+)?\/?([^/]+)?/);
      const sub  = m?.[1] ?? '';
      const seg2 = m?.[2];
      switch (sub) {
        case '':              return { view: 'reports' };
        case 'auto-flags':    return { view: 'auto-flags' };
        case 'banned-words':  return { view: 'banned-words' };
        case 'bans':          return seg2 ? { view: 'apply', id: seg2 } : { view: 'bans' };
        case 'audit-log':     return { view: 'audit' };
        default:              return { view: 'reports' };
      }
    }

    export default function ModerationLayout() {
      const [path, setPath] = useState(window.location.pathname);
      useEffect(() => {
        const onPop = () => setPath(window.location.pathname);
        window.addEventListener('popstate', onPop);
        return () => window.removeEventListener('popstate', onPop);
      }, []);
      const { view, id } = useMemo(() => resolveView(path), [path]);

      return (
        <Suspense fallback={<Skeleton />}>
          {view === 'reports'       && <ReportsQueue mode="all" />}
          {view === 'auto-flags'    && <ReportsQueue mode="auto-flags" />}
          {view === 'banned-words'  && <BannedWordsEditor />}
          {view === 'bans'          && <UserBansRoster />}
          {view === 'apply'         && <ApplyModerationForm userId={id!} />}
          {view === 'audit'         && <AuditLogViewer />}
        </Suspense>
      );
    }
    ```

    import { supabase } from '@/lib/supabase';

    export async function triageReport(reportId: string) {
      const { error } = await supabase.rpc('triage_report', { p_report_id: reportId });
      if (error) throw error;
    }
    export async function dismissReport(reportId: string, reason: string) {
      const { error } = await supabase.rpc('dismiss_report', { p_report_id: reportId, p_reason: reason });
      if (error) throw error;
    }
    export async function resolveReport(reportId: string) {
      const { error } = await supabase.rpc('resolve_report', { p_report_id: reportId });
      if (error) throw error;
    }
    export async function applyUserModeration(args: { userId: string; status: 'active'|'muted'|'banned'|'temp_suspended'; reason: string | null; expiresAt: string | null; }) {
      const { error } = await supabase.rpc('apply_user_moderation', {
        p_user_id: args.userId, p_status: args.status, p_reason: args.reason, p_expires_at: args.expiresAt,
      });
      if (error) throw error;
    }
    export async function bannedWordUpsert(word: string, severity: 'warn'|'flag'|'escalate') {
      const { data, error } = await supabase.rpc('banned_word_upsert', { p_word: word, p_severity: severity });
      if (error) throw error;
      return data as string;
    }
    export async function bannedWordRemove(id: string) {
      const { error } = await supabase.rpc('banned_word_remove', { p_id: id });
      if (error) throw error;
    }
    export async function invokeBannedWordsSweep(table: 'community_posts'|'community_comments', startCursor: string | null) {
      const { data, error } = await supabase.functions.invoke('banned-words-sweep', {
        body: { table, start_cursor: startCursor, batch_size: 100 },
      });
      if (error) throw error;
      return data as { processed: number; matches: number; next_cursor: string | null; done: boolean };
    }

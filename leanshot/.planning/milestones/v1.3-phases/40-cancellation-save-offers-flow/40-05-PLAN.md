create or replace function public.save_offer_rule_create(
  p_title text,
  p_cohort_id uuid,                -- nullable; null = "any cohort"
  p_tenure_buckets text[],         -- subset of {'<30d','30-180d','>180d'}; empty = any
  p_reasons text[],                -- subset of 7 reason codes; empty = any
  p_org_type text,                 -- 'any'|'consumer'|'clinic'
  p_offer_type text,               -- 'pause'|'discount'|'extended_trial'|'downgrade'|'contact_csm'
  p_pause_months integer,          -- 1|2|3|null
  p_coupon_id text,                -- 'SAVE-{20|25|30}-{2|3}MO' or null
  p_extension_days integer,        -- 7|14|30|null
  p_downgrade_target text,         -- 'price_monthly' or null
  p_priority integer
) returns uuid;
-- Gate: is_admin_at_least('admin'); raise 'forbidden' (42501) otherwise

create or replace function public.save_offer_rule_update(
  p_id uuid,
  p_title text,
  -- same column args as create
  p_active boolean                  -- separate field for activate-toggle behavior
) returns void;
-- Gate: is_admin_at_least('admin')

create or replace function public.save_offer_rule_archive(
  p_id uuid
) returns void;
-- Gate: is_admin_at_least('superadmin') — destructive lifecycle gate per cohort_set_status precedent
-- Sets active=false + records archived_at via log_admin_action; rule history preserved for ROI dashboard reconciliation

create or replace function public.save_offer_rule_reorder(
  p_ids uuid[],                    -- ordered array; index = priority (0 = highest)
) returns void;
-- Gate: is_admin_at_least('admin')
-- Single transactional UPDATE reassigning priority = array_position(p_ids, id) for each id in p_ids

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| admin browser→SECDEF RPCs | user JWT verified inside RPC via auth.uid; role gate via is_admin_at_least |
| AdminShell→CancellationModule | client-side surfaceCheck UX hint; DB-side SECDEF RPC role enforcement is authoritative |
| RuleEditor→Stripe coupon catalog | catalog hardcoded in client + validated server-side in RPC body (catalog regex) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-40-05-01 | Elevation of Privilege | Staff-role user calls save_offer_rule_create via supabase.rpc directly (bypassing UI) | mitigate | SECDEF RPC checks `is_admin_at_least('admin')` server-side; raises 42501 'forbidden'; pgTAP proof asserts this |
| T-40-05-02 | Elevation of Privilege | Admin (not superadmin) archives a rule via direct rpc call | mitigate | save_offer_rule_archive gates at 'superadmin'; pgTAP proof asserts |
| T-40-05-03 | Tampering | Client passes arbitrary coupon_id like 'SAVE-99-99MO' to apply discounts outside catalog | mitigate | RPC validates regex `^SAVE-(20|25|30)-(2|3)MO$`; raises 22023 invalid_input |
| T-40-05-04 | Tampering | Race: two admins reorder rules concurrently producing inconsistent priorities | mitigate | save_offer_rule_reorder runs in single transaction with atomic UPDATE; last writer wins (acceptable for v1.3 low-write-volume admin surface) |
| T-40-05-05 | Information Disclosure | Manifest entry leaks admin-only UI to non-admin users | mitigate | Manifest minRole='admin' gates UI render; module entry returns NoAccess for sub-admin; surfaceCheck is hint-only — DB is authoritative |
| T-40-05-06 | Repudiation | Rule changes occur without audit | mitigate | Every SECDEF RPC body calls `log_admin_action(p_action_name=..., p_metadata=...)`; audit log preserved per app.suppress_audit pattern |
| T-40-05-07 | Tampering | RPC SQL injection via p_title (text input) | mitigate | Parameterized via supabase-js + plpgsql bind variables; never string-concatenated into SQL |
</threat_model>

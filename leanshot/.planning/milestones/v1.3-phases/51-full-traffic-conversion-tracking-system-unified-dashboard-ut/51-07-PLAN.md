State + fetch + anomaly join:
```typescript
type FunnelRow = { channel_group: string; stage_in: string; stage_out: string; in_count: number; out_count: number; rate: number };
type AnomalyPayload = { channel_group: string; audience: string; stage_in: string; stage_out: string; sigmas: number };

export const TrafficFunnelsTab: React.FC = () => {
  const [audience, setAudience] = useState<Audience>('consumer');
  const [rows, setRows] = useState<FunnelRow[] | null>(null);
  const [anomalies, setAnomalies] = useState<AnomalyPayload[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [drillStage, setDrillStage] = useState<{ stage_in: string; stage_out: string } | null>(null);

  const orgFilter = useStore((s) => s.user?.role === 'clinic_owner' ? s.user.org?.id ?? null : null);

  const today = new Date().toISOString().slice(0, 10);
  const sevenAgo = new Date(Date.now() - 7 * 86400000).toISOString().slice(0, 10);

  const fetchFunnel = useCallback(async () => {
    setLoading(true); setError(null);
    const [{ data: funnelData, error: rpcErr }, { data: anomalyData }] = await Promise.all([
      supabase.rpc('get_traffic_funnel_rollup', {
        p_org_id: orgFilter, p_start_date: sevenAgo, p_end_date: today, p_audience: audience,
      }),
      supabase.from('admin_notifications').select('payload').eq('kind', 'traffic_funnel_drop').gte('created_at', new Date(Date.now() - 24 * 3600 * 1000).toISOString()),
    ]);
    if (rpcErr) {
      if (rpcErr.code === '42501') setError('Permission denied — admin role required');
      else setError(`Failed to load funnel — ${rpcErr.message}`);
      setRows(null);
    } else {
      setRows((funnelData as FunnelRow[]) ?? []);
      setAnomalies((anomalyData ?? []).map((r) => r.payload as AnomalyPayload));
    }
    setLoading(false);
  }, [orgFilter, audience, sevenAgo, today]);

  useEffect(() => { void fetchFunnel(); }, [fetchFunnel]);

  // Aggregate per stage_pair (across channel_groups + days)
  const stageRows = useMemo(() => {
    if (!rows) return null;
    return STAGE_LABELS[audience].map((sp) => {
      const matching = rows.filter((r) => r.stage_in === sp.stage_in && r.stage_out === sp.stage_out);
      const in_count = matching.reduce((acc, r) => acc + r.in_count, 0);
      const out_count = matching.reduce((acc, r) => acc + r.out_count, 0);
      const rate = in_count > 0 ? out_count / in_count : 0;
      const has_anomaly = anomalies.some((a) => a.audience === audience && a.stage_in === sp.stage_in && a.stage_out === sp.stage_out);
      const max_sigma = anomalies.filter((a) => a.audience === audience && a.stage_in === sp.stage_in && a.stage_out === sp.stage_out).reduce((m, a) => Math.max(m, a.sigmas), 0);
      return { ...sp, in_count, out_count, rate, has_anomaly, max_sigma };
    });
  }, [rows, anomalies, audience]);

  return (
    <section>
      <div className="flex items-center justify-between mb-3">
        <h2 className="text-[11px] font-semibold text-[var(--color-text-secondary)] uppercase tracking-[0.06em]">Conversion Funnels</h2>
        <div className="flex items-center gap-2">
          <PillGroup ariaLabel="Audience">
            <Pill size="sm" aria-pressed={audience === 'consumer'} onClick={() => setAudience('consumer')}>Consumer</Pill>
            <Pill size="sm" aria-pressed={audience === 'clinic-org'} onClick={() => setAudience('clinic-org')}>Clinic-org</Pill>
            <Pill size="sm" aria-pressed={audience === 'affiliate'} onClick={() => setAudience('affiliate')}>Affiliate</Pill>
          </PillGroup>
          <Button variant="ghost" size="sm" leadingIcon={<RefreshCw size={12} aria-hidden />} onClick={() => void fetchFunnel()} aria-label="Refresh">Refresh</Button>
        </div>
      </div>

      {loading && <Skeleton className="h-64 rounded-card" />}
      {error && !loading && <Card padding="md"><p className="text-[13px] text-[var(--color-danger)]">{error}</p></Card>}
      {!loading && !error && stageRows && stageRows.every((s) => s.in_count === 0) && (
        <EmptyState heading={`No funnel data for ${audience}`} body={`No ${audience} cohort recorded in the selected window. Switch audience or widen the range.`} />
      )}

      {!loading && !error && stageRows && stageRows.some((s) => s.in_count > 0) && (
        <Card padding="md">
          <BaseChart
            type="bar"
            data={{
              labels: stageRows.map((s) => `${s.label_in} → ${s.label_out}`),
              datasets: [
                { label: 'Entered', data: stageRows.map((s) => s.in_count), backgroundColor: 'var(--color-text-secondary)' },
                { label: 'Advanced', data: stageRows.map((s) => s.out_count), backgroundColor: 'var(--color-primary)' },
              ],
            }}
            options={{ responsive: true, plugins: { legend: { display: true, position: 'bottom' } }, scales: { y: { beginAtZero: true } } }}
          />
          <ul className="mt-4 space-y-2">
            {stageRows.map((s) => (
              <li key={`${s.stage_in}_${s.stage_out}`} className="flex items-center justify-between gap-3">
                <button className="text-[13px] text-left hover:underline" type="button" onClick={() => setDrillStage({ stage_in: s.stage_in, stage_out: s.stage_out })}>
                  {s.label_in} → {s.label_out}: <span className="numerals-tabular">{(s.rate * 100).toFixed(1)}%</span> • <span className="numerals-tabular">{s.in_count.toLocaleString()}</span> entered
                </button>
                {s.has_anomaly && (
                  <Badge tone="warning"><AlertTriangle size={12} aria-hidden /> Below 7-day baseline by {s.max_sigma.toFixed(1)}σ</Badge>
                )}
              </li>
            ))}
          </ul>
        </Card>
      )}

      <Sheet open={drillStage != null} onOpenChange={(open) => !open && setDrillStage(null)}>
        {drillStage && rows && (
          <div className="p-4">
            <button className="flex items-center gap-1 text-[13px] text-[var(--color-primary)] mb-4 hover:underline" onClick={() => setDrillStage(null)} type="button">← Back to funnel</button>
            <h3 className="text-[18px] font-bold mb-2">Channel Origin (top 5 per stage)</h3>
            <p className="text-[11px] text-[var(--color-text-secondary)] mb-3">{drillStage.stage_in} → {drillStage.stage_out}</p>
            <table className="w-full text-[13px]">
              <thead>
                <tr className="text-[11px] font-semibold text-[var(--color-text-secondary)] uppercase tracking-[0.06em] text-left">
                  <th>Channel</th><th className="text-right">Entered</th><th className="text-right">Advanced</th><th className="text-right">Rate</th>
                </tr>
              </thead>
              <tbody>
                {Object.entries(
                  rows
                    .filter((r) => r.stage_in === drillStage.stage_in && r.stage_out === drillStage.stage_out)
                    .reduce((acc, r) => {
                      acc[r.channel_group] = acc[r.channel_group] ?? { in: 0, out: 0 };
                      acc[r.channel_group].in += r.in_count;
                      acc[r.channel_group].out += r.out_count;
                      return acc;
                    }, {} as Record<string, { in: number; out: number }>)
                ).sort(([, a], [, b]) => b.in - a.in).slice(0, 5).map(([cg, vals]) => (
                  <tr key={cg}><td>{cg}</td><td className="text-right numerals-tabular">{vals.in.toLocaleString()}</td><td className="text-right numerals-tabular">{vals.out.toLocaleString()}</td><td className="text-right numerals-tabular">{vals.in > 0 ? ((vals.out / vals.in) * 100).toFixed(1) + '%' : '—'}</td></tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </Sheet>
    </section>
  );
};
```

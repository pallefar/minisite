**RPC body:**
```sql
create or replace function public.compute_channel_stage_rate(
  p_channel_group text,
  p_audience text,
  p_stage_in text,
  p_stage_out text,
  p_window_days int default 7
) returns table (
  observed_rate numeric,
  expected_rate numeric,
  expected_stddev numeric
)
language plpgsql security definer stable
set search_path = pg_catalog, public, extensions
as $$
begin
  return query
  with windowed as (
    select day, rate
    from public.traffic_funnel_rollup
    where channel_group = p_channel_group
      and audience = p_audience
      and stage_in = p_stage_in
      and stage_out = p_stage_out
      and day >= (current_date - make_interval(days => p_window_days))
      and day < current_date
  ),
  today as (
    select rate as today_rate
    from public.traffic_funnel_rollup
    where channel_group = p_channel_group
      and audience = p_audience
      and stage_in = p_stage_in
      and stage_out = p_stage_out
      and day = current_date
    limit 1
  )
  select
    coalesce((select today_rate from today), 0)::numeric as observed_rate,
    coalesce(avg(rate), 0)::numeric as expected_rate,
    coalesce(stddev_samp(rate), 0)::numeric as expected_stddev
  from windowed;
end;
$$;

revoke execute on function public.compute_channel_stage_rate(text, text, text, text, int) from public, anon, authenticated;
grant execute on function public.compute_channel_stage_rate(text, text, text, text, int) to service_role;
```

```typescript
// Per-channel-stage anomaly scan (Phase 51 TRAFFIC-11 extension)
const audiences = ['consumer', 'clinic-org', 'affiliate'] as const;
const stagePairs: Record<typeof audiences[number], Array<[string, string]>> = {
  'consumer': [['visit','signup'], ['signup','activation'], ['activation','paid']],
  'clinic-org': [['visit','clinic_signup'], ['clinic_signup','first_patient_added'], ['first_patient_added','first_paid_seat']],
  'affiliate': [['visit','affiliate_signup'], ['affiliate_signup','first_referral_conversion']],
};

// Read channel_groups from taxonomy
const { data: channelGroups, error: cgErr } = await admin()
  .from('channel_groups')
  .select('label')
  .order('priority', { ascending: true });
if (cgErr) {
  console.warn('[funnel-anomaly-cron] channel_groups read failed:', cgErr.message);
} else {
  const sigmaThreshold = 2; // 2σ — mirror existing per-funnel threshold
  for (const cg of channelGroups ?? []) {
    for (const audience of audiences) {
      for (const [stageIn, stageOut] of stagePairs[audience]) {
        const { data: rateData, error: rateErr } = await admin().rpc('compute_channel_stage_rate', {
          p_channel_group: cg.label,
          p_audience: audience,
          p_stage_in: stageIn,
          p_stage_out: stageOut,
          p_window_days: 7,
        });
        if (rateErr || !rateData || !rateData[0]) continue;
        const { observed_rate, expected_rate, expected_stddev } = rateData[0];
        if (expected_stddev === 0 || expected_rate === 0) continue; // not enough baseline
        const sigmas = (expected_rate - observed_rate) / expected_stddev;
        if (sigmas < sigmaThreshold) continue; // not anomalous

        // Dedup key: (channel_group, audience, funnel, stage_pair, date)
        const today = new Date().toISOString().slice(0, 10);
        const dedupKey = `traffic_funnel_drop:${cg.label}:${audience}:${stageIn}_${stageOut}:${today}`;

        // 4h suppression — mirror existing per-funnel pattern (lines 217-232 of existing file)
        const fourHrAgo = new Date(Date.now() - 4 * 60 * 60 * 1000).toISOString();
        const { data: recent } = await admin()
          .from('admin_notifications')
          .select('id')
          .eq('dedup_key', dedupKey)
          .gte('created_at', fourHrAgo)
          .limit(1);
        if (recent && recent.length > 0) continue; // already alerted within 4h

        const { error: insErr } = await admin()
          .from('admin_notifications')
          .upsert({
            kind: 'traffic_funnel_drop',
            dedup_key: dedupKey,
            payload: {
              channel_group: cg.label,
              audience,
              funnel: audience,
              stage_in: stageIn,
              stage_out: stageOut,
              observed_rate,
              expected_rate,
              expected_stddev,
              sigmas,
              date: today,
            },
          }, { onConflict: 'dedup_key', ignoreDuplicates: true });
        if (insErr) console.warn('[funnel-anomaly-cron] admin_notifications upsert failed:', insErr.message);
      }
    }
  }
}
```

```sql
-- Phase 62 — pg_cron schedules
-- Daily 02:00 UTC: REFRESH MATERIALIZED VIEW CONCURRENTLY for 5 insights_*_rollup matviews
-- Nightly 01:00 UTC: purge revoked-consent users' insights data (last_purged_at < consent_revoked_at)
-- Both use SQL-only paths — no Edge Fn calls — so Fn deployment ordering only matters for upstream publishers.

begin;

-- 1) Matview refresh (daily 02:00 UTC) — mirrors phase60_eval_nightly cron pattern
-- IMPORTANT: REFRESH MATERIALIZED VIEW CONCURRENTLY requires UNIQUE INDEX on each matview (Plan 62-01 ships these).
do $$
begin
  if exists (select 1 from cron.job where jobname = 'phase62_matview_refresh') then
    perform cron.unschedule('phase62_matview_refresh');
  end if;
end $$;

select cron.schedule(
  'phase62_matview_refresh',
  '0 2 * * *',
  $cron$
    refresh materialized view concurrently public.insights_dose_rollup;
    refresh materialized view concurrently public.insights_body_metrics_rollup;
    refresh materialized view concurrently public.insights_retention_rollup;
    refresh materialized view concurrently public.insights_engagement_rollup;
    refresh materialized view concurrently public.insights_ai_interaction_rollup;
  $cron$
);

-- 2) Revoked-consent purge (nightly 01:00 UTC)
do $$
begin
  if exists (select 1 from cron.job where jobname = 'phase62_purge_revoked_consent') then
    perform cron.unschedule('phase62_purge_revoked_consent');
  end if;
end $$;

select cron.schedule(
  'phase62_purge_revoked_consent',
  '0 1 * * *',
  $cron$
    select public.purge_research_data_for_revoked();
  $cron$
);

commit;
```

DO NOT use `current_setting('app.service_role_key')` — neither cron calls an Edge Fn so vault bearer is NOT needed here. If a future cron job DOES call a Fn, switch to the vault bearer pattern per [[reference_supabase_pg_cron_vault_service_role_pattern]].

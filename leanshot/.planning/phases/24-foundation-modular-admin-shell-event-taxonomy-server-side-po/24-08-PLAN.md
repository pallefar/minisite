    ```bash
    #!/usr/bin/env bash
    # Phase 24 D-18..20 — per-chunk bundle-ceiling enforcement.
    # Hard-fails CI on any overage. Always prints a table of (chunk, ceiling, actual, status).
    set -euo pipefail

    DIST="${1:-dist/assets}"

    # Per-chunk ceilings (kB gz). Keep alphabetized for diff hygiene.
    declare -A CEILINGS=(
      [admin-shell]=30
      [community-feed]=20
      [course-player]=30
      [gamification-burst]=8
      [helpdesk-widget]=25
      [i18n-runtime]=15
      [index]=50
    )

    # Remediation hints surfaced on overage per CONTEXT specifics.
    declare -A HINTS=(
      [admin-shell]="Split a lazy route, defer with sync-defer.ts, or revisit ceiling at next planning cycle."
      [community-feed]="Defer feed-virtualization with sync-defer.ts; split heavy media render path."
      [course-player]="Lazy-load video player; consider dynamic import for chapter-list."
      [gamification-burst]="Move particle-animation to sync-defer.ts; drop framer-motion preset if used."
      [helpdesk-widget]="Lazy-load ticket-form; defer markdown renderer."
      [i18n-runtime]="Ship only the active locale's bundle; lazy-load others on language switch."
      [index]="Verify no static heavy-SDK imports leaked; route through sync-defer.ts per [[project_phase5_bundle_regression]]."
    )

    failed=0
    printf "%-20s %10s %10s %8s\n" "CHUNK" "CEILING_KB" "ACTUAL_KB" "STATUS"
    printf "%-20s %10s %10s %8s\n" "-----" "----------" "---------" "------"

    for chunk in "${!CEILINGS[@]}"; do
      ceiling="${CEILINGS[$chunk]}"
      # Hash-hyphen-safe regex per [[reference_bundle_budget_hash_hyphen]]:
      # match `${chunk}-<hash>.js` where hash is 8+ hex chars (Vite default).
      # The chunk name itself may contain hyphens; anchor with start-of-name + exact match up to the hyphen-hex-hash.
      files=$(find "$DIST" -maxdepth 1 -type f -regex ".*/${chunk}-[a-f0-9]\{8,\}\.js$" 2>/dev/null)
      if [ -z "$files" ]; then
        printf "%-20s %10s %10s %8s\n" "$chunk" "$ceiling" "0" "MISSING"
        # Chunks for code not yet shipped (helpdesk, i18n, etc.) are MISSING-OK, not failures.
        continue
      fi
      total_bytes=0
      for f in $files; do
        bytes=$(gzip -c "$f" | wc -c | tr -d ' ')
        total_bytes=$((total_bytes + bytes))
      done
      actual_kb=$(awk -v b="$total_bytes" 'BEGIN { printf "%.2f", b/1024 }')
      status="OK"
      if awk -v a="$actual_kb" -v c="$ceiling" 'BEGIN { exit !(a > c) }'; then
        status="OVER"
        failed=$((failed+1))
        hint="${HINTS[$chunk]:-}"
        printf "%-20s %10s %10s %8s -- OVER by %s kB. Hint: %s\n" "$chunk" "$ceiling" "$actual_kb" "$status" "$(awk -v a="$actual_kb" -v c="$ceiling" 'BEGIN { printf "%.2f", a-c }')" "$hint" >&2
      else
        printf "%-20s %10s %10s %8s\n" "$chunk" "$ceiling" "$actual_kb" "$status"
      fi
    done

    if [ "$failed" -gt 0 ]; then
      echo "FAIL: $failed chunk(s) over ceiling per D-19 (hard-fail)." >&2
      exit 1
    fi
    ```

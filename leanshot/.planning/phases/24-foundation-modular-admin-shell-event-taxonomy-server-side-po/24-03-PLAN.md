    export const ROLE_ORDER: Record<AdminRole, number> = {

      admin: 1,

    };

    Create `src/components/admin/AdminShell.tsx`:
    ```tsx
    import { Suspense, useEffect, useMemo, useState } from 'react';
    import { Route, Routes, NavLink } from 'react-router-dom'; // OR existing routing primitive — match v1.2 pattern
    import posthog from 'posthog-js';
    import { ADMIN_MODULES } from '@/lib/admin/modules';
    import { hasMinRole, type AdminRole } from '@/lib/admin/roles';
    import NotAuthorizedCard from '@/components/admin/NotAuthorizedCard';
    // ... etc

    export default function AdminShell({ adminRole }: { adminRole: AdminRole | null }) {
      const visibleModules = useMemo(() => {
        return ADMIN_MODULES.filter((m) => hasMinRole(adminRole, m.minRole) && posthog.isFeatureEnabled(m.flagKey) !== false);
      }, [adminRole]);

      // Note: posthog.isFeatureEnabled returns true | false | undefined.
      // We treat undefined (flag not yet resolved) as VISIBLE so admins aren't locked out on first paint.
      // Once /decide returns, the filter re-runs (posthog fires onFeatureFlags event).

      const [, force] = useState({});
      useEffect(() => {
        const cb = () => force({});
        posthog.onFeatureFlags(cb);
        return () => { /* posthog-js doesn't expose explicit off; cleanup via no-op */ };
      }, []);

      if (!adminRole) return <NotAuthorizedCard />;

      return (
        <div className="admin-shell">
          <nav>
            {visibleModules.map((m) => (
              <NavLink key={m.key} to={`/admin/${m.route}`}>
                <m.icon size={16} aria-hidden /> {m.label}
              </NavLink>
            ))}
          </nav>
          <main>
            <Suspense fallback={<div>Loading…</div>}>
              <Routes>
                {ADMIN_MODULES.map((m) => {
                  if (!hasMinRole(adminRole, m.minRole) || posthog.isFeatureEnabled(m.flagKey) === false) {
                    return <Route key={m.key} path={`/admin/${m.route}/*`} element={<NotAuthorizedCard />} />;
                  }
                  const LazyComp = lazyOnce(m.lazy);
                  return <Route key={m.key} path={`/admin/${m.route}/*`} element={<LazyComp />} />;
                })}
              </Routes>
            </Suspense>
          </main>
        </div>
      );
    }

    // Memoize React.lazy() per entry so module isn't re-created on re-render.
    const lazyCache = new WeakMap<() => Promise<{ default: ComponentType }>, ReturnType<typeof React.lazy>>();
    function lazyOnce(fn: () => Promise<{ default: ComponentType }>) {
      let l = lazyCache.get(fn);
      if (!l) { l = React.lazy(fn); lazyCache.set(fn, l); }
      return l;
    }
    ```

| Boundary | Description |
|----------|-------------|
| URL bar → AdminShell route handler | User can type `/admin/ai` directly without nav |
| posthog-js feature flag → module visibility | Failure mode (PostHog unreachable) defaults to allow (per design — D-02 says "failure = hidden" but we choose "default visible until /decide returns" to avoid first-paint flash; flip if reviewer rejects) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-24-01 | Elevation of privilege | URL-typed access to higher-role module | mitigate | (1) AdminShell route handler returns NotAuthorizedCard for disabled modules; (2) Every admin RPC re-checks `is_admin_at_least` (Pattern S1 dual layer) |

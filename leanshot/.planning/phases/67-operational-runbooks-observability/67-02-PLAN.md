### `leanshot/middleware.ts` — Edge Middleware fallback
If Vercel project supports Edge Middleware (Next.js convention), create:
```ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const RATE_LIMITS = {
  '/api/lead-capture': { limit: 30, windowMs: 60_000 },
  '/api/og/': { limit: 60, windowMs: 60_000 },
  '/api/affiliate-impression': { limit: 10, windowMs: 60_000 },
};

// In-memory bucket — works per-instance, NOT across regions/instances.
// For production-grade limiting, replace with Upstash Redis (deferred to Phase 70).
const buckets = new Map<string, { count: number; resetAt: number }>();

export function middleware(req: NextRequest) {
  const ip = req.headers.get('x-forwarded-for')?.split(',')[0] ?? 'unknown';
  const path = req.nextUrl.pathname;

  const matched = Object.entries(RATE_LIMITS).find(([prefix]) => path.startsWith(prefix));
  if (!matched) return NextResponse.next();

  const [prefix, { limit, windowMs }] = matched;
  const key = `${ip}:${prefix}`;
  const now = Date.now();
  const bucket = buckets.get(key) ?? { count: 0, resetAt: now + windowMs };
  if (now > bucket.resetAt) { bucket.count = 0; bucket.resetAt = now + windowMs; }
  bucket.count++;
  buckets.set(key, bucket);

  if (bucket.count > limit) {
    return new NextResponse('Too Many Requests', {
      status: 429,
      headers: { 'Retry-After': Math.ceil((bucket.resetAt - now) / 1000).toString() },
    });
  }
  return NextResponse.next();
}

export const config = {
  matcher: ['/api/lead-capture', '/api/og/:path*', '/api/affiliate-impression'],
};
```

```typescript
export async function signInWithOAuthProvider(
  provider: 'google' | 'apple',
): Promise<{ error: { message: string } | null }> {
  if (provider === 'apple' && !isAppleEnabled()) {
    return { error: { message: 'apple_disabled' } };
  }
  const { error } = await supabase.auth.signInWithOAuth({
    provider,
    options: { redirectTo: `${window.location.origin}/auth/callback` },
  });
  return { error: error ? { message: error.message } : null };
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| iOS native dialog → app | Apple's ASAuthorization framework mints the identityToken; the app receives a signed JWT it must not trust blindly |
| app → Supabase GoTrue | identityToken passed to signInWithIdToken; GoTrue verifies the JWT signature against Apple's public keys server-side |
| npm registry → build | New third-party package enters the dependency tree |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-59-04 | Tampering (token integrity) | signInWithAppleNative identityToken | mitigate | Token is passed to `supabase.auth.signInWithIdToken` which verifies the Apple JWT signature, exp, and nonce server-side (GoTrue). The client NEVER asserts identity itself or hand-rolls JWT verification (RESEARCH Don't-Hand-Roll) |
| T-59-05 | Tampering (replay) | old/expired identityToken | mitigate | A fresh `nonce: crypto.randomUUID()` is sent per authorize() call; GoTrue rejects expired tokens via exp validation |
| T-59-06 | Information Disclosure | private-relay email | accept | Apple private-relay email lands in auth.users.email only; profiles has no email column (handle_new_user keys on id). No PII assumption added; nothing stored from result.response.email |
| T-59-07 | Spoofing | unconfigured-provider half-auth | mitigate | Native path is reachable ONLY when isAppleEnabled()===true AND detectPlatform()==='ios'; both must hold or the call short-circuits, so an unconfigured provider cannot half-authenticate |
| T-59-SC | Tampering (supply chain) | @capacitor-community/apple-sign-in install | mitigate | Legitimacy verified at PLAN time via registry JSON API ([ASSUMED]→approved: official Capacitor Community org, no install scripts, peerDep compatible) and re-confirmed autonomously in Task 1 before install; STOP-on-discrepancy guard if repo/maintainer/scripts changed |

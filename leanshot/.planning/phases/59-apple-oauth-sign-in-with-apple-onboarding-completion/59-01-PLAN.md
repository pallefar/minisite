```typescript
export function isAppleEnabled(): boolean; // reads VITE_AUTH_APPLE_ENABLED or localStorage; false by default. Called at render time (non-reactive — that's fine).
export async function signInWithOAuthProvider(
  provider: 'google' | 'apple',
): Promise<{ error: { message: string } | null }>; // returns { error: { message: 'apple_disabled' } } if Apple gated off (no Supabase call made)
```

Reference render pattern already shipped in ConsumerOnboardingRenderer.tsx (lines 170-176, 326-335):
```tsx
const onOAuth = async (provider: 'google' | 'apple'): Promise<void> => {
  setSubmitting(true);
  setAuthMessage(null);
  const { error } = await signInWithOAuthProvider(provider);
  setSubmitting(false);
  if (error) setAuthMessage(error.message);
};
// render:
{isAppleEnabled() && (
  <Button variant="ghost" onClick={() => void onOAuth('apple')} disabled={submitting} className="min-h-[44px] w-full">
    {t('onboarding:consumer.auth.sign_in_apple')}
  </Button>
)}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser → Supabase GoTrue | User initiates OAuth; the Apple PKCE redirect + token exchange happen server-side in GoTrue (client never holds Apple credentials) |
| browser → i18n catalog | Static locale JSON served to client; no user input crosses it |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-59-01 | Information Disclosure | signInWithOAuthProvider when Apple unconfigured | mitigate | Gate returns `{ error: { message: 'apple_disabled' } }` and makes NO Supabase call — no GoTrue config state leaks (existing Phase 34 behavior; this plan only adds the call-site behind `isAppleEnabled()`) |
| T-59-02 | Spoofing (redirect hijack) | OAuth redirectTo | mitigate | `signInWithOAuthProvider` hardcodes `redirectTo: ${window.location.origin}/auth/callback` (app-owned origin only); this plan does not alter that |
| T-59-03 | Elevation/Tampering | Apple button in promote/anon-completion mode | mitigate | Button gated on `!isPromote` / `!isAnon` so a credential-completion flow cannot be diverted into a fresh OAuth identity |
| T-59-SC | Tampering | npm installs | mitigate | This plan installs NO packages (web path uses already-installed @supabase/supabase-js); package-install legitimacy gate is owned by Plan 02 |
</threat_model>

**`KBSearchTypeahead.tsx`** — controlled input + debounced RPC:
```typescript
import { useEffect, useState, useDeferredValue } from 'react';
import { Input } from '@/components/ui/Input';
import { supabase } from '@/lib/supabase';
import { track } from '@/lib/analytics/events';
import type { TabId } from '@/types';

type Result = { id: string; slug: string; title: string; locale: string; rank: number };

export default function KBSearchTypeahead({ onSelectArticle }: { onSelectArticle: (id: string) => void }) {
  const [query, setQuery] = useState('');
  const [locale, setLocale] = useState<'en' | 'es'>('en');
  const [results, setResults] = useState<Result[]>([]);
  const [loading, setLoading] = useState(false);
  const deferredQuery = useDeferredValue(query);

  useEffect(() => {
    if (deferredQuery.trim().length < 2) { setResults([]); return; }
    const handle = setTimeout(async () => {
      setLoading(true);
      const { data, error } = await supabase.rpc('search_kb_articles', {
        p_query: deferredQuery, p_locale: locale, p_limit: 10,
      });
      if (!error && data) {
        setResults(data as Result[]);
        track('helpdesk.kb_search.performed', { query_length: deferredQuery.length, results_count: data.length, locale });
      }
      setLoading(false);
    }, 250);
    return () => clearTimeout(handle);
  }, [deferredQuery, locale]);

  return (
    <div className="flex flex-col gap-2">
      <div className="flex items-center gap-2">
        <Input aria-label="Search the knowledge base" placeholder="Search help articles…" value={query} onChange={(e) => setQuery(e.target.value)} />
        <button aria-pressed={locale === 'en'} onClick={() => setLocale('en')} className="text-xs px-2 py-1">EN</button>
        <button aria-pressed={locale === 'es'} onClick={() => setLocale('es')} className="text-xs px-2 py-1">ES</button>
      </div>
      <ul role="region" aria-live="polite" aria-busy={loading} className="flex flex-col gap-1">
        {results.map((r) => (
          <li key={r.id}>
            <button onClick={() => { track('helpdesk.kb_article.viewed', { article_id: r.id, slug: r.slug, locale: r.locale }); onSelectArticle(r.id); }} className="text-left w-full p-2 hover:bg-accent">
              {r.title}
            </button>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

**`KBArticleView.tsx`** — markdown render with hard sanitization:
```typescript
import { useEffect, useState } from 'react';
import ReactMarkdown from 'react-markdown';
import remarkGfm from 'remark-gfm';
import rehypeRaw from 'rehype-raw';
import DOMPurify from 'dompurify';
import { Button } from '@/components/ui/Button';
import { supabase } from '@/lib/supabase';

type Article = { id: string; slug: string; title: string; body: string; title_es: string | null; body_es: string | null; locale_set: string[] };

export default function KBArticleView({ articleId, onBack }: { articleId: string; onBack: () => void }) {
  const [article, setArticle] = useState<Article | null>(null);
  const [locale, setLocale] = useState<'en' | 'es'>('en');

  useEffect(() => {
    void supabase.from('kb_articles').select('id, slug, title, body, title_es, body_es, locale_set').eq('id', articleId).single()
      .then(({ data }) => setArticle((data ?? null) as Article | null));
  }, [articleId]);

  if (!article) return <div className="p-4 text-sm">Loading article…</div>;

  const title = locale === 'es' ? (article.title_es ?? article.title) : article.title;
  const body  = locale === 'es' ? (article.body_es  ?? article.body)  : article.body;
  const sanitized = DOMPurify.sanitize(body, { USE_PROFILES: { html: true } });

  return (
    <div className="flex flex-col gap-3 p-3">
      <Button onClick={onBack} variant="ghost" aria-label="Back to search">← Back</Button>
      <h2 className="text-lg font-semibold">{title}</h2>
      {article.locale_set?.includes('es') && (
        <div className="flex gap-2 text-xs">
          <button aria-pressed={locale === 'en'} onClick={() => setLocale('en')}>EN</button>
          <button aria-pressed={locale === 'es'} onClick={() => setLocale('es')}>ES</button>
        </div>
      )}
      <article className="prose prose-sm max-w-none">
        <ReactMarkdown remarkPlugins={[remarkGfm]} rehypePlugins={[rehypeRaw]}>
          {sanitized}
        </ReactMarkdown>
      </article>
    </div>
  );
}
```

```typescript
type Message = { id: string; ticket_id: string; author_kind: string; body: string; created_at: string };

export function useTicketChannel(ticketId: string, currentUserId: string) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [typingUserIds, setTypingUserIds] = useState<string[]>([]);
  const channelRef = useRef<RealtimeChannel | null>(null);
  const typingTimeouts = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map());

  useEffect(() => {
    if (!ticketId) return;
    let cancelled = false;
    const setup = async () => {
      const channel = supabase.channel(`ticket:${ticketId}`, { config: { broadcast: { self: false }}});
      channel.on('broadcast', { event: 'typing' }, (payload) => {
        const userId = String(payload.payload?.userId ?? '');
        if (!userId || userId === currentUserId) return;
        setTypingUserIds((prev) => prev.includes(userId) ? prev : [...prev, userId]);
        const existing = typingTimeouts.current.get(userId);
        if (existing) clearTimeout(existing);
        typingTimeouts.current.set(userId, setTimeout(() => {
          setTypingUserIds((prev) => prev.filter((u) => u !== userId));
          typingTimeouts.current.delete(userId);
        }, 3000));
      });
      channel.on('postgres_changes', {
        event: 'INSERT', schema: 'public', table: 'ticket_messages',
        filter: `ticket_id=eq.${ticketId}`,
      }, (payload) => {
        setMessages((prev) => [...prev, payload.new as Message]);
      });
      await supabase.realtime.setAuth();   // INVARIANT — Phase 9 pattern
      if (cancelled) return;
      channel.subscribe();
      channelRef.current = channel;
    };
    void setup();
    // Initial backfill
    void supabase.from('ticket_messages').select('id, ticket_id, author_kind, body, created_at').eq('ticket_id', ticketId).order('created_at', { ascending: true })
      .then(({ data }) => { if (!cancelled && data) setMessages(data as Message[]); });
    return () => {
      cancelled = true;
      const c = channelRef.current; if (c) supabase.removeChannel(c);
      for (const t of typingTimeouts.current.values()) clearTimeout(t);
      typingTimeouts.current.clear();
    };
  }, [ticketId, currentUserId]);

  // Throttled typing broadcast (500ms)
  const lastSent = useRef(0);
  const sendTyping = () => {
    const now = Date.now();
    if (now - lastSent.current < 500) return;
    lastSent.current = now;
    channelRef.current?.send({ type: 'broadcast', event: 'typing', payload: { userId: currentUserId, isTyping: true }});
  };

  return { messages, typingUserIds, sendTyping };
}
```

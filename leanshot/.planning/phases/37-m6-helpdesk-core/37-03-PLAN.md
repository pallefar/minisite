**Outcome enum + helper:**
```typescript
type Outcome =
  | 'ticket_created'
  | 'reply_appended'
  | 'unknown_sender_bounced'
  | 'hmac_failed_bounced'
  | 'duplicate'
  | 'rejected_loop';

async function logInboundEvent(args: { resendEmailId: string; from: string; to: string; outcome: Outcome; ticketId?: string | null }) {
  await admin.from('ticket_inbound_events').insert({
    resend_email_id: args.resendEmailId,
    ticket_id: args.ticketId ?? null,
    from_email: args.from,
    to_email: args.to,
    outcome: args.outcome,
  });
}
```

3. **Svix signature verify** with replay-protection timestamp window (±5 minutes — Svix default):
```typescript
const rawBody = await req.text();
const wh = new Webhook(RESEND_WEBHOOK_SECRET);
let event: any;
try {
  event = wh.verify(rawBody, {
    'svix-id':        req.headers.get('svix-id') ?? '',
    'svix-timestamp': req.headers.get('svix-timestamp') ?? '',
    'svix-signature': req.headers.get('svix-signature') ?? '',
  });
} catch { return new Response('invalid signature', { status: 401 }); }
if (event?.type !== 'email.received') return new Response('ignored', { status: 200 });
```

7. **Branch — `reply+<token>@app.leanshot.app`:**
```typescript
const replyMatch = toEmail.match(/^reply\+([a-z0-9_-]+)@app\.leanshot\.app$/);
if (replyMatch) {
  const token = replyMatch[1]!;
  // We don't yet know ticketId/userId — must look up by trial verification against the sender.
  // Strategy: find candidate ticket(s) for sender email; verify token against each candidate.
  const { data: userRow } = await admin.from('profiles').select('id').eq('email', fromEmail).maybeSingle();
  if (!userRow) {
    // sender unknown → bounce + log
    await sendEmail(admin, { template: 'helpdesk_unknown_sender', to: fromEmail, vars: { signup_url: SIGNUP_URL }, phi: false });
    await logInboundEvent({ resendEmailId: emailId, from: fromEmail, to: toEmail, outcome: 'unknown_sender_bounced' });
    return new Response('unknown-sender', { status: 200 });
  }
  const { data: candidates } = await admin
    .from('tickets').select('id').eq('user_id', userRow.id)
    .order('created_at', { ascending: false }).limit(50);
  let matchedTicketId: string | null = null;
  for (const c of (candidates ?? [])) {
    if (await verifyReplyToken(token, HMAC_SECRET, c.id, userRow.id)) { matchedTicketId = c.id; break; }
  }
  if (!matchedTicketId) {
    // HMAC fail → bounce
    await sendEmail(admin, { template: 'helpdesk_unknown_sender', to: fromEmail, vars: { signup_url: SIGNUP_URL }, phi: false });
    await logInboundEvent({ resendEmailId: emailId, from: fromEmail, to: toEmail, outcome: 'hmac_failed_bounced' });
    return new Response('hmac-failed', { status: 200 });
  }
  // Append message; reopen ticket if closed.
  const { data: msg } = await admin.from('ticket_messages').insert({
    ticket_id: matchedTicketId, author_user_id: userRow.id,
    author_kind: 'user', body, via: 'email', resend_email_id: emailId,
  }).select('id').single();
  await admin.from('tickets').update({ status: 'open', updated_at: new Date().toISOString(), last_user_message_at: new Date().toISOString() })
    .eq('id', matchedTicketId).in('status', ['waiting_on_customer','resolved','closed']);
  await ingestAttachments(email.attachments ?? [], matchedTicketId, msg!.id, /* phi */ false);
  await logInboundEvent({ resendEmailId: emailId, from: fromEmail, to: toEmail, outcome: 'reply_appended', ticketId: matchedTicketId });
  captureServer({ event: 'helpdesk.inbound_email.received', distinctId: userRow.id, properties: { outcome: 'reply_appended', ticket_id: matchedTicketId }});
  return new Response(JSON.stringify({ ok: true, outcome: 'reply_appended' }), { status: 200, headers: corsHeaders });
}
```

**Attachment ingest helper (inline in same file, scoped to closure of `admin`):**
```typescript
const ALLOWED_MIME = new Set(['image/png','image/jpeg','image/gif','image/webp','application/pdf','text/plain','text/csv']);
async function ingestAttachments(
  list: Array<{ filename: string; content_type: string; size: number; download_url: string }>,
  ticketId: string,
  messageId: string,
  phi: boolean,
): Promise<void> {
  for (const att of list.slice(0, 10)) {
    if (!ALLOWED_MIME.has(att.content_type)) { console.warn('attachment-mime-rejected'); continue; }
    if (att.size > 10_485_760)                { console.warn('attachment-size-rejected'); continue; }
    const fileRes = await fetch(att.download_url, { headers: { Authorization: `Bearer ${RESEND_API_KEY}` }});
    if (!fileRes.ok) { console.warn('attachment-fetch-failed'); continue; }
    const buf = new Uint8Array(await fileRes.arrayBuffer());
    const storagePath = `${ticketId}/${messageId}/${crypto.randomUUID()}-${att.filename.replace(/[^a-zA-Z0-9._-]/g, '_').slice(0,80)}`;
    const up = await admin.storage.from('ticket-attachments').upload(storagePath, buf, { contentType: att.content_type, upsert: false });
    if (up.error) { console.warn('attachment-upload-failed', up.error.message); continue; }
    await admin.from('ticket_attachments').insert({
      ticket_id: ticketId, message_id: messageId,
      storage_path: storagePath, mime_type: att.content_type,
      byte_size: att.size, original_filename: att.filename,
      // TODO [DEFERRED v1.4]: ClamAV scan — see .planning/phases/37-m6-helpdesk-core/deferred-items.md
      scan_status: 'deferred',
    });
    captureServer({ event: 'helpdesk.attachment.scan_deferred', distinctId: ticketId, properties: { mime: att.content_type, size: att.size, phi }});
  }
}
```

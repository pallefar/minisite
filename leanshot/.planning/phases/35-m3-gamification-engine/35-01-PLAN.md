**Function 1 — public.compute_level(xp_total int) returns int:**

```sql
create or replace function public.compute_level(xp_total int)
returns int
language sql
immutable
parallel safe
set search_path = public, pg_catalog
as $body$
  -- D-02: Level N requires N² × 100 XP cumulative; level = floor(sqrt(xp/100))
  -- D-04: Pure function — no now(), no current_setting, no auth.uid().
  select greatest(1, floor(sqrt(greatest(xp_total, 0)::float / 100))::int);
$body$;

comment on function public.compute_level(int) is
  'Phase 35 D-04 — deterministic pure SQL level computation. '
  'Level N requires N²×100 XP cumulative (D-02). Rollback test in supabase/tests/35_xp_ledger_replay.sql '
  'exercises with random sequences; result MUST be identical regardless of insert order.';

revoke all on function public.compute_level(int) from public;
grant execute on function public.compute_level(int) to authenticated, service_role;
```

Use named tag `$body$` (NOT bare `$$`) per memory reference_postgres_dollar_quote_nesting_in_cron_body — even though this body is short, the project convention is named tags throughout for grep-ability.

`IMMUTABLE` (not STABLE) is load-bearing per memory reference_supabase_migration_gotchas: allows the fn to be used in partial-index predicates and in parallel query plans (needed by the replay test against 100k+ rows).

**`supabase/tests/35_xp_ledger_replay.sql`** — D-04 determinism contract (THE load-bearing test for this phase):

```sql
begin;
select plan(8);

-- Setup: insert 50 random ledger rows for two synthetic users in different orders.
-- Sequences MUST produce identical level results because compute_level is pure.
do $body$
declare
  uid_a uuid := gen_random_uuid();
  uid_b uuid := gen_random_uuid();  -- same XP deltas, different insert order
  deltas int[] := array[25, 25, 10, 50, 100, 25, 25, 250, 25, 50,
                         25, 10, 50, 100, 25, 25, 10, 25, 250, 25,
                         50, 25, 100, 25, 10];  -- 25-row total = 1250 XP
  i int;
begin
  -- Insert in forward order for user A
  for i in 1..array_length(deltas, 1) loop
    insert into public.xp_ledger (user_id, action_type, xp_delta)
      values (uid_a, 'injection_log', deltas[i]);
  end loop;
  -- Insert in reverse order for user B
  for i in reverse array_length(deltas, 1)..1 loop
    insert into public.xp_ledger (user_id, action_type, xp_delta)
      values (uid_b, 'injection_log', deltas[i]);
  end loop;
  perform set_config('p35_test.uid_a', uid_a::text, true);
  perform set_config('p35_test.uid_b', uid_b::text, true);
end $body$;

select is(
  public.compute_level(public.xp_total_for(current_setting('p35_test.uid_a')::uuid)),
  public.compute_level(public.xp_total_for(current_setting('p35_test.uid_b')::uuid)),
  'D-04: ledger replay yields identical level regardless of insert order'
);

select is(public.compute_level(0),      1::int, 'compute_level(0) = 1');
select is(public.compute_level(100),    1::int, 'compute_level(100) = 1 (level 1 boundary)');
select is(public.compute_level(400),    2::int, 'compute_level(400) = 2 (D-02)');
select is(public.compute_level(2500),   5::int, 'compute_level(2500) = 5 (D-02)');
select is(public.compute_level(10000),  10::int, 'compute_level(10000) = 10 (D-02)');
select is(public.compute_level(40000),  20::int, 'compute_level(40000) = 20 (D-02)');
select is(public.compute_prestige(1000000), 1::int, 'compute_prestige caps display at 100 + prestige tier (D-03)');

select * from finish();
rollback;
```

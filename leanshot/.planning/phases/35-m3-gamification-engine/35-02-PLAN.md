evaluate_streak_for_user SECDEF function:

```sql
create or replace function public.evaluate_streak_for_user(p_user uuid)
returns void
language plpgsql
security definer
set search_path = public, pg_catalog
as $body$
declare
  v_tz text;
  v_today_local date;
  v_yesterday_local date;
  v_has_action_yesterday boolean;
  v_state record;
  v_freeze_balance int;
begin
  select timezone into v_tz from public.profiles where id = p_user;
  if v_tz is null then v_tz := 'UTC'; end if;

  v_today_local := (now() at time zone v_tz)::date;
  v_yesterday_local := v_today_local - 1;

  -- Idempotency: if we've already evaluated this user for today, no-op.
  -- This handles DST double-fire (Pitfall 2) and cron-retry safely.
  select * into v_state from public.streak_state where user_id = p_user;
  if v_state.last_eval_date is not null and v_state.last_eval_date >= v_today_local then
    return;
  end if;

  -- D-06 cross-action: ANY qualifying XP action yesterday keeps streak.
  select exists(
    select 1 from public.xp_ledger
     where user_id = p_user
       and action_type in ('injection_log','weight_log','symptom_log','workout_log')
       and (created_at at time zone v_tz)::date = v_yesterday_local
  ) into v_has_action_yesterday;

  if v_has_action_yesterday then
    -- Survived yesterday: increment.
    insert into public.streak_state (user_id, current_streak_days, longest_streak_days, last_action_at, last_eval_date, updated_at)
    values (p_user, 1, 1, now(), v_today_local, now())
    on conflict (user_id) do update set
      current_streak_days = streak_state.current_streak_days + 1,
      longest_streak_days = greatest(streak_state.longest_streak_days, streak_state.current_streak_days + 1),
      last_action_at = now(),
      last_eval_date = v_today_local,
      updated_at = now();
  else
    -- Missed yesterday: try auto-apply freeze token before breaking.
    if coalesce(v_state.current_streak_days, 0) > 0 then
      v_freeze_balance := public.freeze_tokens_remaining(p_user);
      if v_freeze_balance >= 1 then
        -- Consume one token; keep streak.
        insert into public.freeze_tokens_ledger (user_id, delta, reason, source_ref)
        values (p_user, -1, 'auto_apply', 'streak:' || v_yesterday_local::text);
        insert into public.streak_state (user_id, current_streak_days, longest_streak_days, last_action_at, last_eval_date, updated_at)
        values (p_user, coalesce(v_state.current_streak_days, 0), coalesce(v_state.longest_streak_days, 0), v_state.last_action_at, v_today_local, now())
        on conflict (user_id) do update set
          last_eval_date = v_today_local,
          updated_at = now();
        -- Plan 35-09 wires the user-facing "freeze auto-applied" notification.
      else
        -- No tokens; break streak.
        insert into public.streak_state (user_id, current_streak_days, longest_streak_days, last_action_at, last_eval_date, updated_at)
        values (p_user, 0, coalesce(v_state.longest_streak_days, 0), v_state.last_action_at, v_today_local, now())
        on conflict (user_id) do update set
          current_streak_days = 0,
          last_eval_date = v_today_local,
          updated_at = now();
      end if;
    else
      -- No prior streak; just record the eval.
      insert into public.streak_state (user_id, current_streak_days, longest_streak_days, last_eval_date, updated_at)
      values (p_user, 0, 0, v_today_local, now())
      on conflict (user_id) do update set last_eval_date = v_today_local, updated_at = now();
    end if;
  end if;
end;
$body$;

revoke all on function public.evaluate_streak_for_user(uuid) from public;
grant execute on function public.evaluate_streak_for_user(uuid) to service_role;
```

Use named `$body$` tag. Note `freeze_tokens_remaining` is created in the NEXT migration; Postgres resolves it at function-call time, not at function-create time (deferred resolution is fine).

**`20270708000007_p35_freeze_monthly_grant_cron.sql`** — monthly +1 grant at month boundary.

```sql
create or replace function public.grant_monthly_freeze_tokens()
returns void
language plpgsql
security definer
set search_path = public, pg_catalog
as $body$
declare r record; v_month_key text;
begin
  v_month_key := to_char(now() at time zone 'UTC', 'YYYY-MM');
  for r in
    select id as user_id from public.profiles where id is not null
  loop
    begin
      -- D-08: grant +1 free monthly; UNIQUE INDEX on (user_id, reason, source_ref) prevents double-grant
      -- on cron retry. ON CONFLICT DO NOTHING for idempotency.
      insert into public.freeze_tokens_ledger (user_id, delta, reason, source_ref)
      values (r.user_id, 1, 'monthly_grant', v_month_key)
      on conflict (user_id, reason, source_ref) do nothing;
      -- D-08 max stockpile 3: enforced by freeze_tokens_remaining clamp at read time.
      -- Overflow rows still get inserted (audit trail), they just don't count toward usable balance.
      if (select sum(delta) from public.freeze_tokens_ledger where user_id = r.user_id) > 3 then
        raise notice 'phase35-freeze-monthly-grant: user % already at cap, grant logged but capped at 3', r.user_id;
      end if;
    exception when others then
      raise notice 'phase35-freeze-monthly-grant: user % error % — continuing', r.user_id, sqlerrm;
    end;
  end loop;
end;
$body$;

revoke all on function public.grant_monthly_freeze_tokens() from public;
grant execute on function public.grant_monthly_freeze_tokens() to service_role;

do $unschedule$
declare job_name text;
begin
  for job_name in select jobname from cron.job where jobname = 'phase35-freeze-monthly-grant' loop
    perform cron.unschedule(job_name);
  end loop;
exception when others then null;
end $unschedule$;

select cron.schedule(
  'phase35-freeze-monthly-grant',
  '15 0 1 * *',                        -- 1st of month, 00:15 UTC
  $cron$
  do $grant$ begin
    perform public.grant_monthly_freeze_tokens();
  exception when others then
    raise notice 'phase35-freeze-monthly-grant: outer error % — continuing', sqlerrm;
  end $grant$;
  $cron$
);
```

Both crons use named-tag dollar-quoting throughout. Cron expression for monthly: `15 0 1 * *` (1st of month, 00:15 UTC — avoids midnight pile-up).

```ts
  try {
    // 1. User-JWT auth
    const auth = req.headers.get('authorization') ?? '';
    const jwt = auth.toLowerCase().startsWith('bearer ') ? auth.slice(7) : '';
    if (!jwt) return jsonError(401, 'unauthenticated');

    const a = admin();
    const { data: { user } } = await a.auth.getUser(jwt);
    if (!user) return jsonError(401, 'unauthenticated');

    // 2. Admin role check (server-side mirror of surfaceCheck per D-10 audit + memory feedback_admin_module_manifest_vs_router_branch_drift)
    const { data: profile } = await a.from('profiles').select('admin_role').eq('id', user.id).maybeSingle();
    if (!profile?.admin_role || !['support_admin','support_lead','superadmin'].includes(profile.admin_role)) {
      return jsonError(403, 'forbidden_not_admin');
    }

    // 3. Body validation
    const raw = await req.json().catch(() => null);
    const parsed = BodySchema.safeParse(raw);
    if (!parsed.success) return jsonError(400, 'invalid_body');
    const { target_user_id, delta, reason_note } = parsed.data;

    // 4. Insert freeze_tokens_ledger row with granted_by_admin_user_id audit per D-10
    const { error: insErr } = await a.from('freeze_tokens_ledger').insert({
      user_id: target_user_id,
      delta,
      reason: 'admin_grant',
      granted_by_admin_user_id: user.id,
      source_ref: `admin_grant:${reason_note.slice(0, 80)}`,
    });
    if (insErr) {
      console.error('[admin-grant-freeze-token] insert failed', insErr);
      return jsonError(500, 'db_error');
    }

    // 5. Server-side PostHog capture per D-05
    captureServer({
      distinctId: user.id,
      event: 'freeze_token_granted',
      properties: { target_user_id, delta, reason: 'admin_grant', granted_by_admin_user_id: user.id },
    });

    return jsonResponse(200, { ok: true });
  } catch (e) {
    console.error('[admin-grant-freeze-token] handler error', e);
    return jsonError(500, 'internal_error');
  } finally {
    try { await shutdownPostHog(); } catch (e) { console.error('[admin-grant-freeze-token] shutdown failed', e); }
  }
});
```

**`leanshot/src/lib/gamification-defer.ts`** — mirror `leanshot/src/lib/sync-defer.ts` shape. Read sync-defer.ts first to understand the exact FIFO buffer + idle-callback + dynamic-import pattern.

```ts
// Mirrors leanshot/src/lib/sync-defer.ts (Phase 6 D-12 idle-deferred-init wrapper)
// Type-only imports keep canvas-confetti / framer-motion OUT of entry chunk (Pitfall 12)
import type confetti from 'canvas-confetti';

type GamificationCall =
  | { kind: 'levelUpBurst'; level: number }
  | { kind: 'challengeCompleteBurst'; rewardKind: 'xp' | 'badge' | 'freeze' | 'combo' };

const BUFFER_CAP = 16;
const buffer: GamificationCall[] = [];
const COOLDOWN_KEY = 'leanshot_confetti_cooldown_until';
const COOLDOWN_MS = 60_000;  // Pitfall 11 + CONTEXT discretion: 1 burst per 60s

let loadedApi: { confetti: typeof confetti } | null = null;
let loadingPromise: Promise<typeof loadedApi> | null = null;

async function loadGamification(): Promise<typeof loadedApi> {
  if (loadedApi) return loadedApi;
  if (loadingPromise) return loadingPromise;
  loadingPromise = (async () => {
    const mod = await import('canvas-confetti');
    loadedApi = { confetti: mod.default };
    return loadedApi;
  })();
  return loadingPromise;
}

function isOnCooldown(): boolean {
  try {
    const until = Number(localStorage.getItem(COOLDOWN_KEY) ?? 0);
    return Date.now() < until;
  } catch { return false; }
}

function setCooldown(): void {
  try { localStorage.setItem(COOLDOWN_KEY, String(Date.now() + COOLDOWN_MS)); } catch {}
}

function fireBurst(api: NonNullable<typeof loadedApi>, call: GamificationCall): void {
  try {
    // Brand color from CSS variable — anti-pattern guard (do NOT hardcode)
    const root = document.documentElement;
    const primary = getComputedStyle(root).getPropertyValue('--color-primary').trim() || '#22c55e';
    const accent = getComputedStyle(root).getPropertyValue('--color-accent').trim() || '#f59e0b';
    api.confetti({
      particleCount: 80,
      spread: 60,
      origin: { y: 0.6 },
      colors: [primary, accent],
      disableForReducedMotion: true,  // defense-in-depth #2 (library-level gate)
    });
  } catch (e) {
    console.error('[gamification-defer] burst failed', e);
  }
}

function dispatch(api: NonNullable<typeof loadedApi>, call: GamificationCall): void {
  // Caller MUST have already verified !reducedMotion at React level (defense-in-depth #1)
  if (isOnCooldown()) return;
  setCooldown();
  fireBurst(api, call);
}

export function fireLevelUpBurst(level: number): void {
  const call: GamificationCall = { kind: 'levelUpBurst', level };
  if (loadedApi) { dispatch(loadedApi, call); return; }
  if (buffer.length >= BUFFER_CAP) buffer.shift();
  buffer.push(call);
  void loadGamification().then((api) => {
    if (!api) return;
    while (buffer.length > 0) {
      const next = buffer.shift();
      if (next) dispatch(api, next);
    }
  });
}

export function fireChallengeCompleteBurst(rewardKind: 'xp' | 'badge' | 'freeze' | 'combo'): void {
  const call: GamificationCall = { kind: 'challengeCompleteBurst', rewardKind };
  if (loadedApi) { dispatch(loadedApi, call); return; }
  if (buffer.length >= BUFFER_CAP) buffer.shift();
  buffer.push(call);
  void loadGamification().then((api) => {
    if (!api) return;
    while (buffer.length > 0) {
      const next = buffer.shift();
      if (next) dispatch(api, next);
    }
  });
}
```

```tsx
import { useEffect } from 'react';
import { useReducedMotion } from '@/hooks/useReducedMotion';
import { fireLevelUpBurst, fireChallengeCompleteBurst } from '@/lib/gamification-defer';

interface ConfettiBurstProps {
  trigger: 'level-up' | 'challenge-complete' | null;
  level?: number;
  rewardKind?: 'xp' | 'badge' | 'freeze' | 'combo';
}

export function ConfettiBurst({ trigger, level, rewardKind }: ConfettiBurstProps): null {
  const reduced = useReducedMotion();
  useEffect(() => {
    if (reduced) return;             // defense-in-depth #1: React-level gate
    if (!trigger) return;
    if (trigger === 'level-up') fireLevelUpBurst(level ?? 1);
    else fireChallengeCompleteBurst(rewardKind ?? 'xp');
  }, [reduced, trigger, level, rewardKind]);
  return null;
}
```

**`leanshot/src/lib/gamification/dashboard-data.ts`** — aggregated SECDEF RPC reader used by all 4 cards. Single hook + single round trip = cleaner card render.

```ts
import { supabase } from '@/lib/supabase';
import { computeLevel, xpToNextLevel, computePrestige } from '@/lib/gamification/xp';

export interface GamificationDashboardData {
  xpTotal: number;
  level: number;
  prestige: number;
  xpToNext: number;
  xpInLevel: number;
  nextLevelXp: number;
  streak: { current: number; longest: number; lastActionAt: string | null };
  freezeTokens: number;
}

export async function fetchGamificationDashboard(userId: string): Promise<GamificationDashboardData> {
  // 4 parallel reads — all SECDEF helpers from Plans 35-01 / 35-02
  const [xpTotal, freeze, streak] = await Promise.all([
    supabase.rpc('xp_total_for', { p_user: userId }).then((r) => Number(r.data ?? 0)),
    supabase.rpc('freeze_tokens_remaining', { p_user: userId }).then((r) => Number(r.data ?? 0)),
    supabase.from('streak_state').select('current_streak_days, longest_streak_days, last_action_at')
      .eq('user_id', userId).maybeSingle().then((r) => r.data),
  ]);
  const level = computeLevel(xpTotal);
  const prestige = computePrestige(xpTotal);
  const { xpInLevel, xpToNext, nextLevelXp } = xpToNextLevel(xpTotal);
  return {
    xpTotal,
    level,
    prestige,
    xpToNext,
    xpInLevel,
    nextLevelXp,
    streak: {
      current: streak?.current_streak_days ?? 0,
      longest: streak?.longest_streak_days ?? 0,
      lastActionAt: streak?.last_action_at ?? null,
    },
    freezeTokens: freeze,
  };
}
```

**`leanshot/src/components/dashboard/burst/LevelUpBurst.tsx`** — framer-motion overlay with reduced-motion fallback:

```tsx
import { motion, AnimatePresence } from 'framer-motion';
import { useEffect, useState } from 'react';
import { useReducedMotion } from '@/hooks/useReducedMotion';
import { ConfettiBurst } from './ConfettiBurst';
import { fireXpEvent } from '@/lib/gamification/xp-event-client';

interface LevelUpBurstProps {
  newLevel: number | null;            // null = no level-up; otherwise show overlay
  onDismiss: () => void;
}

export function LevelUpBurst({ newLevel, onDismiss }: LevelUpBurstProps) {
  const reduced = useReducedMotion();
  const [confettiTrigger, setConfettiTrigger] = useState<'level-up' | null>(null);

  useEffect(() => {
    if (newLevel === null) return;
    setConfettiTrigger('level-up');
    // Fire server-side PostHog event per D-05
    fireXpEvent('level_up', { level: newLevel });
    // Auto-dismiss after 5s
    const t = setTimeout(onDismiss, 5_000);
    return () => clearTimeout(t);
  }, [newLevel, onDismiss]);

  if (newLevel === null) return null;

  return (
    <>
      <ConfettiBurst trigger={confettiTrigger} level={newLevel} />
      <AnimatePresence>
        <motion.div
          role="dialog"
          aria-modal="true"
          aria-label={`Level ${newLevel} unlocked`}
          initial={reduced ? { opacity: 0 } : { opacity: 0, scale: 0.8 }}
          animate={reduced ? { opacity: 1 } : { opacity: 1, scale: 1 }}
          exit={reduced ? { opacity: 0 } : { opacity: 0, scale: 0.95 }}
          transition={{ duration: reduced ? 0.1 : 0.4, ease: 'easeOut' }}
          className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur"
          onClick={onDismiss}
        >
          <div className="bg-surface-elevated rounded-2xl p-8 shadow-xl text-center motion-safe:animate-pulse">
            <div className="text-sm text-secondary uppercase tracking-wide">Level Up</div>
            <div className="text-7xl font-bold text-primary mt-2">{newLevel}</div>
            <button
              type="button"
              onClick={onDismiss}
              className="mt-4 text-sm text-secondary hover:text-primary"
            >
              Dismiss
            </button>
          </div>
        </motion.div>
      </AnimatePresence>
    </>
  );
}
```

```tsx
import { useEffect, useState } from 'react';
import { Card, CardHeader } from '@/components/ui/Card';
import { fetchLeaderboardForCohort, type LeaderboardRow } from '@/lib/gamification/leaderboard';
import { Trophy, Users } from 'lucide-react';
import { useStore } from '@/lib/store';
import type { GamificationDashboardData } from '@/lib/gamification/dashboard-data';

interface Props {
  data: GamificationDashboardData;
  cohortId: string | null;          // user's primary cohort with leaderboard_enabled=true; null = nothing to show
  hasOptedIn: boolean;
  onOpenLeaderboardSettings: () => void;
  onDismissNudge: () => void;
  nudgeDismissed: boolean;
}

const NUDGE_THRESHOLD_LEVEL = 5;   // D-12 — one-time nudge after user reaches level 5

export function LeaderboardCard({ data, cohortId, hasOptedIn, onOpenLeaderboardSettings, onDismissNudge, nudgeDismissed }: Props) {
  const [rows, setRows] = useState<LeaderboardRow[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!cohortId || !hasOptedIn) return;
    setLoading(true);
    fetchLeaderboardForCohort(cohortId)
      .then(setRows)
      .catch((e) => setError(e.message))
      .finally(() => setLoading(false));
  }, [cohortId, hasOptedIn]);

  // D-12: opt-in nudge after level 5, single-shot
  const showNudge = data.level >= NUDGE_THRESHOLD_LEVEL && !hasOptedIn && cohortId !== null && !nudgeDismissed;

  if (!cohortId) return null;        // user not in any leaderboard-enabled cohort

  if (showNudge) {
    return (
      <Card span={8} variant="interactive">
        <CardHeader title="Join the leaderboard" icon={<Users />} />
        <div className="text-sm text-secondary">
          You've reached level {data.level}. Want to compete (anonymously) with others in your cohort?
        </div>
        <div className="mt-3 flex gap-2">
          <button type="button" className="btn-primary" onClick={onOpenLeaderboardSettings}>Pick a handle</button>
          <button type="button" className="btn-ghost" onClick={onDismissNudge}>Not now</button>
        </div>
      </Card>
    );
  }

  if (!hasOptedIn) return null;

  return (
    <Card span={8} variant="elevated">
      <CardHeader title="Leaderboard" icon={<Trophy />} />
      {loading && <div>Loading…</div>}
      {error && <div className="text-error">Could not load leaderboard</div>}
      {!loading && rows.length === 0 && <div className="text-secondary">No entries yet</div>}
      <ol className="mt-2 space-y-1">
        {rows.map((r) => (
          <li key={`${r.rank_in_cohort}-${r.handle}`} className={r.is_self ? 'font-semibold text-primary' : ''}>
            <span className="inline-block w-8 text-right text-secondary">{r.rank_in_cohort}.</span>{' '}
            <span>{r.handle}</span>{' '}
            <span className="text-secondary text-sm">— {r.xp_7d.toLocaleString()} XP (7d)</span>
          </li>
        ))}
      </ol>
    </Card>
  );
}
```

```tsx
  const [priorLevel, setPriorLevel] = useState<number | null>(null);

    if (!user?.id) return;

  }, [user?.id, priorLevel]);

  if (!data) return null;
```

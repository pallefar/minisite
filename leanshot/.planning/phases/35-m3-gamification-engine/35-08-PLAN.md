Create `supabase/migrations/20270708000020_p35_user_leaderboard_prefs.sql`:

```sql
create table public.user_leaderboard_prefs (
  user_id uuid primary key references auth.users(id) on delete cascade,
  nudge_dismissed_at timestamptz,
  last_seen_at timestamptz,
  updated_at timestamptz not null default now()
);

alter table public.user_leaderboard_prefs enable row level security;

create policy "user_leaderboard_prefs_select_own" on public.user_leaderboard_prefs
  for select to authenticated using (auth.uid() = user_id);

create policy "user_leaderboard_prefs_upsert_own" on public.user_leaderboard_prefs
  for insert to authenticated with check (auth.uid() = user_id);

create policy "user_leaderboard_prefs_update_own" on public.user_leaderboard_prefs
  for update to authenticated using (auth.uid() = user_id) with check (auth.uid() = user_id);

-- No DELETE policy (preference rows persist across sessions; account deletion cascades from auth.users)

comment on table public.user_leaderboard_prefs is
  'Phase 35 D-12 — per-user UI state for the leaderboard nudge. nudge_dismissed_at is monotonic: '
  'once set, the dashboard nudge does NOT re-surface (single-shot contract).';
```

Unlike the ledger tables, this is plain user-owned key-value preference data: select-own + upsert-own + update-own. INSERT/UPDATE policies for `authenticated` are explicitly allowed (not the append-only pattern from xp_ledger) — users own this row.

Defensive trigger to prevent `nudge_dismissed_at` rollback (monotonic):

```sql
create or replace function public._p35_lb_prefs_monotonic() returns trigger
language plpgsql as $body$
begin
  if old.nudge_dismissed_at is not null and new.nudge_dismissed_at is null then
    raise exception 'nudge_dismissed_at is monotonic (cannot un-dismiss)';
  end if;
  return new;
end;
$body$;

create trigger trg_p35_lb_prefs_monotonic
before update on public.user_leaderboard_prefs
for each row execute function public._p35_lb_prefs_monotonic();
```

**`leanshot/src/components/dashboard/settings/LeaderboardsSubtab.tsx`** — mirror NotificationsSubtab pattern:

```tsx
import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import { useToast } from '@/hooks/useToast';
import { setLeaderboardOptin, suggestLeaderboardHandle, LeaderboardApiError } from '@/lib/gamification/leaderboard';
import { validateHandle } from '@/lib/gamification/handle-validate';
import { Card, CardHeader } from '@/components/ui/Card';
import { Button } from '@/components/ui/Button';
import { Input } from '@/components/ui/Input';
import { Trophy } from 'lucide-react';

interface CohortEntry {
  id: string;
  name: string;
  isMember: boolean;
  hasOptedIn: boolean;
  currentHandle: string | null;
}

export function LeaderboardsSubtab() {
  const toast = useToast();
  const [cohorts, setCohorts] = useState<CohortEntry[]>([]);
  const [loading, setLoading] = useState(true);
  const [submittingFor, setSubmittingFor] = useState<string | null>(null);
  const [handleDraft, setHandleDraft] = useState<Record<string, string>>({});

  useEffect(() => {
    let cancelled = false;
    (async () => {
      // Fetch all leaderboard-enabled cohorts the user belongs to + their opt-in state
      const { data: u } = await supabase.auth.getUser();
      if (!u.user) return;
      const { data: rows } = await supabase
        .from('cohort_definitions')
        .select('id, name, leaderboard_enabled, cohort_membership!inner(user_id), leaderboard_optin(user_id, handle, active)')
        .eq('leaderboard_enabled', true)
        .eq('cohort_membership.user_id', u.user.id);
      if (cancelled) return;
      const entries: CohortEntry[] = (rows ?? []).map((r) => {
        const optin = (r.leaderboard_optin as Array<{ user_id: string; handle: string; active: boolean }> | undefined)
          ?.find((o) => o.user_id === u.user!.id);
        return {
          id: r.id as string,
          name: r.name as string,
          isMember: true,
          hasOptedIn: optin?.active ?? false,
          currentHandle: optin?.handle ?? null,
        };
      });
      setCohorts(entries);
      setLoading(false);
    })();
    return () => { cancelled = true; };
  }, []);

  const onSuggest = async (cohortId: string) => {
    try {
      const suggested = await suggestLeaderboardHandle();
      setHandleDraft((prev) => ({ ...prev, [cohortId]: suggested }));
    } catch (e) {
      toast('Could not suggest handle — try entering manually', 'error');
    }
  };

  const onToggle = async (cohort: CohortEntry, optIn: boolean) => {
    setSubmittingFor(cohort.id);
    try {
      const handle = optIn ? (handleDraft[cohort.id] ?? cohort.currentHandle ?? '') : '';
      if (optIn) {
        const v = validateHandle(handle);
        if (!v.ok) {
          toast(v.message, 'error');
          return;
        }
      }
      await setLeaderboardOptin(cohort.id, optIn ? handle : null, optIn);
      toast(optIn ? `Joined ${cohort.name} leaderboard` : `Left ${cohort.name} leaderboard`, 'success');
      // Update local state
      setCohorts((prev) => prev.map((c) =>
        c.id === cohort.id ? { ...c, hasOptedIn: optIn, currentHandle: optIn ? handle : c.currentHandle } : c
      ));
    } catch (e) {
      if (e instanceof LeaderboardApiError) {
        toast(`Could not update: ${e.code}`, 'error');
      } else {
        toast('Could not update leaderboard settings', 'error');
      }
    } finally {
      setSubmittingFor(null);
    }
  };

  if (loading) return <div>Loading…</div>;
  if (cohorts.length === 0) {
    return (
      <Card variant="elevated">
        <CardHeader title="Leaderboards" icon={<Trophy />} />
        <div className="text-sm text-secondary">
          No leaderboards are available for your cohorts yet. Admins enable leaderboards for specific cohorts;
          this page will populate when you're eligible.
        </div>
      </Card>
    );
  }

  return (
    <div className="space-y-4">
      {cohorts.map((cohort) => {
        const draft = handleDraft[cohort.id] ?? cohort.currentHandle ?? '';
        const v = draft ? validateHandle(draft) : null;
        return (
          <Card key={cohort.id} variant="elevated">
            <CardHeader title={cohort.name} icon={<Trophy />} />
            <div className="space-y-3">
              <label className="flex items-center gap-2">
                <input
                  type="checkbox"
                  checked={cohort.hasOptedIn}
                  disabled={submittingFor === cohort.id}
                  onChange={(e) => onToggle(cohort, e.target.checked)}
                  aria-label={`Join ${cohort.name} leaderboard`}
                />
                <span>Show me on this leaderboard</span>
              </label>
              <div className="text-xs text-secondary">
                Your handle is shown in place of your name. We never display your real name on leaderboards.
              </div>
              {(cohort.hasOptedIn || draft) && (
                <div>
                  <label htmlFor={`handle-${cohort.id}`} className="text-sm">Handle</label>
                  <div className="flex gap-2">
                    <Input
                      id={`handle-${cohort.id}`}
                      value={draft}
                      onChange={(e) => setHandleDraft((p) => ({ ...p, [cohort.id]: e.target.value }))}
                      aria-invalid={v && !v.ok ? true : undefined}
                      placeholder={cohort.currentHandle ?? 'Pick a handle'}
                    />
                    <Button variant="secondary" onClick={() => onSuggest(cohort.id)}>Suggest</Button>
                  </div>
                  {v && !v.ok && <div className="text-error text-sm">{v.message}</div>}
                </div>
              )}
            </div>
          </Card>
        );
      })}
    </div>
  );
}
```

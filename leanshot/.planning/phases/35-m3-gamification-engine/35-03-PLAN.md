**grant_xp_for_action SECDEF function:**

```sql
create or replace function public.grant_xp_for_action(
  p_user uuid,
  p_action_type text,
  p_xp_delta int,
  p_source_ref text
)
returns void
language plpgsql
security definer
set search_path = public, pg_catalog
as $body$
declare
  v_prior_total int;
  v_prior_level int;
  v_new_total int;
  v_new_level int;
begin
  -- 1. Compute prior total + level (deterministic)
  v_prior_total := public.xp_total_for(p_user);
  v_prior_level := public.compute_level(v_prior_total);

  -- 2. Append ledger row (D-04 determinism; D-05 source of truth)
  insert into public.xp_ledger (user_id, action_type, xp_delta, source_ref)
  values (p_user, p_action_type, p_xp_delta, p_source_ref);

  -- 3. Detect level-up
  v_new_total := v_prior_total + p_xp_delta;
  v_new_level := public.compute_level(v_new_total);

  if v_new_level > v_prior_level then
    -- Insert level badge_unlock (badge_catalog id pattern: 'level-<N>' or 'prestige-<N>')
    -- Spec: badge_catalog has level-1, level-5, level-10, level-25, level-50, level-100, prestige-1 (Plan 35-01 Task 3 seed)
    -- Only insert if a matching badge_catalog row exists; otherwise skip (no row, no unlock)
    for v_prior_level in v_prior_level + 1 .. v_new_level loop
      if exists(select 1 from public.badge_catalog where id = 'level-' || v_prior_level::text) then
        insert into public.badge_unlocks (user_id, badge_id, source, source_ref)
        values (p_user, 'level-' || v_prior_level::text, 'level', 'level:' || v_prior_level::text)
        on conflict (user_id, badge_id) do nothing;
      end if;
    end loop;
    -- Prestige check (D-03)
    if v_new_level / 100 > v_prior_level / 100 then
      if exists(select 1 from public.badge_catalog where id = 'prestige-' || (v_new_level / 100)::text) then
        insert into public.badge_unlocks (user_id, badge_id, source, source_ref)
        values (p_user, 'prestige-' || (v_new_level / 100)::text, 'level', 'prestige:' || (v_new_level / 100)::text)
        on conflict (user_id, badge_id) do nothing;
      end if;
    end if;
  end if;
end;
$body$;

revoke all on function public.grant_xp_for_action(uuid, text, int, text) from public;
grant execute on function public.grant_xp_for_action(uuid, text, int, text) to service_role;
```

```sql
-- Trigger function template
create or replace function public._p35_xp_on_injection() returns trigger
language plpgsql security definer set search_path = public, pg_catalog
as $body$
begin
  perform public.grant_xp_for_action(new.user_id, 'injection_log', 25, 'injections:' || new.id::text);
  return new;
exception when others then
  raise notice 'p35_xp_on_injection: error % — continuing (do not block source insert)', sqlerrm;
  return new;
end;
$body$;

create trigger trg_p35_xp_on_injection
after insert on public.injections
for each row execute function public._p35_xp_on_injection();
```

Repeat the pattern for the 3 other source tables. Each trigger function:
- Takes `new.user_id` (NOT `auth.uid()`)
- Uses `source_ref` like `<tablename>:<id>` for audit trace
- Wraps the `perform` in `exception when others` so a gamification error does NOT block the source insert (defense-in-depth — never let XP grant failure cascade into broken user flow)
- Returns `new` (AFTER triggers can also return null without effect, but `return new` is convention)

**Trigger ordering note** (per Research OQ-4): AFTER INSERT is safe with Phase 5 Realtime sync because Realtime LISTEN/NOTIFY fires from logical decoding (WAL-based), NOT from row triggers. The trigger function INSERTing into `xp_ledger` will itself emit a separate WAL event, which Realtime can also subscribe to (Plan 35-06 dashboard subscribes to xp_ledger for "level-up live notification").

**Idempotency note:** If a user retries inserting the same injection (e.g., offline → online sync resolves a duplicate), Phase 5 sync layer already de-dupes via `injections.id` primary key. The xp_ledger does NOT have a UNIQUE (source_ref) constraint because the same source_ref can legitimately appear twice if a different action type references it (rare; defensive). However, the UNIQUE INDEX from Plan 35-01 on `(user_id, action_type, cycle_id) WHERE cycle_id IS NOT NULL` blocks streak/weekly cycle double-grants, which is the actual risk.

```sql
create or replace function public._p35_combo_badge_check() returns trigger
language plpgsql security definer set search_path = public, pg_catalog
as $body$
declare
  v_streak int;
begin
  -- GAME-09: combo unlocks when challenge_complete + active streak (>= 7 days; "compound consistency")
  if new.source <> 'challenge' then
    return new;
  end if;

  select current_streak_days into v_streak from public.streak_state where user_id = new.user_id;
  if coalesce(v_streak, 0) >= 7 then
    -- Insert combo badge if not already unlocked
    insert into public.badge_unlocks (user_id, badge_id, source, source_ref)
    values (new.user_id, 'combo-cross-streak', 'combo', 'combo:' || new.source_ref)

    -- Grant +50 XP combo bonus
    perform public.grant_xp_for_action(new.user_id, 'combo_unlock', 50, 'combo:' || new.source_ref);
  end if;

  return new;
exception when others then
  raise notice 'p35_combo_badge_check: error % — continuing', sqlerrm;
  return new;
end;
```

**`index.ts`** — Deno.serve handler:

```ts
import { z } from 'zod';
import { captureServer, shutdownPostHog } from '../_shared/posthog-server.ts';
import { createClient } from '@supabase/supabase-js';

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

const BodySchema = z.object({
  event: z.enum(['xp_earned', 'level_up', 'streak_milestone', 'freeze_token_granted', 'challenge_completed', 'badge_unlocked']),
  properties: z.record(z.unknown()),  // open; sanitized server-side below
});

const ALLOWED_PROPERTY_KEYS = new Set([
  'action_type', 'xp_delta', 'xp_total', 'level', 'prior_level', 'badge_id',
  'challenge_id', 'streak_days', 'reason', 'cohort_id_anonymized',
]);

function sanitizeProperties(props: Record<string, unknown>): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  for (const [k, v] of Object.entries(props)) {
    if (ALLOWED_PROPERTY_KEYS.has(k)) out[k] = v;
  }
  return out;
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: corsHeaders });
  if (req.method !== 'POST') return new Response(JSON.stringify({ error: 'method_not_allowed' }), { status: 405, headers: { 'content-type': 'application/json', ...corsHeaders }});

  try {
    const auth = req.headers.get('authorization') ?? '';
    const jwt = auth.toLowerCase().startsWith('bearer ') ? auth.slice(7) : '';
    if (!jwt) return new Response(JSON.stringify({ error: 'unauthenticated' }), { status: 401, headers: { 'content-type': 'application/json', ...corsHeaders }});

    const url = Deno.env.get('SUPABASE_URL') ?? '';
    const serviceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? '';
    const admin = createClient(url, serviceKey, { auth: { autoRefreshToken: false } });
    const { data: { user } } = await admin.auth.getUser(jwt);
    if (!user) return new Response(JSON.stringify({ error: 'unauthenticated' }), { status: 401, headers: { 'content-type': 'application/json', ...corsHeaders }});

    const raw = await req.json().catch(() => null);
    const parsed = BodySchema.safeParse(raw);
    if (!parsed.success) return new Response(JSON.stringify({ error: 'invalid_body' }), { status: 400, headers: { 'content-type': 'application/json', ...corsHeaders }});

    // Sanitize properties — drop any keys NOT in the allowlist (defense-in-depth against PHI leak per Phase 25 HIPAA-08)
    const sanitized = sanitizeProperties(parsed.data.properties);

    captureServer({
      distinctId: user.id,
      event: parsed.data.event,
      properties: sanitized,
    });

    return new Response(JSON.stringify({ ok: true }), { status: 200, headers: { 'content-type': 'application/json', ...corsHeaders }});
  } catch (e) {
    console.error('[xp-event] handler error', e);
    return new Response(JSON.stringify({ error: 'internal_error' }), { status: 500, headers: { 'content-type': 'application/json', ...corsHeaders }});
  } finally {
    try { await shutdownPostHog(); } catch (e) { console.error('[xp-event] shutdown failed', e); }
  }
});
```

**`leanshot/src/lib/gamification/xp.ts`** — client-side compute_level mirror:

```ts
/**
 * Client-side mirror of Postgres compute_level() for UI hints.
 * Single source of truth = DB function (Plan 35-01). This is for "XP to next level" display only.
 * If this diverges from the SQL function, the UI hint is wrong — DB always wins.
 */
export function computeLevel(xpTotal: number): number {
  // D-02: Level N requires N²·100 XP cumulative
  return Math.max(1, Math.floor(Math.sqrt(Math.max(xpTotal, 0) / 100)));
}

export function xpForLevel(level: number): number {
  // D-02 inverse: cumulative XP at start of level N
  return Math.max(0, level * level * 100);
}

export function xpToNextLevel(xpTotal: number): { level: number; xpInLevel: number; xpToNext: number; nextLevelXp: number } {
  const level = computeLevel(xpTotal);
  const currentLevelStart = xpForLevel(level);
  const nextLevelXp = xpForLevel(level + 1);
  return {
    level,
    xpInLevel: xpTotal - currentLevelStart,
    xpToNext: nextLevelXp - xpTotal,
    nextLevelXp,
  };
}

export function computePrestige(xpTotal: number): number {
  return Math.max(0, Math.floor(computeLevel(xpTotal) / 100));
}
```

**`supabase/functions/challenge-evaluate-cron/index.ts`** — hourly cron-invoked Edge Fn:

```ts
import { createClient } from '@supabase/supabase-js';
import { checkServiceRoleBearer, makeLazyAdmin } from '../_shared/lifecycle-utils.ts';
import { captureServer, shutdownPostHog } from '../_shared/posthog-server.ts';

const admin = makeLazyAdmin();

Deno.serve(async (req) => {
  if (req.method !== 'POST') return new Response('method_not_allowed', { status: 405 });
  if (!checkServiceRoleBearer(req)) return new Response('unauthorized', { status: 401 });

  try {
    const a = admin();
    // 1. Find all users with at least one active challenge
    const { data: activeUsers, error } = await a.from('challenge_progress')
      .select('user_id, challenge_id, weekly_challenges!inner(status, ends_at)')
      .eq('weekly_challenges.status', 'active')
      .gt('weekly_challenges.ends_at', new Date().toISOString());
    if (error) throw error;

    const uniqueUsers = Array.from(new Set((activeUsers ?? []).map((r: { user_id: string }) => r.user_id)));

    let processed = 0;
    let errors = 0;
    for (const userId of uniqueUsers) {
      try {
        await a.rpc('evaluate_challenge_progress_for_user', { p_user: userId });
        processed++;
      } catch (e) {
        console.error('[challenge-evaluate-cron]', userId, e);
        errors++;
      }
    }

    // 2. Trigger lifecycle-behavior-triggered to dispatch any new notifications
    //    (fire-and-forget; let lifecycle-behavior-triggered's own cron also pick up)
    void a.functions.invoke('lifecycle-behavior-triggered', { body: { trigger: 'gamification' }}).catch((e) => {
      console.warn('[challenge-evaluate-cron] lifecycle dispatch failed', e);
    });

    captureServer({
      distinctId: 'system',
      event: 'gamification_cron_completed',
      properties: { users_processed: processed, errors, kind: 'challenge_evaluate' },
    });

    return new Response(JSON.stringify({ ok: true, processed, errors }), {
      status: 200,
      headers: { 'content-type': 'application/json' },
    });
  } catch (e) {
    console.error('[challenge-evaluate-cron] handler failed', e);
    return new Response(JSON.stringify({ error: 'internal_error' }), { status: 500, headers: { 'content-type': 'application/json' } });
  } finally {
    try { await shutdownPostHog(); } catch (e) { console.error('[challenge-evaluate-cron] shutdown failed', e); }
  }
});
```

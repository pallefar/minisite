### `billing.ts` exports (`leanshot/src/lib/billing.ts`)

```typescript
export type FeatureKey = 'pharma-forecast' | 'advanced-ai' | 'ad-free';
export type TierGateMode = 'blur-upsell' | 'hard-block-no-ui' | 'hard-block-cta';

export const TIER_GATE_REGISTRY: Record<FeatureKey, TierGateMode>;

/**
 * Collapse Stripe subscription state (status + current_period_end) into the
 * 3-state UX tier per RESEARCH Pitfall 6.
 *
 * @param stripeStatus  one of Stripe's 8 statuses or null if no row
 * @param currentPeriodEnd  ISO timestamp from Stripe subscription
 * @param now  injected for deterministic testing
 */
export function getActiveTier(
  stripeStatus:
    | 'incomplete'
    | 'incomplete_expired'
    | 'trialing'
    | 'active'
    | 'past_due'
    | 'canceled'
    | 'unpaid'
    | 'paused'
    | null,
  currentPeriodEnd: string | null,
  now: Date,
): Tier;

/** Reset module-level tier cache. Wired into useStore.signOut. */
export function clearTierCache(): void;
```

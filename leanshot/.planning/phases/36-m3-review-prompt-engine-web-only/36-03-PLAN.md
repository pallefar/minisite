**W4 — analytics-trigger-bus.ts (NEW FILE — modeled on src/lib/nps/quarterly-modal.ts)**:
```typescript
// src/lib/nps/analytics-trigger-bus.ts
export const ANALYTICS_TRIGGER_EVENT = 'leanshot:analytics-event';

export interface AnalyticsTriggerDetail {
  name: string; // matches an entry key in EVENTS registry
  properties?: Record<string, string | number | boolean>;
}

export function isAnalyticsTriggerDetail(detail: unknown): detail is AnalyticsTriggerDetail {
  if (!detail || typeof detail !== 'object') return false;
  const d = detail as { name?: unknown };
  return typeof d.name === 'string' && d.name.length > 0;
}

export function dispatchAnalyticsTrigger(detail: AnalyticsTriggerDetail): void {
  try {
    if (typeof window === 'undefined') return;
    window.dispatchEvent(
      new CustomEvent<AnalyticsTriggerDetail>(ANALYTICS_TRIGGER_EVENT, { detail }),
    );
  } catch {
    /* fail-soft: SSR contexts (none in this SPA) and locked-down iframes. */
  }
}
```

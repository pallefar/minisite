**`leanshot/src/lib/onboarding-builder/use-consumer-onboarding-flow.ts`** — mirror `use-org-onboarding-flow.ts` state machine; specialize for the consumer table.

```typescript
import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import { useStore } from '@/lib/store';
import posthog from 'posthog-js';
// Reuse OnboardingStepNode from the org hook to avoid type drift.
import type { OnboardingStepNode } from '@/lib/onboarding-builder/use-org-onboarding-flow';

export type ConsumerOnboardingStatus = 'loading' | 'preview' | 'consumer' | 'completed';
export interface ConsumerOnboardingFlow {
  id: string;
  config: OnboardingStepNode[];
  version: number;
  variant_id: string | null;
}
export interface ConsumerOnboardingFlowState {
  status: ConsumerOnboardingStatus;
  flow: ConsumerOnboardingFlow | null;
}

const INITIAL_STATE: ConsumerOnboardingFlowState = { status: 'loading', flow: null };

function waitForFeatureFlags(): Promise<void> {
  return new Promise<void>((resolve) => {
    let done = false;
    const finish = () => { if (!done) { done = true; resolve(); } };
    try { posthog.onFeatureFlags(() => finish()); } catch { finish(); }
    setTimeout(finish, 2000);
  });
}

export function useConsumerOnboardingFlow(): ConsumerOnboardingFlowState {
  const signedInUser = useStore((s) => s.signedIn?.user ?? null);
  const localUser = useStore((s) => s.user);
  const [state, setState] = useState<ConsumerOnboardingFlowState>(INITIAL_STATE);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        if (signedInUser?.id) {
          const { data: prof } = await supabase
            .from('profiles')
            .select('completed_onboarding_at')
            .eq('id', signedInUser.id)
            .maybeSingle();
          if (cancelled) return;
          if (prof?.completed_onboarding_at) {
            setState({ status: 'completed', flow: null });
            return;
          }
        }
        await waitForFeatureFlags();
        if (cancelled) return;
        let versionId: string | null = null;
        try {
          const payload = posthog.getFeatureFlagPayload('onboarding-ab');
          if (payload && typeof payload === 'object' && 'version_id' in payload) {
            versionId = String((payload as { version_id: unknown }).version_id);
          }
        } catch { /* fall back to is_active */ }
        const base = supabase.from('onboarding_flows').select('id, config, version');
        const { data: flowRow } = versionId
          ? await base.eq('id', versionId).maybeSingle()
          : await base.eq('is_active', true).maybeSingle();
        if (cancelled) return;
        if (!flowRow) {
          setState({ status: signedInUser ? 'consumer' : 'preview', flow: null });
          return;
        }
        setState({
          status: signedInUser ? 'consumer' : 'preview',
          flow: {
            id: flowRow.id,
            config: (flowRow.config ?? []) as OnboardingStepNode[],
            version: flowRow.version,
            variant_id: versionId,
          },
        });
      } catch (err) {
        if (!cancelled) {
          console.warn('[useConsumerOnboardingFlow] fail-open:', err);
          setState({ status: signedInUser ? 'consumer' : 'preview', flow: null });
        }
      }
    })();
    return () => { cancelled = true; };
  }, [signedInUser?.id, localUser?.id]);

  return state;
}
```

import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';

const OPTOUT_KEY = 'leanshot_social_proof_optout';

export default function LiveSignupCounter(): JSX.Element | null {
  const optedOut = (() => { try { return localStorage.getItem(OPTOUT_KEY) === 'true'; } catch { return false; } })();
  const [count, setCount] = useState<number | null>(null);
  useEffect(() => {
    if (optedOut) return;
    let cancelled = false;
    async function tick() {
      try {
        const { data } = await supabase.rpc('get_rolling_signup_count');
        if (!cancelled && typeof data === 'number') setCount(data);
      } catch { /* silent */ }
    }
    void tick();
    const interval = setInterval(() => {
      if (document.visibilityState === 'visible') void tick();
    }, 30_000);
    const visHandler = () => { if (document.visibilityState === 'visible') void tick(); };
    document.addEventListener('visibilitychange', visHandler);
    return () => { cancelled = true; clearInterval(interval); document.removeEventListener('visibilitychange', visHandler); };
  }, [optedOut]);
  if (optedOut) return null;
  if (count === null) return <div aria-hidden className="h-5 w-32 bg-[var(--color-surface-elevated)] rounded animate-pulse" />;
  return <p aria-live="polite" className="text-sm text-[var(--color-text-muted)]">{count.toLocaleString()} people joined this week</p>;
}

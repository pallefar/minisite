```typescript
function vendorHealthCheck(): Response | null {
  const apiKey = Deno.env.get('POSTHOG_PERSONAL_API_KEY');
  const projectId = Deno.env.get('POSTHOG_PROJECT_ID');
  if (!apiKey || !projectId) {
    console.warn('[ship-winner-flag] vendor_unconfigured: POSTHOG_PERSONAL_API_KEY or POSTHOG_PROJECT_ID missing');
    return jsonResponse(503, { error: 'vendor_unconfigured', service: 'posthog' });
  }
  return null;
}

const BodySchema = z.object({
  flag_id: z.string().regex(/^\d+$/), // PostHog flag IDs are integers
  variant: z.string().min(1).max(64),
});

async function handleShipWinner(req: Request): Promise<Response> {
  if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: corsHeaders });
  if (req.method !== 'POST') return jsonError(405, 'method_not_allowed');

  const health = vendorHealthCheck();
  if (health) return health;

  // Auth + role check
  const auth = req.headers.get('authorization');
  if (!auth) return jsonError(401, 'unauthenticated');
  const admin = getAdmin();
  const { data: userRes } = await admin.auth.getUser(auth.replace(/^Bearer\s+/i, ''));
  if (!userRes?.user) return jsonError(401, 'unauthenticated');
  const callerUid = userRes.user.id;

  const { data: prof } = await admin.from('profiles').select('admin_role').eq('id', callerUid).maybeSingle();
  if (prof?.admin_role !== 'superadmin') return jsonError(403, 'forbidden_not_superadmin');

  // Body
  const parsed = BodySchema.safeParse(await req.json().catch(() => ({})));
  if (!parsed.success) return jsonError(400, 'invalid_body');
  const { flag_id, variant } = parsed.data;

  // PATCH PostHog
  const apiKey = Deno.env.get('POSTHOG_PERSONAL_API_KEY')!;
  const projectId = Deno.env.get('POSTHOG_PROJECT_ID')!;
  const url = `https://us.i.posthog.com/api/projects/${projectId}/feature_flags/${flag_id}/`;
  const phRes = await fetch(url, {
    method: 'PATCH',
    headers: { 'Authorization': `Bearer ${apiKey}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({
      filters: {
        groups: [{ properties: [], rollout_percentage: 100, variant }],
        multivariate: null,
      },
    }),
  });
  if (!phRes.ok) {
    const text = await phRes.text().catch(() => '');
    console.error('[ship-winner-flag] PostHog PATCH failed:', phRes.status, text);
    return jsonError(phRes.status === 403 ? 403 : 502, 'posthog_patch_failed');
  }

  // Audit log (best-effort — non-blocking on insert failure)
  try {
    await admin.from('audit_logs').insert({
      actor: callerUid,
      action: 'onboarding.ship_winner',
      target_kind: 'posthog_feature_flag',
      target_id: flag_id,
      after: { variant, rollout_percentage: 100 },
    });
  } catch (err) {
    console.warn('[ship-winner-flag] audit log insert failed (non-blocking):', err);
  }

  return jsonResponse(200, { ok: true, flag_id, variant });
}
```

```typescript
function vendorHealthCheck(): Response | null {
  const apiKey = Deno.env.get('POSTHOG_PERSONAL_API_KEY');
  const projectId = Deno.env.get('POSTHOG_PROJECT_ID');
  if (!apiKey || !projectId) return jsonResponse(503, { error: 'vendor_unconfigured', service: 'posthog' });
  return null;
}

const BodySchema = z.object({
  flow_id: z.string().uuid(),
  time_range_days: z.number().int().min(1).max(90),
});

interface CacheEntry { ts: number; payload: unknown; }
const CACHE = new Map<string, CacheEntry>();
const CACHE_TTL_MS = 60_000;

async function handleFunnelQuery(req: Request): Promise<Response> {
  if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: corsHeaders });
  if (req.method !== 'POST') return jsonError(405, 'method_not_allowed');
  const health = vendorHealthCheck();
  if (health) return health;

  // Auth + admin-or-higher
  const auth = req.headers.get('authorization');
  if (!auth) return jsonError(401, 'unauthenticated');
  const admin = getAdmin();
  const { data: userRes } = await admin.auth.getUser(auth.replace(/^Bearer\s+/i, ''));
  if (!userRes?.user) return jsonError(401, 'unauthenticated');
  const { data: prof } = await admin.from('profiles').select('admin_role').eq('id', userRes.user.id).maybeSingle();
  if (!prof?.admin_role || !['admin', 'superadmin'].includes(prof.admin_role)) return jsonError(403, 'forbidden');

  // Body
  const parsed = BodySchema.safeParse(await req.json().catch(() => ({})));
  if (!parsed.success) return jsonError(400, 'invalid_body');
  const cacheKey = `${parsed.data.flow_id}:${parsed.data.time_range_days}`;
  const cached = CACHE.get(cacheKey);
  if (cached && Date.now() - cached.ts < CACHE_TTL_MS) {
    return jsonResponse(200, cached.payload);
  }

  // PostHog Insights HogQL query
  const apiKey = Deno.env.get('POSTHOG_PERSONAL_API_KEY')!;
  const projectId = Deno.env.get('POSTHOG_PROJECT_ID')!;
  const hogQL = `
    SELECT
      properties.step_id AS step_id,
      countIf(event = 'onboarding.step.shown') AS views,
      countIf(event = 'onboarding.step.completed') AS completions,
      avg(properties.time_on_step_ms) AS avg_time_ms
    FROM events
    WHERE properties.flow_id = '${parsed.data.flow_id}'
      AND timestamp >= now() - INTERVAL ${parsed.data.time_range_days} DAY
    GROUP BY properties.step_id
    ORDER BY views DESC
  `;
  const phRes = await fetch(`https://us.i.posthog.com/api/projects/${projectId}/query/`, {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${apiKey}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({ query: { kind: 'HogQLQuery', query: hogQL } }),
  });
  if (!phRes.ok) return jsonError(502, 'posthog_query_failed');
  const phData = await phRes.json();
  const rows = (phData?.results ?? []) as Array<[string, number, number, number]>;
  const steps = rows.map(([step_id, views, completions, avg_time_ms]) => ({
    step_id,
    views,
    completions,
    drop_off_pct: views > 0 ? Math.round(((views - completions) / views) * 100) : 0,
    avg_time_ms: Math.round(avg_time_ms ?? 0),
  }));
  const payload = { steps };
  CACHE.set(cacheKey, { ts: Date.now(), payload });
  return jsonResponse(200, payload);
}
```

```tsx
import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import { useToast } from '@/hooks/useToast';

interface Experiment { id: number; key: string; name: string; current_rollout: number; variants: string[]; }

export default function OnboardingABPanel(): JSX.Element {
  const toast = useToast();
  const [experiments, setExperiments] = useState<Experiment[] | null>(null);
  const [unconfigured, setUnconfigured] = useState(false);
  const [busyId, setBusyId] = useState<number | null>(null);
  const [adminRole, setAdminRole] = useState<string | null>(null);

  useEffect(() => {
    (async () => {
      // Cache admin_role for the session
      const { data: { user } } = await supabase.auth.getUser();
      if (user) {
        const { data: prof } = await supabase.from('profiles').select('admin_role').eq('id', user.id).maybeSingle();
        setAdminRole(prof?.admin_role ?? null);
      }
      const { data, error } = await supabase.functions.invoke('onboarding-funnel-query', { body: { kind: 'list_experiments' } });
      if (error || (data as { error?: string })?.error === 'vendor_unconfigured') {
        setUnconfigured(true);
        setExperiments([]);
        return;
      }
      setExperiments((data as { experiments?: Experiment[] })?.experiments ?? []);
    })();
  }, []);

  async function handleShip(exp: Experiment, variant: string): Promise<void> {
    setBusyId(exp.id);
    try {
      const { data, error } = await supabase.functions.invoke('ship-winner-flag', { body: { flag_id: String(exp.id), variant } });
      if (error || (data as { error?: string })?.error) {
        if ((data as { error?: string })?.error === 'forbidden_not_superadmin') {
          toast.error('Superadmin role required');
        } else {
          toast.error(`Ship failed: ${error?.message ?? 'unknown'}`);
        }
        return;
      }
      toast.success(`Shipped variant ${variant} to 100%`);
      // Reload list
      const { data: refresh } = await supabase.functions.invoke('onboarding-funnel-query', { body: { kind: 'list_experiments' } });
      setExperiments((refresh as { experiments?: Experiment[] })?.experiments ?? []);
    } finally {
      setBusyId(null);
    }
  }

  if (unconfigured) {
    return (
      <div role="status" className="rounded-2xl border border-[var(--color-warning)] bg-[var(--color-warning-soft)] p-4">
        <p className="font-medium">PostHog API key not yet configured.</p>
        <p className="text-sm text-[var(--color-text-muted)] mt-1">Plan 34-10 owns the human checkpoint that sets POSTHOG_PERSONAL_API_KEY + POSTHOG_PROJECT_ID as Supabase Function Secrets.</p>
      </div>
    );
  }

  if (!experiments) return <div className="text-[var(--color-text-muted)]">Loading experiments…</div>;
  if (experiments.length === 0) return <p className="text-[var(--color-text-muted)]">No onboarding experiments yet. Create one in PostHog → Experiments.</p>;

  const canShip = adminRole === 'superadmin';
  return (
    <ul className="space-y-3">
      {experiments.map((exp) => (
        <li key={exp.id} className="bg-[var(--color-surface)] border border-[var(--color-border)] rounded-2xl p-4">
          <header className="flex items-center justify-between">
            <div>
              <h3 className="font-semibold">{exp.name}</h3>
              <p className="text-sm text-[var(--color-text-muted)] font-mono">{exp.key} · rollout {exp.current_rollout}%</p>
            </div>
          </header>
          <div className="mt-3 flex flex-wrap gap-2">
            {exp.variants.map((v) => (
              <Button
                key={v}
                variant="primary"
                disabled={!canShip}
                title={!canShip ? 'Superadmin only' : undefined}
                loading={busyId === exp.id}
                onClick={() => void handleShip(exp, v)}
                className="min-h-[44px]"
              >
                Ship "{v}" to 100%
              </Button>
            ))}
          </div>
        </li>
      ))}
    </ul>
  );
}
```

```tsx
import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import BaseChart from '@/components/dashboard/charts/BaseChart';

interface StepRow { step_id: string; views: number; completions: number; drop_off_pct: number; avg_time_ms: number; }

export default function OnboardingFunnelTab({ flowId }: { flowId: string | null }): JSX.Element {
  const [steps, setSteps] = useState<StepRow[] | null>(null);
  const [unconfigured, setUnconfigured] = useState(false);

  useEffect(() => {
    if (!flowId) return;
    let cancelled = false;
    async function tick(): Promise<void> {
      const { data } = await supabase.functions.invoke('onboarding-funnel-query', {
        body: { kind: 'step_funnel', flow_id: flowId, time_range_days: 14 },
      });
      if (cancelled) return;
      if ((data as { error?: string })?.error === 'vendor_unconfigured') { setUnconfigured(true); return; }
      setSteps(((data as { steps?: StepRow[] })?.steps) ?? []);
    }
    void tick();
    const interval = setInterval(() => { if (document.visibilityState === 'visible') void tick(); }, 30_000);
    return () => { cancelled = true; clearInterval(interval); };
  }, [flowId]);

  if (unconfigured) {
    return (

        <p className="font-medium">PostHog Insights not yet configured.</p>
        <p className="text-sm text-[var(--color-text-muted)] mt-1">Funnel analytics ship once Plan 34-10's vendor checkpoint completes.</p>
      </div>

  }
  if (!flowId) return <p className="text-[var(--color-text-muted)]">Save a flow to enable funnel tracking.</p>;
  if (!steps) return <p className="text-[var(--color-text-muted)]">Loading funnel…</p>;
  if (steps.length === 0) return <p className="text-[var(--color-text-muted)]">No step events captured yet (14-day window).</p>;

  // Summary KPI strip + per-step table
  const totalViews = steps[0]?.views ?? 0;
  const completionRate = totalViews > 0 ? Math.round((steps[steps.length - 1].completions / totalViews) * 100) : 0;

  return (
    <div className="space-y-4">
      <div className="grid grid-cols-3 gap-3">
        <KpiTile label="Total views (14d)" value={totalViews.toLocaleString()} />
        <KpiTile label="End-to-end completion" value={`${completionRate}%`} />
        <KpiTile label="Steps" value={steps.length.toString()} />
      </div>
      <BaseChart
        config={{
          type: 'bar',
          data: {
            labels: steps.map((s) => s.step_id.slice(0, 8)),
            datasets: [
              { label: 'Views', data: steps.map((s) => s.views) },
              { label: 'Completions', data: steps.map((s) => s.completions) },
            ],
          },
          options: { responsive: true, maintainAspectRatio: false },
        }}
      />
      <table className="w-full text-sm">
        <thead><tr><th>Step</th><th>Views</th><th>Completions</th><th>Drop-off</th><th>Avg time</th></tr></thead>
        <tbody>
          {steps.map((s) => (
            <tr key={s.step_id}>
              <td className="font-mono">{s.step_id}</td>
              <td>{s.views}</td>
              <td>{s.completions}</td>
              <td>{s.drop_off_pct}%</td>
              <td>{(s.avg_time_ms / 1000).toFixed(1)}s</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

function KpiTile({ label, value }: { label: string; value: string }): JSX.Element {
  return (
    <div className="bg-[var(--color-surface)] border border-[var(--color-border)] rounded-2xl p-3">
      <div className="text-xs text-[var(--color-text-muted)]">{label}</div>
      <div className="text-2xl font-semibold">{value}</div>
    </div>
  );
}
```

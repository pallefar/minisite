import { STEP_TYPE_LABELS, type OnboardingStepType, type OnboardingStepNode } from '@/types/onboarding-step';

interface StepPaletteProps {
  onAddStep: (type: OnboardingStepType) => void;
}

export default function StepPalette({ onAddStep }: StepPaletteProps): JSX.Element {
  const TYPES: OnboardingStepType[] = ['text', 'single-select', 'multi-select', 'scale', 'weight', 'date', 'nps', 'custom-component'];
  return (
    <div role="toolbar" aria-label="Step palette" className="flex flex-wrap gap-2 p-3 bg-[var(--color-surface-elevated)] rounded-2xl">
      {TYPES.map((t) => (
        <button
          key={t}
          type="button"
          onClick={() => onAddStep(t)}
          className="min-h-[44px] px-3 py-2 rounded-full border border-[var(--color-border)] bg-[var(--color-surface)] text-sm font-medium hover:border-[var(--color-primary)]"
          aria-label={`Add ${STEP_TYPE_LABELS[t]} step`}
        >
          + {STEP_TYPE_LABELS[t]}
        </button>
      ))}
    </div>
  );
}

export function createStepOfType(type: OnboardingStepType): OnboardingStepNode {
  return {
    id: crypto.randomUUID(),
    type,
    copy: { title: '', subtitle: '' },
    field: type === 'custom-component' ? undefined : { key: '', required: false },
    options: (type === 'single-select' || type === 'multi-select') ? [] : undefined,
  };
}

**`OnboardingBuilderModule.tsx`** (~150 lines): top-level shell.

```tsx
import { useEffect, useMemo, useState } from 'react';
import { supabase } from '@/lib/supabase';
import { useToast } from '@/hooks/useToast';
import { SortableTreePanel } from '@/components/ui/SortableTreePanel';
import { AdminLayout } from '@/components/admin/layout/AdminLayout'; // or whatever the project uses
import StepPalette, { createStepOfType } from './StepPalette';
import StepRow from './StepRow';
import StepPropertyPanel from './StepPropertyPanel';
import type { OnboardingStepNode, OnboardingStepType } from '@/types/onboarding-step';

type Tab = 'builder' | 'ab' | 'funnel';

export default function OnboardingBuilderModule(): JSX.Element {
  const toast = useToast();
  const [tab, setTab] = useState<Tab>('builder');
  const [steps, setSteps] = useState<OnboardingStepNode[]>([]);
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [adminRole, setAdminRole] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);

  const isSuperadmin = adminRole === 'superadmin';

  useEffect(() => {
    (async () => {
      const { data: { user } } = await supabase.auth.getUser();
      if (user) {
        const { data: prof } = await supabase.from('profiles').select('admin_role').eq('id', user.id).maybeSingle();
        setAdminRole(prof?.admin_role ?? null);
      }
      const { data: flow } = await supabase
        .from('onboarding_flows')
        .select('config')
        .eq('is_active', true)
        .maybeSingle();
      setSteps((flow?.config ?? []) as OnboardingStepNode[]);
      setLoading(false);
    })();
  }, []);

  async function handleSave(): Promise<void> {
    setSaving(true);
    try {
      const { error } = await supabase.rpc('save_consumer_onboarding_flow', { p_steps: steps });
      if (error) {
        if (error.code === '42501' || /superadmin/.test(error.message)) {
          toast.error('Superadmin role required to save');
        } else {
          toast.error(`Save failed: ${error.message}`);
        }
        return;
      }
      toast.success('Saved as new version');
    } finally {
      setSaving(false);
    }
  }

  const selected = useMemo(() => steps.find((s) => s.id === selectedId) ?? null, [steps, selectedId]);

  return (
    <AdminLayout>
      <header className="flex items-center justify-between mb-6">
        <h1 className="text-2xl font-semibold">Onboarding builder</h1>
        <Button
          variant="primary"
          loading={saving}
          disabled={!isSuperadmin}
          title={!isSuperadmin ? 'Superadmin only' : undefined}
          onClick={() => void handleSave()}
        >
          Save flow
        </Button>
      </header>
      <nav role="tablist" className="flex gap-2 border-b border-[var(--color-border)] mb-6">
        {(['builder', 'ab', 'funnel'] as const).map((t) => (
          <button
            key={t}
            role="tab"
            aria-selected={tab === t}
            onClick={() => setTab(t)}
            className={`px-4 py-2 min-h-[44px] ${tab === t ? 'border-b-2 border-[var(--color-primary)] font-semibold' : ''}`}
          >
            {t === 'builder' ? 'Builder' : t === 'ab' ? 'A/B Experiments' : 'Funnel Analytics'}
          </button>
        ))}
      </nav>

      {tab === 'builder' && (
        <div className="grid grid-cols-1 lg:grid-cols-[1fr,360px] gap-4">
          <div className="space-y-3">
            <StepPalette onAddStep={(t) => setSteps((prev) => [...prev, createStepOfType(t)])} />
            {loading ? (
              <div className="text-[var(--color-text-muted)]">Loading flow…</div>
            ) : (
              <SortableTreePanel<OnboardingStepNode>
                items={steps}
                getId={(s) => s.id}
                onReorder={(next) => setSteps(next)}
                renderItem={(s, i, dragging) => (
                  <StepRow step={s} index={i} dragging={dragging} onClick={() => setSelectedId(s.id)} />
                )}
                announceItemLabel={(s) => `${s.type}: ${s.copy?.title || 'untitled'}`}
              />
            )}
          </div>
          <aside className="bg-[var(--color-surface-elevated)] p-4 rounded-2xl min-h-[400px]">
            {selected ? (
              <StepPropertyPanel
                step={selected}
                onChange={(next) => setSteps((prev) => prev.map((s) => s.id === selected.id ? next : s))}
                onDelete={() => { setSteps((prev) => prev.filter((s) => s.id !== selected.id)); setSelectedId(null); }}
              />
            ) : (
              <p className="text-[var(--color-text-muted)]">Select a step to edit its properties.</p>
            )}
          </aside>
        </div>
      )}
      {tab === 'ab' && <TabPlaceholder name="A/B Experiments" plan="34-09" />}
      {tab === 'funnel' && <TabPlaceholder name="Funnel Analytics" plan="34-09" />}
    </AdminLayout>
  );
}

function TabPlaceholder({ name, plan }: { name: string; plan: string }): JSX.Element {
  return (
    <div className="text-center py-12 text-[var(--color-text-muted)]">
      <p>{name} tab ships in Plan {plan}.</p>
    </div>
  );
}
```

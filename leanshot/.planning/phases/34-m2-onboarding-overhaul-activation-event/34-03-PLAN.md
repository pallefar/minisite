```typescript
// Imports: @supabase/supabase-js + zod (via esm.sh) + ../_shared/posthog-server.ts
import { captureServer, shutdownPostHog } from '../_shared/posthog-server.ts';

const BodySchema = z.object({
  goal_type: z.enum([
    'lose-weight','build-muscle','new-prescription','build-habit',
    'doctor-monitored','family-supporter','manage-symptoms','track-with-vial-supply',
  ]),
  action_type: z.string().min(1),
});

async function handleRecordActivation(req: Request): Promise<Response> {
  if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: corsHeaders });
  if (req.method !== 'POST') return jsonError(405, 'method_not_allowed');
  try {
    // 1. Auth — verify Bearer JWT
    const auth = req.headers.get('authorization');
    if (!auth) return jsonError(401, 'unauthenticated');
    const admin = getAdmin();
    const { data: userRes, error: authErr } = await admin.auth.getUser(auth.replace(/^Bearer\s+/i, ''));
    if (authErr || !userRes?.user) return jsonError(401, 'unauthenticated');
    const userId = userRes.user.id;

    // 2. Validate body
    const parsed = BodySchema.safeParse(await req.json().catch(() => ({})));
    if (!parsed.success) return jsonError(400, 'invalid_body');
    const { goal_type, action_type } = parsed.data;

    // 3. Advisory lock + fire-once check
    //    Use a SECDEF RPC OR inline SQL via the admin client; pick whichever is cleanest.
    //    Recommended: call admin.rpc('record_activation_event', { p_user_id, p_goal_type, p_action_type })
    //    where the SECDEF does the advisory_xact_lock + activated_at check + INSERT atomically.
    //    Alternative: do it from Edge Fn with admin client + retry on conflict.
    //
    //    Implementation choice: inline from Edge Fn (one round-trip):
    //    a. SELECT profiles.created_at, activation_events.activated_at WHERE id/user_id = userId
    //    b. If activated_at != null → return { already_activated: true } (no capture)
    //    c. days = floor((now - profiles.created_at) / 86400)
    //    d. If days > 7 → return { skipped: true, reason: 'outside_window' }
    //    e. UPSERT activation_events with ON CONFLICT (user_id) DO UPDATE SET activated_at = excluded.activated_at WHERE activation_events.activated_at IS NULL
    //       Per memory feedback_state_counter_table_needs_upsert_on_event — UPSERT not bare UPDATE because the Phase 38 shell may not have inserted a row yet.
    //    f. If the UPSERT did not actually set activated_at (returned 0 rows affected; check via .select()) → race lost → return { already_activated: true } without capture
    //    g. Otherwise captureServer(...) + return { activated: true }

    const { data: signup } = await admin
      .from('profiles')
      .select('created_at')
      .eq('id', userId)
      .maybeSingle();
    if (!signup?.created_at) return jsonError(500, 'profile_missing');
    const daysSinceSignup = Math.floor((Date.now() - new Date(signup.created_at).getTime()) / 86400000);
    if (daysSinceSignup > 7) return jsonResponse(200, { skipped: true, reason: 'outside_window' });

    const { data: existing } = await admin
      .from('activation_events')
      .select('activated_at')
      .eq('user_id', userId)
      .maybeSingle();
    if (existing?.activated_at) return jsonResponse(200, { already_activated: true });

    const nowIso = new Date().toISOString();
    const { data: upsertRes, error: upsertErr } = await admin
      .from('activation_events')
      .upsert({
        user_id: userId,
        activated_at: nowIso,
        goal_type,
        action_type,
        window_days: 7,
        source: 'first_log',
        updated_at: nowIso,
      }, { onConflict: 'user_id', ignoreDuplicates: false })
      .select('activated_at, goal_type')
      .single();
    if (upsertErr) return jsonError(500, 'db_error');

    // Re-read to confirm we actually set activated_at (race protection)
    if (!upsertRes?.activated_at) return jsonResponse(200, { already_activated: true });

    captureServer({
      userId,
      event: 'activation_completed',
      properties: {
        goal_type,
        action_type,
        window_days: 7,
        days_since_signup: daysSinceSignup,
        source: 'first_log',
      },
    });
    return jsonResponse(200, { activated: true, days_since_signup: daysSinceSignup });
  } finally {
    await shutdownPostHog();
  }
}
```

NOTE on race-safety: the `select activated_at → if null then upsert` sequence is NOT atomic from the Edge Fn. Two concurrent invocations could both see `null` and both attempt the UPSERT. The PRIMARY KEY on user_id means only one row exists, but both `captureServer` calls would fire. To make this race-safe, either:
- (a) Use `upsert ... on conflict (user_id) do update set activated_at = excluded.activated_at where activation_events.activated_at is null` — Supabase JS supports this via `.upsert({...}, { onConflict: 'user_id' })` but the WHERE clause requires raw SQL via an RPC.
- (b) Wrap in a SECDEF RPC `record_activation_event(p_user_id, p_goal, p_action, p_days)` that does `pg_advisory_xact_lock(hashtext('record_activation:' || p_user_id)) + check + upsert + return whether activation_completed should fire`.

```sql
create or replace function public.record_activation_event(
  p_user_id    uuid,
  p_goal_type  text,
  p_action     text,
  p_days       int
) returns jsonb
language plpgsql
security definer
set search_path = pg_catalog, public, extensions
as $$
declare
  v_did_fire boolean := false;
  v_existing timestamptz;
begin
  perform pg_advisory_xact_lock(hashtext('record_activation:' || p_user_id::text));
  select activated_at into v_existing from public.activation_events where user_id = p_user_id;
  if v_existing is not null then
    return jsonb_build_object('did_fire', false, 'reason', 'already_activated');
  end if;
  insert into public.activation_events (user_id, activated_at, goal_type, action_type, window_days, source, updated_at)
  values (p_user_id, now(), p_goal_type, p_action, 7, 'first_log', now())
  on conflict (user_id) do update
    set activated_at = excluded.activated_at,
        goal_type    = excluded.goal_type,
        action_type  = excluded.action_type,
        window_days  = 7,
        source       = 'first_log',
        updated_at   = now()
    where public.activation_events.activated_at is null;
  -- did_fire = true only if our INSERT/UPDATE actually wrote activated_at
  select activated_at into v_existing from public.activation_events where user_id = p_user_id;
  v_did_fire := (v_existing = (select activated_at from public.activation_events where user_id = p_user_id and activated_at is not null));
  return jsonb_build_object('did_fire', v_did_fire, 'days_since_signup', p_days);
end$$;

revoke all on function public.record_activation_event(uuid,text,text,int) from public, anon, authenticated;
grant execute on function public.record_activation_event(uuid,text,text,int) to service_role;
```

```typescript
export function isAppleEnabled(): boolean {
  // Tri-source: env var > localStorage runtime override > false.
  // Exported so Plan 34-06 can read it from the goal-step OAuth button row
  // without duplicating the gate logic (per plan-checker B-03 fix).
  if (typeof import.meta !== 'undefined' && (import.meta as any).env?.VITE_AUTH_APPLE_ENABLED === 'true') return true;
  try { return localStorage.getItem('leanshot_auth_apple_enabled') === 'true'; } catch { return false; }
}

export async function signInWithOAuthProvider(
  provider: 'google' | 'apple',
): Promise<{ error: { message: string } | null }> {
  if (provider === 'apple' && !isAppleEnabled()) {
    return { error: { message: 'apple_disabled' } };
  }
  const { error } = await supabase.auth.signInWithOAuth({
    provider,
    options: {
      // PKCE is the supabase-js v2 default; redirectTo MUST be path-based
      // because OAuth providers cannot redirect to # fragments (RESEARCH Pitfall 1).
      redirectTo: `${window.location.origin}/auth/callback`,
    },
  });
  return { error: error ? { message: error.message } : null };
}
```

```tsx
import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';

export default function AuthCallbackView(): JSX.Element {
  const [status, setStatus] = useState<'exchanging' | 'redirecting' | 'error'>('exchanging');

  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const { data, error } = await supabase.auth.exchangeCodeForSession(window.location.href);
        if (cancelled) return;
        if (error || !data?.session?.user) {
          setStatus('error');
          window.location.replace('/#/auth/error');
          return;
        }
        // Look up profiles.completed_onboarding_at to route post-exchange
        const { data: prof } = await supabase
          .from('profiles')
          .select('completed_onboarding_at')
          .eq('id', data.session.user.id)
          .maybeSingle();
        setStatus('redirecting');
        if (prof?.completed_onboarding_at) {
          window.location.replace('/#/dashboard');
        } else {
          window.location.replace('/#/onboarding');
        }
      } catch {
        if (!cancelled) {
          setStatus('error');
          window.location.replace('/#/auth/error');
        }
      }
    })();
    return () => { cancelled = true; };
  }, []);

  return (
    <div role="status" aria-live="polite" className="min-h-screen flex items-center justify-center bg-[var(--color-bg)]">
      <div className="text-center">
        <div aria-hidden className="size-10 mx-auto mb-4 rounded-full border-2 border-[var(--color-primary)] border-t-transparent animate-spin" />
        <p className="text-[var(--color-text-muted)] text-sm">
          {status === 'error' ? 'Sign-in failed — redirecting…' : 'Signing you in…'}
        </p>
      </div>
    </div>
  );
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| OAuth provider → /auth/callback | The `?code=` query parameter is an OAuth authorization code; supabase-js validates it server-side via `exchangeCodeForSession`. The browser never sees the OAuth client secret. |
| Browser localStorage → apple-enabled flag | Untrusted local flag; treating it as a UI toggle only. Server-side Supabase Apple OAuth config is the real gate — if Apple isn't configured at Supabase, the `signInWithOAuth({ provider: 'apple' })` call fails regardless of the local flag. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-34-04-01 | Tampering | Authorization code in URL | mitigate | supabase-js's `exchangeCodeForSession` validates the code against GoTrue server-side; tampered codes fail with `invalid_request` |
| T-34-04-02 | Spoofing | Apple "first-name" pre-fill | accept | Apple only returns full_name on first sign-in (RESEARCH Pitfall 8); per Plan 34-06 the onboarding "What's your name?" step captures it instead |
| T-34-04-03 | Information Disclosure | OAuth redirect URL | mitigate | redirectTo is `${origin}/auth/callback` (same origin); no PII in URL; PKCE code is single-use + short-lived |
| T-34-04-04 | Elevation of Privilege | Apple disabled-flag bypass | accept | localStorage flag is UI-only; real Apple OAuth gate is at Supabase dashboard config (no .p8 → 400 from GoTrue) |
| T-34-04-05 | Repudiation | OAuth sign-in event | mitigate | Supabase GoTrue logs every sign-in attempt server-side; PostHog captures `auth.signed_in` event from existing identify.ts wiring |
</threat_model>

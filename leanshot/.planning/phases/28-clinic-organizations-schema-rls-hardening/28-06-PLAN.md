    -- Phase 28 Plan 06 — ORG-07 path-based routing RPC.
    -- Anti-enumeration: non-existent slug AND existing-slug-no-relationship both return {state:'not_found'}.
    -- Constant-cost path: single index lookup on organizations.slug + single join.

    create or replace function public.resolve_clinic_slug(p_slug text)
    returns jsonb
    language plpgsql
    stable
    security definer
    set search_path = pg_catalog, public, extensions
    as $$
    declare
      v_org_row record;
      v_member_row record;
      v_invite_row record;
      v_email citext;
    begin
      -- Look up the org by slug (single index lookup).
      select id, slug, name into v_org_row from public.organizations where slug = p_slug limit 1;

      if v_org_row.id is null then
        return jsonb_build_object('state', 'not_found');
      end if;

      -- Check membership for the calling user.
      select role into v_member_row from public.org_members
        where org_id = v_org_row.id and user_id = auth.uid()
        limit 1;

      if v_member_row.role is not null then
        return jsonb_build_object(
          'state', 'member',
          'org_summary', jsonb_build_object(
            'id', v_org_row.id,
            'slug', v_org_row.slug,
            'name', v_org_row.name,
            'role', v_member_row.role
          )
        );
      end if;

      -- Check pending invite for the calling user's email.
      select email into v_email from auth.users where id = auth.uid() limit 1;
      if v_email is null then
        return jsonb_build_object('state', 'not_found');
      end if;

      select id, invited_role, expires_at into v_invite_row from public.org_invites
        where org_id = v_org_row.id
          and email = v_email
          and status = 'pending'
          and expires_at > now()
        limit 1;

      if v_invite_row.id is not null then
        return jsonb_build_object(
          'state', 'pending_invite',
          'invite_summary', jsonb_build_object(
            'invite_id', v_invite_row.id,
            'role', v_invite_row.invited_role,
            'expires_at', v_invite_row.expires_at
          )
        );
      end if;

      -- Slug exists but caller has no relationship — anti-enumeration: return not_found.
      return jsonb_build_object('state', 'not_found');
    end;
    $$;

    revoke execute on function public.resolve_clinic_slug(text) from public, anon;
    grant execute on function public.resolve_clinic_slug(text) to authenticated;
    ```

    import { useEffect, useState } from 'react';
    import { useStore } from '@/lib/store';
    import { supabase } from '@/lib/supabase';
    import { Spinner } from '@/components/ui/Spinner';
    // Imports for existing InviteAcceptance + NotFound components (discovered above).

    type ResolveSlugResult =
      | { state: 'member'; org_summary: { id: string; slug: string; name: string; role: 'admin'|'staff'|'viewer' } }
      | { state: 'pending_invite'; invite_summary: { invite_id: string; role: 'admin'|'staff'|'viewer'; expires_at: string } }
      | { state: 'not_found' };

    export function RouteOrgGuard({ slug, children }: { slug: string; children: React.ReactNode }): React.ReactElement {
      const [result, setResult] = useState<ResolveSlugResult | null>(null);

      useEffect(() => {
        let cancelled = false;
        (async () => {
          const { data, error } = await supabase.rpc('resolve_clinic_slug', { p_slug: slug });
          if (cancelled) return;
          if (error || !data) { setResult({ state: 'not_found' }); return; }
          setResult(data as ResolveSlugResult);
          if ((data as ResolveSlugResult).state === 'member') {
            const r = data as Extract<ResolveSlugResult, {state:'member'}>;
            useStore.getState().setCurrentOrg(
              { id: r.org_summary.id, slug: r.org_summary.slug, name: r.org_summary.name },
              r.org_summary.role,
            );
          }
        })();
        return () => { cancelled = true; useStore.getState().clearCurrentOrg(); };
      }, [slug]);

      if (result === null) return <Spinner size="md" data-testid="route-org-guard-loading" />;
      if (result.state === 'member') return <>{children}</>;
      if (result.state === 'pending_invite') return <InviteAcceptance inviteId={result.invite_summary.invite_id} role={result.invite_summary.role} expiresAt={result.invite_summary.expires_at} />;
      return <NotFound />;
    }
    ```

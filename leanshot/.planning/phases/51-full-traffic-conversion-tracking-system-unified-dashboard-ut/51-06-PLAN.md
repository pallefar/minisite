State + fetch (CAC pattern — native useEffect, no TanStack Query):
```typescript
type Row = {
  channel_group: string;
  audience: string;
  day: string;
  org_id: string | null;
  visits: number; signups: number; activations: number; paids: number;
  ad_spend_usd: number;
  d1_retained_count: number; d7_retained_count: number; d14_retained_count: number;
  d30_retained_count: number; d60_retained_count: number;
  cac_to_activation: number | null; cac_to_paid: number | null;
};

type DateRange = 'today' | '7d' | '30d';
type TouchMode = 'first' | 'last';

const PAID_CHANNELS = new Set(['Paid Search', 'Paid Social']);

export const TrafficChannelsTab: React.FC = () => {
  const [rows, setRows] = useState<Row[] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [dateRange, setDateRange] = useState<DateRange>('7d');
  const [touchMode, setTouchMode] = useState<TouchMode>('last');
  const [drawerChannel, setDrawerChannel] = useState<string | null>(null);
  const [toast, setToast] = useState<string | null>(null);

  const orgFilter = useStore((s) => s.user?.role === 'clinic_owner' ? s.user.org?.id ?? null : null);

  const dateBounds = useMemo(() => {
    const end = new Date();
    const start = new Date();
    if (dateRange === 'today') start.setDate(end.getDate());
    else if (dateRange === '7d') start.setDate(end.getDate() - 7);
    else start.setDate(end.getDate() - 30);
    return { start: start.toISOString().slice(0, 10), end: end.toISOString().slice(0, 10) };
  }, [dateRange]);

  const fetchChannels = useCallback(async () => {
    setLoading(true); setError(null);
    const { data, error: rpcErr } = await supabase.rpc('get_traffic_channel_rollup', {
      p_org_id: orgFilter,
      p_start_date: dateBounds.start,
      p_end_date: dateBounds.end,
    });
    if (rpcErr) {
      if (rpcErr.code === '42501' || rpcErr.message.toLowerCase().includes('permission')) {
        setError('Permission denied — admin role required');
      } else {
        setError(`Failed to load channels — ${rpcErr.message}`);
      }
      setRows(null);
    } else {
      setRows((data as Row[]) ?? []);
    }
    setLoading(false);
  }, [orgFilter, dateBounds.start, dateBounds.end]);

  useEffect(() => { void fetchChannels(); }, [fetchChannels]);

  // Aggregate per channel_group for table (group days within range)
  const tableRows = useMemo(() => {
    if (!rows) return null;
    const m = new Map<string, Row & { sparkline: number[] }>();
    for (const r of rows) {
      const key = r.channel_group;
      const prev = m.get(key);
      if (prev) {
        prev.visits += r.visits; prev.signups += r.signups; prev.activations += r.activations; prev.paids += r.paids;
        prev.ad_spend_usd += Number(r.ad_spend_usd ?? 0);
        prev.d7_retained_count += r.d7_retained_count; prev.d30_retained_count += r.d30_retained_count;
        // sparkline collects per-day visits desc within window
      } else {
        m.set(key, { ...r, sparkline: [r.d1_retained_count, r.d7_retained_count, r.d14_retained_count, r.d30_retained_count, r.d60_retained_count] });
      }
    }
    return Array.from(m.values()).sort((a, b) => b.visits - a.visits);
  }, [rows]);

  return (
    <section>
      <div className="flex items-center justify-between mb-3">
        <h2 className="text-[11px] font-semibold text-[var(--color-text-secondary)] uppercase tracking-[0.06em]">
          Channel Rollup ({dateRange === 'today' ? 'Today' : dateRange === '7d' ? 'Last 7 Days' : 'Last 30 Days'})
        </h2>
        <div className="flex items-center gap-2">
          <PillGroup ariaLabel="Date range">
            <Pill size="sm" aria-pressed={dateRange === 'today'} onClick={() => setDateRange('today')}>Today</Pill>
            <Pill size="sm" aria-pressed={dateRange === '7d'} onClick={() => setDateRange('7d')}>7d</Pill>
            <Pill size="sm" aria-pressed={dateRange === '30d'} onClick={() => setDateRange('30d')}>30d</Pill>
          </PillGroup>
          <PillGroup ariaLabel="Touch attribution mode">
            <Pill size="sm" aria-pressed={touchMode === 'first'} onClick={() => setTouchMode('first')}>First-touch</Pill>
            <Pill size="sm" aria-pressed={touchMode === 'last'} onClick={() => setTouchMode('last')}>Last-touch</Pill>
          </PillGroup>
          <Button variant="ghost" size="sm" leadingIcon={<RefreshCw size={12} aria-hidden />} onClick={() => void fetchChannels()} aria-label="Refresh">Refresh</Button>
        </div>
      </div>

      {loading && <Skeleton className="h-64 rounded-card" />}
      {error && !loading && (
        <Card padding="md"><p className="text-[13px] text-[var(--color-danger)]">{error}</p><Button variant="ghost" size="sm" onClick={() => void fetchChannels()}>Retry</Button></Card>
      )}
      {!loading && !error && tableRows && tableRows.length === 0 && (
        <EmptyState heading="No channel data yet" body="Channel rollup populates after the first matview refresh. Check back in ~15 minutes after first traffic lands." />
      )}
      {!loading && !error && tableRows && tableRows.length > 0 && (
        <Card padding="md">
          <table className="w-full text-[13px]">
            <thead>
              <tr className="text-[11px] font-semibold text-[var(--color-text-secondary)] uppercase tracking-[0.06em] text-left">
                <th>Channel</th>
                <th className="text-right">Visits</th>
                <th className="text-right">Signups</th>
                <th className="text-right">Activations</th>
                <th className="text-right">Paid</th>
                <th className="text-right">D7 Retained</th>
                <th className="text-right">D30 Retained</th>
                <th className="text-right">CAC</th>
                <th></th>
              </tr>
            </thead>
            <tbody>
              {tableRows.map((r) => (
                <tr key={r.channel_group} role="button" tabIndex={0}
                    aria-label={`View ${r.channel_group} retention curve`}
                    className="cursor-pointer hover:bg-[var(--color-surface-elevated)]"
                    onClick={() => setDrawerChannel(r.channel_group)}
                    onKeyDown={(e) => { if (e.key === 'Enter' || e.key === ' ') setDrawerChannel(r.channel_group); }}>
                  <td className="py-2"><span className="numerals-tabular">{r.channel_group}</span></td>
                  <td className="py-2 text-right numerals-tabular">{r.visits.toLocaleString()}</td>
                  <td className="py-2 text-right numerals-tabular">{r.signups.toLocaleString()}</td>
                  <td className="py-2 text-right numerals-tabular">{r.activations.toLocaleString()}</td>
                  <td className="py-2 text-right numerals-tabular">{r.paids.toLocaleString()}</td>
                  <td className="py-2 text-right numerals-tabular">{r.d7_retained_count.toLocaleString()}</td>
                  <td className="py-2 text-right numerals-tabular">{r.d30_retained_count.toLocaleString()}</td>
                  <td className="py-2 text-right numerals-tabular">{r.cac_to_activation != null ? `$${Number(r.cac_to_activation).toFixed(2)}` : '—'}</td>
                  <td className="py-2 text-right"><ChevronRight size={14} className="text-[var(--color-primary)]" aria-hidden /></td>
                </tr>
              ))}
            </tbody>
          </table>
        </Card>
      )}

      {/* Drill-in Sheet (retention curve + CAC deep-link) */}
      <Sheet open={drawerChannel != null} onOpenChange={(open) => !open && setDrawerChannel(null)}>
        {drawerChannel && tableRows && (() => {
          const row = tableRows.find((r) => r.channel_group === drawerChannel);
          if (!row) return null;
          return (
            <div className="p-4">
              <button className="flex items-center gap-1 text-[13px] text-[var(--color-primary)] mb-4 hover:underline" onClick={() => setDrawerChannel(null)} type="button">← Back to channels</button>
              <h3 className="text-[18px] font-bold mb-2">{row.channel_group}</h3>
              <p className="text-[11px] text-[var(--color-text-secondary)] uppercase tracking-[0.06em] mb-3">Retention curve</p>
              <Sparkline values={[row.d1_retained_count, row.d7_retained_count, row.d14_retained_count, row.d30_retained_count, row.d60_retained_count]} />
              <p className="text-[11px] text-[var(--color-text-secondary)] mt-2">
                D1 {row.d1_retained_count.toLocaleString()} • D7 {row.d7_retained_count.toLocaleString()} • D14 {row.d14_retained_count.toLocaleString()} • D30 {row.d30_retained_count.toLocaleString()} • D60 {row.d60_retained_count.toLocaleString()}
              </p>
              {PAID_CHANNELS.has(row.channel_group) && (
                <a href={`/admin/growth/cac?channel=${encodeURIComponent(row.channel_group)}`}
                   className="inline-flex items-center gap-1 text-[13px] text-[var(--color-primary)] mt-4 hover:underline">
                  View ad-spend detail <ExternalLink size={12} aria-hidden />
                </a>
              )}
            </div>
          );
        })()}
      </Sheet>

      {toast && <div role="status" aria-live="polite" className="fixed bottom-4 right-4 bg-[var(--color-surface-elevated)] p-3 rounded-card text-[13px]">{toast}</div>}
    </section>
  );
};
```

**Typography**: only `text-[11px]`, `text-[13px]`, `text-[18px]`. Weights 400 + 700 + 600 (caption-only).

**touchMode toggle implementation note**: in v1.3, the SECDEF accessor reads from `traffic_channel_rollup` which is grouped on `last_touch_channel_group` (Plan 51-03 §matview SQL). To honor first-touch toggle WITHOUT a second matview, the executor has two options:
- **Option A (recommended for v1.3):** disable first-touch column variant — toggle becomes a no-op on the row data, but updates a header caption "(Showing last-touch attribution)". Document as a known v1.3 limitation in SUMMARY.
- **Option B:** add a `p_touch_mode` parameter to `get_traffic_channel_rollup` accessor + matview reads from first_touch_channel_group OR last_touch_channel_group based on param — requires re-spinning Plan 51-03. **DO NOT pursue** unless plan-checker explicitly requests; v1.3 ships Option A.

Pick Option A: leave the toggle in the UI for v1.4 parity but show a small `text-[11px]` caption "(showing last-touch — first-touch coming in v1.4)" when touchMode === 'first'.

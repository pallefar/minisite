**Page body** (UI-SPEC §Page-level + Tabs):
```typescript
type TabKey = 'channels' | 'funnels' | 'landing' | 'realtime' | 'taxonomy';

const TAB_FROM_PATH: Record<string, TabKey> = {
  '/admin/growth/traffic/channels': 'channels',
  '/admin/growth/traffic/funnels': 'funnels',
  '/admin/growth/traffic/landing': 'landing',
  '/admin/growth/traffic/realtime': 'realtime',
  '/admin/growth/traffic/taxonomy': 'taxonomy',
};

function deriveTab(): TabKey {
  const p = window.location.pathname.replace(/\/$/, '');
  return TAB_FROM_PATH[p] ?? 'channels';
}

export const TrafficDashboardPage: React.FC = () => {
  const [activeTab, setActiveTab] = useState<TabKey>(() => deriveTab());

  // URL-driven tab state (no react-router in this project — popstate listener + history.pushState)
  useEffect(() => {
    const onPop = () => setActiveTab(deriveTab());
    window.addEventListener('popstate', onPop);
    return () => window.removeEventListener('popstate', onPop);
  }, []);

  const goTab = useCallback((tab: TabKey) => {
    const target = `/admin/growth/traffic/${tab === 'landing' ? 'landing' : tab}`;
    if (window.location.pathname !== target) {
      window.history.pushState(null, '', target);
    }
    setActiveTab(tab);
  }, []);

  // matview last-refresh — Plan 51-06 wires the real fetch; for shell, render placeholder via useStore + lazy fallback
  const userRole = useStore((s) => s.user?.role ?? 'admin');
  const orgName = useStore((s) => s.user?.org?.name ?? null);

  const subtitle = userRole === 'clinic_owner' && orgName
    ? `Traffic & Conversion — ${orgName}`
    : 'Multi-channel acquisition + funnel intelligence — refreshed just now';

  return (
    <main className="bg-[var(--color-bg)] min-h-full p-6">
      {/* Page header (UI-SPEC §Page-level) */}
      <div className="flex items-center justify-between mb-6">
        <div className="flex items-center gap-3">
          <span className="size-9 rounded-xl bg-[var(--color-surface-elevated)] text-[var(--color-primary)] inline-flex items-center justify-center">
            <Activity size={18} aria-hidden />
          </span>
          <div>
            <h1 className="text-[18px] font-bold">Traffic & Conversion</h1>
            <p className="text-[11px] text-[var(--color-text-secondary)]">{subtitle}</p>
          </div>
        </div>
      </div>

      {/* Tabs */}
      <PillGroup role="tablist" ariaLabel="Traffic dashboard tabs">
        <Pill aria-pressed={activeTab === 'channels'} onClick={() => goTab('channels')} leadingIcon={<LayoutList size={12} aria-hidden />}>Channels</Pill>
        <Pill aria-pressed={activeTab === 'funnels'} onClick={() => goTab('funnels')} leadingIcon={<Filter size={12} aria-hidden />}>Funnels</Pill>
        <Pill aria-pressed={activeTab === 'landing'} onClick={() => goTab('landing')} leadingIcon={<MapIcon size={12} aria-hidden />}>Landing Pages</Pill>
        <Pill aria-pressed={activeTab === 'realtime'} onClick={() => goTab('realtime')} leadingIcon={<Activity size={12} aria-hidden />}>Real-time</Pill>
        <Pill aria-pressed={activeTab === 'taxonomy'} onClick={() => goTab('taxonomy')} leadingIcon={<Settings2 size={12} aria-hidden />}>Taxonomy</Pill>
      </PillGroup>

      {/* Tab content */}
      <div className="pt-8">
        <Suspense fallback={<Skeleton className="h-64 rounded-card" />}>
          {activeTab === 'channels' && <TrafficChannelsTab />}
          {activeTab === 'funnels' && <TrafficFunnelsTab />}
          {activeTab === 'landing' && <TrafficLandingPagesTab />}
          {activeTab === 'realtime' && <TrafficRealtimeTab />}
          {activeTab === 'taxonomy' && <TrafficTaxonomyPage />}
        </Suspense>
      </div>
    </main>
  );
};
```

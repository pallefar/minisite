import { useCallback, useEffect, useMemo, useState } from 'react';
import { RefreshCw } from 'lucide-react';

import { Pill, PillGroup } from '@/components/ui/Pill';

```typescript
type Row = {
  landing_path: string;
  page_variant_id: string | null;
  audience: string;
  day: string;
  org_id: string | null;
  visits: number; signups: number; activations: number; paids: number;
};

type Audience = 'consumer' | 'clinic-org' | 'affiliate' | 'all';
type TopN = 10 | 25 | 50;
type SortKey = 'visits' | 'signup_rate' | 'paid_conv' | 'bounce';

export const TrafficLandingPagesTab: React.FC = () => {
  const [rows, setRows] = useState<Row[] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [audience, setAudience] = useState<Audience>('all');
  const [topN, setTopN] = useState<TopN>(25);
  const [filter, setFilter] = useState('');
  const [sortKey, setSortKey] = useState<SortKey>('visits');
  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');

  const orgFilter = useStore((s) => s.user?.role === 'clinic_owner' ? s.user.org?.id ?? null : null);

  const today = new Date().toISOString().slice(0, 10);
  const sevenAgo = new Date(Date.now() - 7 * 86400000).toISOString().slice(0, 10);

  const fetchLanding = useCallback(async () => {
    setLoading(true); setError(null);
    const { data, error: rpcErr } = await supabase.rpc('get_traffic_landing_page_rollup', {
      p_org_id: orgFilter,
      p_start_date: sevenAgo,
      p_end_date: today,
      p_audience: audience === 'all' ? null : audience,
      p_top_n: topN,
    });
    if (rpcErr) {
      if (rpcErr.code === '42501') setError('Permission denied — admin role required');
      else setError(`Failed to load landing pages — ${rpcErr.message}`);
      setRows(null);
    } else {
      setRows((data as Row[]) ?? []);
    }
    setLoading(false);
  }, [orgFilter, audience, topN, sevenAgo, today]);

  useEffect(() => { void fetchLanding(); }, [fetchLanding]);

  // Aggregate per landing_path × page_variant_id; compute rates; apply filter + sort.
  const tableRows = useMemo(() => {
    if (!rows) return null;
    const agg = new Map<string, Row & { signup_rate: number; paid_conv: number; bounce: number }>();
    for (const r of rows) {
      const key = `${r.landing_path}::${r.page_variant_id ?? '_'}`;
      const prev = agg.get(key);
      if (prev) {
        prev.visits += r.visits; prev.signups += r.signups; prev.activations += r.activations; prev.paids += r.paids;
      } else {
        agg.set(key, { ...r, signup_rate: 0, paid_conv: 0, bounce: 0 });
      }
    }
    const list = Array.from(agg.values()).map((r) => ({
      ...r,
      signup_rate: r.visits > 0 ? r.signups / r.visits : 0,
      paid_conv: r.visits > 0 ? r.paids / r.visits : 0,
      bounce: r.visits > 0 ? Math.max(0, 1 - r.signups / r.visits) : 0,
    }));
    const filtered = filter ? list.filter((r) => r.landing_path.toLowerCase().includes(filter.toLowerCase())) : list;
    filtered.sort((a, b) => {
      const va = a[sortKey] as number;
      const vb = b[sortKey] as number;
      return sortDir === 'desc' ? vb - va : va - vb;
    });
    return filtered.slice(0, topN);
  }, [rows, filter, sortKey, sortDir, topN]);

  const handleSort = (key: SortKey) => {
    if (sortKey === key) setSortDir((d) => d === 'desc' ? 'asc' : 'desc');
    else { setSortKey(key); setSortDir('desc'); }
  };

  const sortArrow = (key: SortKey) => sortKey === key ? (sortDir === 'desc' ? '↓' : '↑') : '';

  return (
    <section>
      <div className="flex items-center justify-between mb-3">
        <h2 className="text-[11px] font-semibold text-[var(--color-text-secondary)] uppercase tracking-[0.06em]">Top Landing Pages</h2>
        <div className="flex items-center gap-2">
          <Input placeholder="Filter by path…" value={filter} onChange={(e) => setFilter(e.target.value)} className="w-48" />
          <PillGroup ariaLabel="Audience">
            <Pill size="sm" aria-pressed={audience === 'all'} onClick={() => setAudience('all')}>All</Pill>
            <Pill size="sm" aria-pressed={audience === 'consumer'} onClick={() => setAudience('consumer')}>Consumer</Pill>
            <Pill size="sm" aria-pressed={audience === 'clinic-org'} onClick={() => setAudience('clinic-org')}>Clinic-org</Pill>
            <Pill size="sm" aria-pressed={audience === 'affiliate'} onClick={() => setAudience('affiliate')}>Affiliate</Pill>
          </PillGroup>
          <PillGroup ariaLabel="Row count">
            <Pill size="sm" aria-pressed={topN === 10} onClick={() => setTopN(10)}>Top 10</Pill>
            <Pill size="sm" aria-pressed={topN === 25} onClick={() => setTopN(25)}>Top 25</Pill>
            <Pill size="sm" aria-pressed={topN === 50} onClick={() => setTopN(50)}>Top 50</Pill>
          </PillGroup>
          <Button variant="ghost" size="sm" leadingIcon={<RefreshCw size={12} aria-hidden />} onClick={() => void fetchLanding()} aria-label="Refresh">Refresh</Button>
        </div>
      </div>

      {loading && <Skeleton className="h-64 rounded-card" />}
      {error && !loading && <Card padding="md"><p className="text-[13px] text-[var(--color-danger)]">{error}</p></Card>}
      {!loading && !error && tableRows && tableRows.length === 0 && (
        <EmptyState heading="No landing-page traffic" body="No visits recorded in this window. Verify lt_anon_id cookie is being set on the landing route." />
      )}
      {!loading && !error && tableRows && tableRows.length > 0 && (
        <Card padding="md">
          <table className="w-full text-[13px]">
            <thead>
              <tr className="text-[11px] font-semibold text-[var(--color-text-secondary)] uppercase tracking-[0.06em] text-left">
                <th>Path</th>
                <th>Variant</th>
                <th className="text-right cursor-pointer" onClick={() => handleSort('visits')} role="button" tabIndex={0}>Visits {sortArrow('visits')}</th>
                <th className="text-right cursor-pointer" onClick={() => handleSort('bounce')} role="button" tabIndex={0}>Bounce % {sortArrow('bounce')}</th>
                <th className="text-right cursor-pointer" onClick={() => handleSort('signup_rate')} role="button" tabIndex={0}>Signup Rate {sortArrow('signup_rate')}</th>
                <th className="text-right cursor-pointer" onClick={() => handleSort('paid_conv')} role="button" tabIndex={0}>Paid Conv {sortArrow('paid_conv')}</th>
              </tr>
            </thead>
            <tbody>
              {tableRows.map((r) => (
                <tr key={`${r.landing_path}::${r.page_variant_id ?? '_'}`}>
                  <td className="py-2"><span className="numerals-tabular">{r.landing_path}</span></td>
                  <td className="py-2 numerals-tabular">{r.page_variant_id ? r.page_variant_id.slice(0, 8) : '—'}</td>
                  <td className="py-2 text-right numerals-tabular">{r.visits.toLocaleString()}</td>
                  <td className="py-2 text-right numerals-tabular">{(r.bounce * 100).toFixed(1)}%</td>
                  <td className="py-2 text-right numerals-tabular">{(r.signup_rate * 100).toFixed(1)}%</td>
                  <td className="py-2 text-right numerals-tabular">{(r.paid_conv * 100).toFixed(1)}%</td>
                </tr>
              ))}
            </tbody>
          </table>
        </Card>
      )}
    </section>
  );
};
```

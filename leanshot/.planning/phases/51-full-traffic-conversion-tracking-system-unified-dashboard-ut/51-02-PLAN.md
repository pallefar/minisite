```typescript
import { createClient } from 'npm:@supabase/supabase-js@2';
import type { SupabaseClient } from 'npm:@supabase/supabase-js@2';
import { captureServer } from './posthog-server.ts';

let _admin: SupabaseClient | null = null;
function getAdmin(): SupabaseClient | null {
  if (_admin) return _admin;
  const url = Deno.env.get('SUPABASE_URL');
  const key = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY');
  if (!url || !key) {
    console.warn('[traffic-attribution] SUPABASE_URL/SERVICE_ROLE_KEY missing — recordTouch is a no-op');
    return null;
  }
  _admin = createClient(url, key, { auth: { persistSession: false } });
  return _admin;
}

export interface RecordTouchArgs {
  anonId: string;
  orgId?: string | null;
  userId?: string | null;
  utm: { source?: string; medium?: string; campaign?: string; term?: string; content?: string };
  referrer: string | null;
  landingPath: string;
  pageVariantId?: string | null;
  audience: 'consumer' | 'clinic-org' | 'affiliate';
  now: Date;
}

export async function recordTouch(args: RecordTouchArgs): Promise<{ ok: true; channelGroup: string } | { ok: false; error: string }> {
  const admin = getAdmin();
  if (!admin) return { ok: false, error: 'admin_unavailable' };

  // Extract referrer host (null-safe)
  let refHost: string | null = null;
  if (args.referrer) {
    try { refHost = new URL(args.referrer).host.toLowerCase(); } catch { /* malformed referrer — leave null */ }
  }

  // Server-side classification via classify_channel_group_with_referrer (Plan 51-01)
  const { data: cg, error: cgErr } = await admin.rpc('classify_channel_group_with_referrer', {
    p_utm_source: args.utm.source ?? null,
    p_utm_medium: args.utm.medium ?? null,
    p_utm_campaign: args.utm.campaign ?? null,
    p_referrer_host: refHost,
  });
  if (cgErr) return { ok: false, error: `classify_failed:${cgErr.message}` };
  const channelGroup = (cg as string | null) ?? 'Direct';

  // Authoritative SQL UPSERT
  const { error: rpcErr } = await admin.rpc('upsert_traffic_attribution', {
    p_anon_id: args.anonId,
    p_org_id: args.orgId ?? null,
    p_utm_source: args.utm.source ?? null,
    p_utm_medium: args.utm.medium ?? null,
    p_utm_campaign: args.utm.campaign ?? null,
    p_utm_term: args.utm.term ?? null,
    p_utm_content: args.utm.content ?? null,
    p_referrer: args.referrer,
    p_landing_path: args.landingPath,
    p_channel_group: channelGroup,
    p_now: args.now.toISOString(),
  });
  if (rpcErr) return { ok: false, error: `upsert_failed:${rpcErr.message}` };

  // PostHog mirror + events_mirror dual-write via captureServer
  captureServer({
    userId: args.userId ?? args.anonId,
    event: 'traffic_visit',
    properties: {
      channel_group: channelGroup,
      audience: args.audience,
      landing_path: args.landingPath,
      page_variant_id: args.pageVariantId ?? null,
      $set: {
        last_touch_channel_group: channelGroup,
        last_touch_source: args.utm.source ?? null,
        last_touch_medium: args.utm.medium ?? null,
        last_touch_at: args.now.toISOString(),
      },
      $set_once: {
        first_touch_channel_group: channelGroup,
        first_touch_source: args.utm.source ?? null,
        first_touch_medium: args.utm.medium ?? null,
        first_touch_at: args.now.toISOString(),
      },
    },
  });

  return { ok: true, channelGroup };
}
```

```typescript
import { corsHeaders, jsonError, jsonResponse } from '../_shared/lifecycle-utils.ts';
import { recordTouch } from '../_shared/traffic-attribution.ts';
import { shutdownPostHog } from '../_shared/posthog-server.ts';

const ALLOWED_ORIGINS = new Set([
  'https://app.leanshot.app',
  'https://leanshot.app',
  'https://www.leanshot.app',
]);

const PHI_PATH_REGEX = /^\/(patient|dose-log)(\/|$)|^\/clinic\/[^/]+\/patient\//i;

function clamp(s: string | null | undefined, n: number): string | null {
  if (s == null) return null;
  return s.length > n ? s.slice(0, n) : s;
}

function redactPath(p: string): string {
  return PHI_PATH_REGEX.test(p) ? '/[redacted]' : p;
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response(null, { headers: corsHeaders() });
  if (req.method !== 'POST') return jsonError(405, 'method_not_allowed');

  // Origin gate (Path 1: SPA fires same-origin from app.leanshot.app)
  const origin = req.headers.get('origin') ?? '';
  if (!ALLOWED_ORIGINS.has(origin)) return jsonError(403, 'origin_denied');

  try {
    const body = await req.json();
    const anonId = clamp(body.anonId, 64);
    if (!anonId) return jsonError(400, 'anon_id_required');

    const utm = {
      source: clamp(body.utm?.source, 2048) ?? undefined,
      medium: clamp(body.utm?.medium, 2048) ?? undefined,
      campaign: clamp(body.utm?.campaign, 2048) ?? undefined,
      term: clamp(body.utm?.term, 2048) ?? undefined,
      content: clamp(body.utm?.content, 2048) ?? undefined,
    };
    const referrer = clamp(body.referrer, 4096);
    const landingPath = redactPath(clamp(body.landingPath, 2048) ?? '/');
    const audience = ['consumer','clinic-org','affiliate'].includes(body.audience) ? body.audience : 'consumer';

    const result = await recordTouch({
      anonId,
      orgId: body.orgId ?? null,
      utm,
      referrer,
      landingPath,
      pageVariantId: clamp(body.pageVariantId, 64),
      audience,
      now: new Date(),
    });

    if (!result.ok) {
      console.warn('[traffic-recorder] recordTouch failed:', result.error);
      return jsonResponse(200, { ok: false, error: result.error });  // 200 because middleware is fire-and-forget
    }

    return jsonResponse(200, { ok: true, channel_group: result.channelGroup });
  } catch (err) {
    console.warn('[traffic-recorder] handler exception:', err);
    return jsonResponse(200, { ok: false, error: 'exception' });  // never throw to caller
  } finally {
    await shutdownPostHog();
  }
});
```

**leanshot/src/lib/traffic/fire-touch.ts:**

```typescript
// Reads lt_anon_id cookie + utm params + referrer + landing path,
// POSTs once per page load to the recorder Edge Fn.
// Idempotent via module-scoped sent flag; fire-and-forget; never throws.

let _sent = false;

function readCookie(name: string): string | null {
  if (typeof document === 'undefined') return null;
  const m = document.cookie.match(new RegExp('(?:^|;\\s*)' + name + '=([^;]+)'));
  return m ? decodeURIComponent(m[1]!) : null;
}

function readUtm(search: string): Record<string, string> {
  const out: Record<string, string> = {};
  if (!search) return out;
  const sp = new URLSearchParams(search);
  for (const key of ['utm_source', 'utm_medium', 'utm_campaign', 'utm_term', 'utm_content']) {
    const v = sp.get(key);
    if (v) out[key.replace('utm_', '')] = v;
  }
  return out;
}

export interface FireTouchOptions {
  // For unit tests; production reads from window/document.
  fetchImpl?: typeof fetch;
  recorderUrl?: string;
  windowOverride?: { location: { search: string; pathname: string }; document?: { cookie: string; referrer: string } };
  resetSentForTest?: boolean;
}

export function _resetSentFlagForTest(): void { _sent = false; }

export async function fireTouchOnce(opts: FireTouchOptions = {}): Promise<void> {
  if (opts.resetSentForTest) _sent = false;
  if (_sent) return;
  _sent = true;

  const fetchFn = opts.fetchImpl ?? (typeof fetch !== 'undefined' ? fetch : null);
  if (!fetchFn) return; // SSR or test without fetch
  const w = opts.windowOverride ?? (typeof window !== 'undefined' ? window : null);
  if (!w) return;

  const anonId = readCookie('lt_anon_id');
  if (!anonId) return; // middleware not yet deployed / private-mode browsers — bail

  const url = opts.recorderUrl ?? ((import.meta.env?.VITE_SUPABASE_URL ?? '') + '/functions/v1/traffic-attribution-recorder');
  if (!url || url.startsWith('/functions')) return; // VITE_SUPABASE_URL not configured — bail silently

  const payload = {
    anonId,
    utm: readUtm(w.location.search),
    referrer: (w as { document?: { referrer?: string } }).document?.referrer ?? null,
    landingPath: w.location.pathname,
    audience: 'consumer' as const,
  };

  try {
    await fetchFn(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
      keepalive: true,
      credentials: 'include', // ensure cookie flows for origin verification
    });
  } catch {
    // fire-and-forget — never throw to caller
  }
}
```

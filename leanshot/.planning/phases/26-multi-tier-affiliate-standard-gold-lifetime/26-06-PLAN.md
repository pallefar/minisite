interface SubRow {
  id: string;                            // stripe subscription id
  plan_id: string | null;
  status: string;
  affiliate_conversions: Array<{
    affiliate_id: string;
    tier_at_conversion_time: string;
  }>;
  // Joined affiliate (frozen check):
  // Done via separate query OR nested embed — Supabase REST supports affiliates(frozen_at)
}

async function processLifetimeRecurring(): Promise<{ processed: number; skipped: number; errors: number }> {
  let processed = 0, skipped = 0, errors = 0;

  // 1. Query lifetime-tier conversions joined to active subscriptions and non-frozen affiliates,
  //    in a SINGLE query (no per-row admin round-trip). Embed direction:
  //      affiliate_conversions.subscription_id → subscriptions(id)
  //      affiliate_conversions.affiliate_id    → affiliates(id)
  //    so the embed aliases match exactly.
  // Pitfall 2: subscriptions.status='active' (raw Stripe value).
  const { data: convs, error: queryErr } = await admin
    .from('affiliate_conversions')
    .select(`
      id,
      subscription_id,
      affiliate_id,
      affiliates!inner ( id, frozen_at ),
      subscriptions!inner ( id, plan_id, status )
    `)
    .eq('tier_at_conversion_time', 'lifetime')
    .eq('subscriptions.status', 'active')
    .is('affiliates.frozen_at', null);
  if (queryErr) {
    return { processed: 0, skipped: 0, errors: 1 };
  }

  const now = new Date();
  const yyyymm = now.getUTCFullYear() * 100 + (now.getUTCMonth() + 1);  // e.g. 202705
  const eligibleAt = new Date(now.getTime() + 60 * 86400000).toISOString();

  for (const conv of (convs ?? []) as Array<{
    id: string;
    subscription_id: string;
    affiliate_id: string;
    affiliates: { id: string; frozen_at: string | null };
    subscriptions: { id: string; plan_id: string | null; status: string };
  }>) {
    try {
      const affiliateId = conv.affiliate_id;
      const sub = conv.subscriptions;
      // Frozen filtering already happens in the query above (.is('affiliates.frozen_at', null)).

      // 2. Compute commission from current Stripe price.
      // Use sub.plan_id (snapshot) as primary source; fall back to live stripe.subscriptions.retrieve
      // ONLY if plan_id is null or price needs verification (rare).
      let grossCents = 0;
      if (sub.plan_id) {
        const price = await getStripe().prices.retrieve(sub.plan_id);
        grossCents = price.unit_amount ?? 0;
      }
      if (grossCents === 0) { skipped++; continue; }

      const commissionCents = Math.round(grossCents * 0.25);  // D-01 lifetime rate
      const idemKey = `${affiliateId}|${sub.id}|${yyyymm}`;

      // 3. INSERT ON CONFLICT DO NOTHING — idempotency anchor (D-07).
      const { error: ledgerErr } = await admin
        .from('affiliate_lifetime_recurring_payments')
        .insert({
          affiliate_id: affiliateId,
          stripe_subscription_id: sub.id,
          billing_period_yyyymm: yyyymm,
          gross_subscription_cents: grossCents,
          commission_cents: commissionCents,
          idempotency_key: idemKey,
        }, { count: 'exact' });

      // 23505 (UNIQUE violation) = no-op (cron retry).
      if (ledgerErr && (ledgerErr as { code?: string }).code !== '23505') {
        errors++;
        continue;
      }
      if (ledgerErr && (ledgerErr as { code?: string }).code === '23505') {
        skipped++;
        continue;
      }

      // 4. INSERT affiliate_conversions row (BEFORE INSERT trigger from Plan 26-01 stamps tier).
      // Synthetic invoice_id enforces idempotency at conversion layer too.
      const invoiceId = `lifetime_recurring_${affiliateId}_${sub.id}_${yyyymm}`;
      const { error: convErr } = await admin
        .from('affiliate_conversions')
        .insert({
          affiliate_id: affiliateId,
          subscription_id: sub.id,
          invoice_id: invoiceId,
          commission_cents: commissionCents,
          status: 'pending',
          eligible_at: eligibleAt,
        });
      // 23505 here is also fine — same idempotency.
      if (convErr && (convErr as { code?: string }).code !== '23505') {
        errors++;
        continue;
      }

      processed++;
    } catch (_e) {
      errors++;
    }
  }

  return { processed, skipped, errors };
}

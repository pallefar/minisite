```typescript
import { openDB, type IDBPDatabase } from 'idb';

const DB_NAME = 'leanshot_photo_queue';
const STORE_NAME = 'blobs';
const DB_VERSION = 1;

let dbPromise: Promise<IDBPDatabase> | null = null;

function getDb(): Promise<IDBPDatabase> {
  if (!dbPromise) {
    dbPromise = openDB(DB_NAME, DB_VERSION, {
      upgrade(db) {
        if (!db.objectStoreNames.contains(STORE_NAME)) {
          db.createObjectStore(STORE_NAME); // out-of-line keys
        }
      },
    });
  }
  return dbPromise;
}

export async function putPhotoBlob(key: string, blob: Blob): Promise<void> {
  const db = await getDb();
  await db.put(STORE_NAME, blob, key);
}

export async function getPhotoBlob(key: string): Promise<Blob | undefined> {
  const db = await getDb();
  return db.get(STORE_NAME, key);
}

export async function deletePhotoBlob(key: string): Promise<void> {
  const db = await getDb();
  await db.delete(STORE_NAME, key);
}

export async function listPhotoBlobKeys(): Promise<string[]> {
  const db = await getDb();
  const keys = await db.getAllKeys(STORE_NAME);
  return keys as string[];
```

```typescript
async function flushPhotoOps(): Promise<void> {
  const state = useStore.getState();
  if (!isSyncEnabled()) return;
  const uid = state.signedIn?.userId;
  if (!uid) return;

  const photoOps = (state.pendingOps ?? []).filter((op) => op.table === 'photos');

  // D-09: serial drain, one at a time
  for (const op of photoOps) {
    try {
      if (op.op === 'upload') {
        const { getPhotoBlob, deletePhotoBlob } = await import('@/lib/photo-queue');
        const blob = await getPhotoBlob(op.blob_ref ?? op.key);
        if (!blob) { state.dropOps([op.key], 'photos'); continue; }

        const photo = state.photos.find((p) => p.photo_id === op.key);
        if (!photo) { state.dropOps([op.key], 'photos'); continue; }

        const path = photo.storage_path || `${uid}/photos/${op.key}.jpg`;
        const { error: upErr } = await supabase.storage.from('photos').upload(path, blob, {
          contentType: photo.mime_type,
          upsert: true,
        });
        if (upErr) {
          if (isPermanent4xx(upErr as any)) {
            state.dropOps([op.key], 'photos');
            console.warn('[leanshot] photo upload permanent error', upErr.message);
          }
          continue; // transient — leave queue intact
        }

        // Upload OK — insert the metadata row + drop local blob
        const { error: insErr } = await supabase.from('photos').upsert(
          { user_id: uid, photo_id: photo.photo_id, date: photo.date, weight: photo.weight, storage_path: path, mime_type: photo.mime_type, size_bytes: photo.size_bytes },
          { onConflict: 'user_id,photo_id' }
        );
        if (insErr) {
          if (isPermanent4xx(insErr)) state.dropOps([op.key], 'photos');
          continue;
        }
        state.dropOps([op.key], 'photos');
        await deletePhotoBlob(op.blob_ref ?? op.key); // free IDB space
      } else if (op.op === 'delete') {
        // 06-RESEARCH §2.4 — table delete first, then Storage object delete
        const photo = state.photos.find((p) => p.photo_id === op.key);
        const { error: dbErr } = await supabase.from('photos').delete().eq('user_id', uid).eq('photo_id', op.key);
        if (dbErr) {
          if (isPermanent4xx(dbErr)) state.dropOps([op.key], 'photos');
          continue;

          // Failure to remove the Storage object is logged but doesn't block dequeue (orphan recoverable later)
          await supabase.storage.from('photos').remove([photo.storage_path]).catch((e) => {
            console.warn('[leanshot] photo storage delete failed (orphan)', e);
          });
        }

      } else if (op.op === 'upsert') {
        // Metadata-only update (e.g., editing weight on a photo) — no Storage interaction

      }
    } catch (e) {
      console.error(`[leanshot] flushPhotoOps error for op ${op.key}`, e);
      // Transient — leave queue; next flush re-tries
    }
  }
```

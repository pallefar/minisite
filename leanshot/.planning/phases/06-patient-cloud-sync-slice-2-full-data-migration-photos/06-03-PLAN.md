**Step 1: Extract `flushTableOps<TLocal>` generic helper in sync.ts.**

```typescript
async function flushTableOps<TLocal extends Record<string, unknown>>(
  tableName: string,
  primaryKeyField: string,
  mapLocalToServer: (local: TLocal) => Record<string, unknown>,
): Promise<void> {
  const state = useStore.getState();
  if (!isSyncEnabled()) return;
  const uid = state.signedIn?.userId;
  if (!uid) return;

  const ops = (state.pendingOps ?? []).filter((op) => op.table === tableName);
  if (ops.length === 0) return;

  // Split upsert vs delete
  const upsertOps = ops.filter((o) => o.op === 'upsert');
  const deleteOps = ops.filter((o) => o.op === 'delete');

  if (upsertOps.length > 0) {
    const localRows = (state as any)[tableName] as TLocal[] | undefined; // typed cast per state shape
    const matching = (localRows ?? []).filter((r) => upsertOps.some((o) => o.key === String((r as any)[primaryKeyField])));
    const rows = matching.map((r) => ({
      user_id: uid,
      ...mapLocalToServer(r),
      // INTENTIONALLY OMITTING updated_at (server moddatetime trigger)
    }));
    const { error } = await supabase.from(tableName).upsert(rows, {
      onConflict: tableName === 'supplements' ? 'user_id,date,supplement_name'
        : tableName === 'settings' ? 'user_id'
        : `user_id,${primaryKeyField}`,
    });
    if (error) {
      if (isPermanent4xx(error)) state.dropOps(upsertOps.map((o) => o.key));
      else console.warn(`[leanshot] transient error on ${tableName} upsert`, error.message);
    } else {
      state.dropOps(upsertOps.map((o) => o.key));
    }
  }

  if (deleteOps.length > 0) {
    for (const op of deleteOps) {
      const filters = tableName === 'supplements'
        ? (() => {
            const [date, name] = op.key.split(':');
            return supabase.from('supplements').delete().eq('user_id', uid).eq('date', date).eq('supplement_name', name);
          })()
        : tableName === 'settings'
        ? supabase.from('settings').delete().eq('user_id', uid)
        : supabase.from(tableName).delete().eq('user_id', uid).eq(primaryKeyField, op.key);
      const { error } = await filters;
      if (error) {
        if (isPermanent4xx(error)) state.dropOps([op.key]);
        else console.warn(`[leanshot] transient error on ${tableName} delete`, error.message);
      } else {
        state.dropOps([op.key]);
      }
    }
  }
}
```

```typescript
// src/lib/sync-defer.ts
/**
 * Sync deferral helper (Phase 6 D-12 CI hardening).
 *
 * Phase 5 wired @/lib/sync, @/lib/auth-migration, and (transitively)
 * @supabase/supabase-js into App.tsx's static graph. This pushed the entry
 * chunk gzip past the 50 kB Phase 2.1 ceiling. Phase 6 D-12 inverts the
 * dependency: App.tsx imports THIS module's tiny pre-init buffer + dynamic-
 * import scheduler, and the heavy @/lib/sync module loads after first paint.
 */
import type { Session } from '@supabase/supabase-js';

type SyncCall =
  | { kind: 'onSignedIn'; userId: string; session: Session }
  | { kind: 'onSignedOut'; prevUserId: string | null }
  | { kind: 'flush' };

const BUFFER_CAP = 64;
const buffer: SyncCall[] = [];
let loadedApi: {
  sync: typeof import('@/lib/sync');
  authMig: typeof import('@/lib/auth-migration');
} | null = null;

async function loadSync(): Promise<NonNullable<typeof loadedApi>> {
  const [sync, authMig] = await Promise.all([
    import('@/lib/sync'),
    import('@/lib/auth-migration'),
  ]);
  return { sync, authMig };
}

function drain(): void {
  if (!loadedApi) return;
  while (buffer.length > 0) {
    const call = buffer.shift()!;
    try {
      if (call.kind === 'onSignedIn') {
        void loadedApi.sync.pullInitialInjections(call.userId);
        loadedApi.sync.subscribeInjections(call.userId);
        void loadedApi.sync.flushSyncQueue();
      } else if (call.kind === 'onSignedOut') {
        void loadedApi.sync.unsubscribeInjections();
      } else if (call.kind === 'flush') {
        void loadedApi.sync.flushSyncQueue();
      }
    } catch (e) {
      console.error('[leanshot] deferred sync call failed', call.kind, e);
    }
  }
}

function push(call: SyncCall): void {
  if (loadedApi) {
    buffer.push(call);
    drain();
    return;
  }
  if (buffer.length >= BUFFER_CAP) {
    console.warn('[leanshot] sync-defer buffer full; dropping oldest');
    buffer.shift();
  }
  buffer.push(call);
}

export function deferOnSignedIn(userId: string, session: Session): void {
  push({ kind: 'onSignedIn', userId, session });
}

export function deferOnSignedOut(prevUserId: string | null): void {
  push({ kind: 'onSignedOut', prevUserId });
}

export function deferFlush(): void {
  push({ kind: 'flush' });
}

export function scheduleSyncInit(): void {
  const startLoad = (): void => {
    void loadSync().then((api) => {
      loadedApi = api;
      drain();
    });
  };
  if ('requestIdleCallback' in window) {
    window.requestIdleCallback(startLoad, { timeout: 2000 });
  } else {
    setTimeout(startLoad, 100);
  }
}

/** Test-only reset hook. */
export function _resetForTests(): void {
  buffer.length = 0;
  loadedApi = null;
}
```

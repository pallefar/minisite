    ```sql
    -- Phase 7 D-03: 30-day soft-delete bookkeeping. The single row tracks one
    -- pending deletion per user; T+30 finalize cron reads target_shred_at.
    create table public.pending_account_deletions (
      user_id uuid primary key references auth.users(id) on delete cascade,
      initiated_at timestamptz not null default now(),
      photos_moved_at timestamptz,
      finalize_attempts int not null default 0
    );

    alter table public.pending_account_deletions enable row level security;

    -- Owner can read their own pending row (so the SPA can show "Deletion scheduled for X").
    create policy "pending_account_deletions_select_own"
      on public.pending_account_deletions
      for select to authenticated
      using (auth.uid() = user_id);

    -- NO insert/update/delete policies for authenticated role.
    -- Only the SECURITY DEFINER RPC + service-role bypass.

    create index pending_account_deletions_shred_idx
      on public.pending_account_deletions ((initiated_at + interval '30 days'))
      where finalize_attempts < 5;  -- partial index for cron worker
    ```

    **1B — 20260601000011_initiate_account_deletion_rpc.sql:**
    ```sql
    -- Phase 7 D-03 T+0 RPC. Auth model: auth.uid() ONLY — never accept a user_id
    -- parameter (T-07-07-S2 elevation-of-privilege mitigation).
    create or replace function public.initiate_account_deletion()
    returns void
    language plpgsql
    security definer
    set search_path = public, auth
    as $$
    declare
      v_uid uuid := auth.uid();
      v_last_sign_in timestamptz;
      v_email text;
    begin
      if v_uid is null then
        raise exception 'not authenticated' using errcode = '28000';
      end if;

      -- Spoofing mitigation (T-07-07-S1): require re-auth within 5 minutes.
      select last_sign_in_at, email
        into v_last_sign_in, v_email
      from auth.users where id = v_uid;

      if v_last_sign_in is null or v_last_sign_in < now() - interval '5 minutes' then
        raise exception 'recent_auth_required' using errcode = 'P0007';
      end if;

      -- Idempotent double-click guard.
      if exists (select 1 from public.pending_account_deletions where user_id = v_uid) then
        raise exception 'already_pending' using errcode = 'P0008';
      end if;

      -- T+0 row (audit-skeleton trigger from 07-08 fires on this INSERT and
      -- writes audit_logs(action='account_deleted_initiated', user_id_hash=sha256(uid), ...)).
      insert into public.pending_account_deletions (user_id) values (v_uid);

      -- Move photos to crypto-locked prefix.
      -- Free-tier "crypto-shred" = move objects to a prefix whose Storage RLS
      -- denies all reads (07-07-1E policy); T+30 hard-delete renders bytes
      -- unrecoverable per the AES-256-at-rest invariant (RESEARCH §6 lines 405-411).
      update storage.objects
         set name = 'photos-pending-shred/' || v_uid::text || '/' ||
                    substring(name from position('/photos/' in name) + length('/photos/'))
       where bucket_id = 'photos'
         and name like v_uid::text || '/photos/%';

      update public.pending_account_deletions
         set photos_moved_at = now()
       where user_id = v_uid;

      -- Sign out all sessions for this user.
      delete from auth.sessions where user_id = v_uid;
    end;
    $$;

    revoke all on function public.initiate_account_deletion() from public;
    grant execute on function public.initiate_account_deletion() to authenticated;
    ```

    **1C — 20260601000012_finalize_account_deletion_fn.sql:**
    ```sql
    -- Phase 7 D-03 T+30 finalize. Idempotent + chunked-safe.
    create or replace function public.finalize_account_deletion(p_user_id uuid)
    returns void
    language plpgsql
    security definer
    set search_path = public, auth
    as $$
    declare
      v_row public.pending_account_deletions;
      v_email text;
    begin
      -- Tampering mitigation (T-07-07-T1): schema validation + window check.
      select * into v_row from public.pending_account_deletions where user_id = p_user_id;
      if not found then
        raise exception 'no pending row for %', p_user_id using errcode = 'P0006';
      end if;

      if v_row.initiated_at + interval '30 days' > now() then
        raise exception 'shred window not elapsed (initiated_at=%)', v_row.initiated_at
          using errcode = 'P0009';
      end if;

      -- Audit-skeleton finalize row — written BEFORE auth.users delete so the
      -- trigger from 07-08 has user_id context. audit_logs.user_id is
      -- "on delete set null" so the row survives the cascade with user_id_hash intact.
      insert into public.audit_logs (
        user_id, user_id_hash, table_name, action,
        before_hash, after_hash, ip_hash
      ) values (
        p_user_id,
        encode(digest(p_user_id::text, 'sha256'), 'hex'),
        'auth.users',
        'account_deleted_finalized',
        null, null, null
      );

      -- Hard-delete Storage objects under the pending-shred prefix.
      delete from storage.objects
       where bucket_id = 'photos'
         and name like 'photos-pending-shred/' || p_user_id::text || '/%';

      -- Cascade-delete everything else via auth.users FK cascade.
      -- This single statement purges all 12 child tables in one transaction:
      -- injections, weights, meals, workouts, supplements, mood, sleep,
      -- symptoms, vials, settings, ai_messages, rate_limit_counters, photos,
      -- pending_account_deletions (self).
      delete from auth.users where id = p_user_id;
    exception
      when others then
        -- Increment retry counter; cron will pick up on next tick (idempotent).
        update public.pending_account_deletions
    ```

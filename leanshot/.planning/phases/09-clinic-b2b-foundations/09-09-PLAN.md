    /** Create an operator (auth.users row) + their first org via create_org RPC. */
    export async function createOperatorWithOrg(slugPrefix: string): Promise<{
      operator: { id: string; email: string; password: string };
      orgId: string;
      slug: string;
    }> {
      const id = testRunId();
      const email = `op-${id}@test.leanshot.app`;
      const password = `Op${id}Pass!1`;
      const { data: createResp, error } = await admin.auth.admin.createUser({
        email, password, email_confirm: true,
      });
      if (error || !createResp.user) throw new Error(`createUser: ${error?.message}`);
      const operator = { id: createResp.user.id, email, password };

      // Sign in as operator to set auth.uid() for the RPC
      const operatorClient = createClient(SUPABASE_URL, process.env.SUPABASE_ANON_KEY!);
      await operatorClient.auth.signInWithPassword({ email, password });

      const slug = `${slugPrefix}-${id}`;
      const { data: orgResp, error: orgErr } = await operatorClient.rpc('create_org', {
        p_name: `Test ${slugPrefix}`, p_slug: slug, p_description: null, p_website_url: null,
      });
      if (orgErr || !orgResp) throw new Error(`create_org: ${orgErr?.message}`);
      const orgId = (orgResp as any)[0].org_id;
      return { operator, orgId, slug };
    }

    /** Create an invite + extract the raw token from the invites row (Resend stubbed). */
    export async function createInvite(args: {
      orgId: string;
      operatorId: string;
      patientEmail: string;
      requestedScope?: Partial<Record<string, boolean>>;
    }): Promise<{ inviteId: string; rawToken: string; tokenHash: string }> {
      const tokenBytes = crypto.getRandomValues(new Uint8Array(16));
      const rawToken = Buffer.from(tokenBytes).toString('base64url');
      const tokenHash = require('crypto').createHash('sha256').update(rawToken).digest('hex');
      const scope = args.requestedScope ?? { injections: true, weights: true, photos: true, symptoms: true, meals: true, workouts: true, supplements: true, mood: true, sleep: true, doctor_report: true };

      const { data, error } = await admin.rpc('send_invite', {
        p_email: args.patientEmail.toLowerCase().trim(),
        p_org_id: args.orgId,
        p_invite_token_hash: tokenHash,
        p_requested_scope: scope,
      });
      if (error) throw error;
      return { inviteId: (data as any)[0].invite_id, rawToken, tokenHash };
    }

    /** Cleanup test data — call in afterAll. */
    export async function cleanupClinicFixtures(emailPattern: string): Promise<void> {
      const { data: users } = await admin.auth.admin.listUsers();
      for (const u of users?.users ?? []) {

          await admin.auth.admin.deleteUser(u.id);
        }
      }
      await admin.from('orgs').delete().like('slug', `%${emailPattern}%`);
    }

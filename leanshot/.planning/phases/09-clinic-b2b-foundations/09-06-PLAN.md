    async function sha256Hex(input: string): Promise<string> {
      const bytes = new TextEncoder().encode(input);
      const hash = await crypto.subtle.digest('SHA-256', bytes);
      return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2, '0')).join('');
    }

    Deno.serve(async (req: Request) => {
      if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders });
      const url = new URL(req.url);
      const pathSuffix = url.pathname.replace(/^\/clinic-invite/, '');
      try {
        if (req.method === 'POST' && pathSuffix === '/send') return await handleSend(req);
        if (req.method === 'GET' && pathSuffix === '/lookup') return await handleLookup(req);
        if (req.method === 'POST' && pathSuffix === '/accept') return await handleAccept(req);
        if (req.method === 'POST' && pathSuffix === '/reject') return await handleReject(req);
        return jsonError(404, 'not_found');
      } catch (e) {
        console.error('clinic-invite error', e);
        return jsonError(500, 'internal_error');
      }
    });

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| anonymous → /clinic-invite/lookup | Untrusted token; hash compared; rate-limited 10/min/IP (brute-force defense T-09-32) |
| operator JWT → /clinic-invite/send | JWT validated by admin.auth.getUser; has_permission(members.invite) gate; rate-limited 20/hour/operator |
| patient JWT → /clinic-invite/accept | JWT-authenticated; RPC re-validates email match; UNIQUE(user_id, org_id) partial index single-identity floor |
| Edge Function → Resend HTTPS | Server-only RESEND_API_KEY in Deno.env; key NEVER reaches browser |
| CORS surface | Allow `*` Origin; NO credentials (Pitfall #11) — Phase 8 cookie quirks do NOT apply |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-32 | Spoofing | brute-force invite token guess | mitigate | 128-bit entropy + 10/min/IP rate-limit on lookup; even at 1B guesses/sec for 100 years = 10^17 guesses; infeasible |
| T-09-33 | Information disclosure | email enumeration via /send | mitigate | D-02: send always returns universal 200; auth.users lookup deferred to /accept |
| T-09-34 | Information disclosure | Resend API key leak in logs | mitigate | Key in Deno.env, never echoed; non-2xx Resend response wraps as `{error:'resend_<status>'}` — never echo full body (Phase 4 pattern) |
| T-09-35 | Tampering | open redirect via invite-token URL | mitigate | Token in path segment (not query); no redirect chain; SPA renders ClinicInvitePage which is a local component |
| T-09-36 | Replay | token reuse after acceptance | mitigate | accept_invite_existing UPDATE accepted_at + consumed_at; subsequent /lookup returns state 'already_used' |
| T-09-37 | Information disclosure | non-2xx Resend response leaks dispatch metadata | mitigate | Phase 4 pattern: never echo `res.text()`; wrap as `{ok:false, error:'resend_<status>'}` |
| T-09-38 | Tampering | CORS misconfiguration enables cross-origin invite sends from attacker page | accept | All clinic-invite endpoints require Authorization JWT (or anonymous token-in-URL) — CSRF via cross-origin form post requires the attacker to already hold a victim's JWT; same risk class as Supabase RPCs. Allow `*` matches ai-chat pattern. |

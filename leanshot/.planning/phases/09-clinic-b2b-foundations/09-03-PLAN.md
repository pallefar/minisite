    ```typescript
    import { useEffect, useState } from 'react';
    import { supabase } from './supabase';
    import type { PermissionKey } from '@/types/clinic';

    // Session-scoped Map; cleared on signOut by callers via clearPermissionCache().
    const cache = new Map<string, boolean>();  // key = `${orgId}:${permKey}`

    export function clearPermissionCache(): void { cache.clear(); }

    /**
     * Returns null while loading; true/false once resolved.
     * NOTE: This is a UX hint only. The server (RLS + RPC gates) is the security floor.
     * Hide an "Edit workspace" button when this returns false, but ALSO rely on the RPC
     * to reject if the user somehow triggers the action via dev tools.
     */
    export function useHasPermission(orgId: string | null, permissionKey: PermissionKey): boolean | null {
      const [value, setValue] = useState<boolean | null>(() => {
        if (!orgId) return null;
        const cached = cache.get(`${orgId}:${permissionKey}`);
        return cached === undefined ? null : cached;
      });
      useEffect(() => {
        if (!orgId) { setValue(null); return; }
        const key = `${orgId}:${permissionKey}`;
        if (cache.has(key)) { setValue(cache.get(key)!); return; }
        let cancelled = false;
        (async () => {
          const { data: user } = await supabase.auth.getUser();
          if (!user.user) { if (!cancelled) setValue(false); return; }
          const { data, error } = await supabase.rpc('has_permission', {
            p_user_id: user.user.id, p_org_id: orgId, p_permission_key: permissionKey,
          });
          if (cancelled) return;
          const result = error ? false : Boolean(data);
          cache.set(key, result);
          setValue(result);
        })();
        return () => { cancelled = true; };
      }, [orgId, permissionKey]);
      return value;
    }
    ```

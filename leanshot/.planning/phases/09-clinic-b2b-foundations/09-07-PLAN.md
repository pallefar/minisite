    ```typescript
    /*
     * clinic-photo Edge Function — D-12 3-check gate + 5-min signed URL per D-13.
     *
     * D-13 TRADEOFF: existing signed URLs remain valid for up to 5 min post-revoke.
     * The 3-check gate kills NEW mints immediately; URLs already minted will continue
     * to load photos until their natural 300s TTL expires. Documented Phase 9 tradeoff.
     *
     * Pitfall #11: CORS allow * Origin, NO credentials (this function is JWT-authed,
     * no cookies issued or read). Matches ai-chat/cors.ts verbatim.
     */
    import { createClient } from 'npm:@supabase/supabase-js@2';
    import { corsHeaders } from './cors.ts';

    const admin = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!,
      { auth: { autoRefreshToken: false, persistSession: false } },
    );

    const TTL_SECONDS = 300;

    function jsonError(status: number, code: string) {
      return new Response(JSON.stringify({ error: code }), {
        status,
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    async function logDenied(reason: string, operatorId: string | null, orgId: string | null, patientId: string | null, photoId: string | null) {
      try {
        await admin.rpc('log_clinic_event', {
          p_actor_type: 'org_operator',
          p_action: 'permission_denied',
          p_org_id: orgId,
          p_target_user_id: patientId,
          p_row_id: photoId,
          // Plan 09-01 log_clinic_event RPC may need to accept a `p_context` jsonb parameter
          // to capture the denial reason. If the RPC doesn't, accept the loss for v1.
        });
      } catch { /* swallow audit failures so the response shape is stable */ }
    }

    Deno.serve(async (req: Request) => {
      if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders });

      const url = new URL(req.url);
      const parts = url.pathname.split('/').filter(Boolean);
      // expected: ['clinic-photo', orgId, userId, photoId]
      const [, orgId, userId, photoId] = parts;
      if (!orgId || !userId || !photoId) return jsonError(400, 'bad_path');

      // 1. JWT
      const authHeader = req.headers.get('Authorization');
      if (!authHeader) return jsonError(401, 'no_auth');
      const token = authHeader.replace(/^Bearer\s+/i, '');
      const { data: { user: operator } } = await admin.auth.getUser(token);
      if (!operator) return jsonError(401, 'invalid_jwt');

      // 2. Operator membership active
      const { data: opMembership } = await admin
        .from('memberships')
        .select('id, role_id')
        .eq('user_id', operator.id).eq('org_id', orgId).is('revoked_at', null)
        .maybeSingle();
      if (!opMembership) {
        await logDenied('not_member', operator.id, orgId, userId, photoId);
        return jsonError(401, 'not_member');
      }

      // 3. Role permission
      const { data: hasPerm } = await admin.rpc('has_permission', {
        p_user_id: operator.id,
        p_org_id: orgId,
        p_permission_key: 'patient_photos.read',
      });
      if (!hasPerm) {
        await logDenied('permission_denied', operator.id, orgId, userId, photoId);
        return jsonError(403, 'permission_denied');
      }

      // 4. Patient membership + consent
      const { data: patientMembership } = await admin
        .from('memberships')
        .select('id, consent_scope, revoked_at')
        .eq('user_id', userId).eq('org_id', orgId).is('revoked_at', null)
        .maybeSingle();
      if (!patientMembership) {
        await logDenied('patient_not_member', operator.id, orgId, userId, photoId);
        return jsonError(401, 'patient_not_member');
      }
      if (patientMembership.consent_scope?.photos !== true) {
        await logDenied('consent_excluded', operator.id, orgId, userId, photoId);
        return jsonError(403, 'consent_excluded');
      }

      // 5. Photo row
      const { data: photo } = await admin
        .from('photos')
        .select('storage_path')
        .eq('user_id', userId).eq('photo_id', photoId)
        .maybeSingle();
      if (!photo) return jsonError(404, 'photo_not_found');

      // 6. Mint signed URL
      const { data: signed, error: signErr } = await admin.storage
        .from('photos')
        .createSignedUrl(photo.storage_path, TTL_SECONDS);
      if (signErr || !signed) return jsonError(500, 'sign_failed');

      // 7. Audit success
      await admin.rpc('log_clinic_event', {
        p_actor_type: 'org_operator',
        p_action: 'clinic_photo_view',
        p_org_id: orgId,
        p_target_user_id: userId,
        p_row_id: photoId,
      });

      return new Response(JSON.stringify({ signedUrl: signed.signedUrl, ttl: TTL_SECONDS }), {
        status: 200,
        headers: { ...corsHeaders, 'Content-Type': 'application/json', 'Cache-Control': 'private, no-store' },
      });
    });
    ```

    ```sql
      p_actor_type public.audit_actor_type,
      p_action text,
    ```

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-39 | Information disclosure | photo URL leaked post-revoke (within 5-min TTL) | accept | D-13 accepted tradeoff; documented in function header; mitigated by short TTL; revoke kills NEW mints immediately |
| T-09-40 | Elevation of privilege | operator without patient_photos.read accesses photo | mitigate | has_permission gate (check 2); audit_logs row on every denial |

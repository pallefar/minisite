```typescript
let _ses: SESv2Client | null = null;
function ses(): SESv2Client {
  if (_ses) return _ses;
  _ses = new SESv2Client({

    credentials: {

      secretAccessKey: Deno.env.get('AWS_SES_SECRET_ACCESS_KEY') ?? '',
    },
  });
  return _ses;
}

export type EmailTemplate = 'welcome' | 'receipt' | 'password_reset' | 'clinic_notification' | 'dose_alert' | 'doctor_share' | 'patient_access_log_notification' | ...;

export type SendEmailArgs = {
  template: EmailTemplate;
  to: string;
  vars: Record<string, unknown>;
  phi: boolean; // SINGLE switch — caller is authoritative on whether template contains PHI
};

export type SendEmailResult = { provider: 'ses'|'resend'; id: string; skipped?: boolean };

export async function sendEmail(supabase: SupabaseClient, args: SendEmailArgs): Promise<SendEmailResult> {
  if (!args.phi) {
    // Non-PHI: Resend (existing path)
    return await sendLifecycleEmail({ ...args }); // adapt to existing signature
  }
  // PHI: SES with vendor-gated send + suppression check
  const health = await awsSesHealthCheck();
  if (!health.ok) {
    console.warn('[email-router] SES no-op', health.status, args.template);
    return { provider: 'ses', id: 'noop-' + health.status, skipped: true };
  }
  // Suppression check (Pattern S5 — lookup before send)
  const recipientHash = await sha256Hex(args.to.toLowerCase());
  const { data: suppressed } = await supabase.from('ses_suppression_list')
    .select('recipient_hash').eq('recipient_hash', recipientHash).maybeSingle();
  if (suppressed) {
    console.warn('[email-router] recipient suppressed', recipientHash.slice(0, 8));
    return { provider: 'ses', id: 'suppressed', skipped: true };
  }
  // Render template — re-use email-layout.ts for HTML wrap
  const html = renderTemplate(args.template, args.vars); // ship a minimal switch on EmailTemplate
  try {
    const out = await ses().send(new SendEmailCommand({
      FromEmailAddress: Deno.env.get('AWS_SES_FROM') ?? 'LeanShot <noreply@app.leanshot.app>',
      Destination: { ToAddresses: [args.to] },
      Content: { Simple: { Subject: { Data: subjectFor(args.template, args.vars) }, Body: { Html: { Data: html } } } },
    }));
    return { provider: 'ses', id: out.MessageId ?? 'unknown' };
  } catch (e) {
    // T-25-03-S3 — NEVER echo e.message; may contain Authorization header in network errors
    console.error('[email-router] SES send threw', e instanceof Error ? e.name : 'unknown');
    throw new Error('ses_send_failed'); // stable short-code per T-25-03-I2
  }
}
```

```typescript
/**
 * ses-bounce-webhook Edge Function — Phase 25 Plan 25-03 (HIPAA-05).
 *
 * Single source of truth for AWS SES suppression list inserts.
 *
 * Security invariants:
 *  - RAW BODY read via `request.text()` BEFORE any JSON parse (Pattern: stripe-webhook).
 *  - SNS signature verify: SigningCertURL hostname allowlist `^https://sns\.[a-z0-9-]+\.amazonaws\.com/`;
 *    canonical string → SHA1WithRSA → base64 compare (no third-party sns-validator package).
 *  - Subscription confirmation: on `SubscriptionConfirmation` message type, GET the SubscribeURL
 *    once and return 200; do NOT auto-trust forwarded URLs (verify host pattern first).
 *  - Idempotency: `ses_suppression_list.sns_message_id UNIQUE` + `on conflict do nothing`.
 *  - PII safety: NEVER echo the recipient email in response or logs; hash it with sha256.
 *  - `Cache-Control: private, no-store` on every response.
 */
```

Flow:
1. CORS preflight handler returns 200 (SNS won't OPTIONS in practice but standard).
2. Method MUST be POST else 405.
3. `const raw = await req.text()` BEFORE any parse.
4. Read SNS headers: `X-Amz-Sns-Message-Type` (Notification / SubscriptionConfirmation / UnsubscribeConfirmation), `X-Amz-Sns-Topic-Arn`, `X-Amz-Sns-Message-Id`.
5. Parse `raw` as JSON.
6. Verify `SigningCertURL` host matches the allowlist regex (default `^https://sns\\.[a-z0-9-]+\\.amazonaws\\.com/` or override via `AWS_SES_SNS_SIGNING_CERT_HOSTPATTERN`). Reject 400 if not.
7. Compute the canonical string per SNS docs (different field order for Notification vs SubscriptionConfirmation per AWS spec), SHA1, verify against the `Signature` field (base64) using the cert public key. Cache cert by URL with 24h TTL via a module-level `Map<url, {cert, fetchedAt}>`. On verify fail → 401 + log `signature-verify-failed`.
8. If `Type === 'SubscriptionConfirmation'`: fetch the `SubscribeURL` once with timeout 10s, return 200 `{confirmed: true}`. Log success.
9. If `Type === 'Notification'`: parse the inner `Message` (string-encoded JSON) into the SES bounce/complaint shape per AWS SES Notification Contents docs. Extract:
   - `notificationType`: 'Bounce' | 'Complaint' | 'Delivery'. Only persist Bounce + Complaint; Delivery returns 200 ack only.
   - For Bounce: each `bounce.bouncedRecipients[].emailAddress`.
   - For Complaint: each `complaint.complainedRecipients[].emailAddress`.
10. For each recipient: lowercase, sha256-hex hash, INSERT into `ses_suppression_list (recipient_hash, suppression_reason, sns_message_id, raw_payload)` values (`hash`, `'bounce'|'complaint'`, `X-Amz-Sns-Message-Id`, the full raw_payload). Use `on conflict (sns_message_id) do nothing` for idempotency; if INSERT returns 0 rows (already processed) return 200 `{duplicate: true}`.
11. Return 200 `{accepted: true, recipients: <count>}` on success.

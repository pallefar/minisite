      language plpgsql
      security definer

    declare
      v_action text;
    begin
      if TG_OP = 'INSERT' then v_action := 'locale_override_create';

        case when TG_OP = 'DELETE' then null else row_to_json(NEW)::jsonb end,
        coalesce(NEW.org_id, OLD.org_id)
      );

      return coalesce(NEW, OLD);
    end;

    Structure:
    ```typescript
    import type { PostProcessorModule, i18n as I18nType } from 'i18next';
    import { supabase } from '@/lib/supabase';
    import { useStore } from '@/lib/store';

    // PostProcessor is a registration shim only — real work happens via i18n event listeners
    export const overrideBackend: PostProcessorModule = {
      type: 'postProcessor',
      name: 'localeOverrides',
      process(value: string) { return value; },
    };

    interface OverrideRow { lng: string; ns: string; key: string; value: string; }

    async function fetchOverrides(lng: string, ns: string): Promise<OverrideRow[]> {
      const orgId = useStore.getState().org?.id ?? null;
      let query = supabase
        .from('locale_overrides')
        .select('lng, ns, key, value')
        .eq('lng', lng).eq('ns', ns).eq('published', true);
      if (orgId) {
        query = query.or(`org_id.is.null,org_id.eq.${orgId}`);
      } else {
        query = query.is('org_id', null);
      }
      const { data, error } = await query;
      if (error) { console.warn('[i18n] override fetch failed', error); return []; }
      return data ?? [];
    }

    function buildNestedMap(rows: OverrideRow[]): Record<string, unknown> {
      const map: Record<string, unknown> = {};
      for (const row of rows) {
        const parts = row.key.split('.');
        let cursor: Record<string, unknown> = map;
        parts.forEach((part, i) => {
          if (i === parts.length - 1) cursor[part] = row.value;
          else { cursor[part] = (cursor[part] as Record<string, unknown>) ?? {}; cursor = cursor[part] as Record<string, unknown>; }
        });
      }
      return map;
    }

    async function applyOverrides(i18n: I18nType, lng: string, ns: string) {
      const rows = await fetchOverrides(lng, ns);
      if (!rows.length) return;
      const bundle = buildNestedMap(rows);
      i18n.addResourceBundle(lng, ns, bundle, true /*deep*/, true /*overwrite*/);
    }

    let channel: ReturnType<typeof supabase.channel> | null = null;

    export function subscribeToOverrideUpdates(i18n: I18nType): () => void {
      // Apply overrides when any namespace finishes loading
      const onLoaded = (loaded: Record<string, Record<string, boolean>>) => {
        for (const lng of Object.keys(loaded)) {
          for (const ns of Object.keys(loaded[lng])) {
            void applyOverrides(i18n, lng, ns);
          }
        }
      };
      i18n.on('loaded', onLoaded);

      // Realtime subscription scoped to current org (or global if anonymous)
      const orgId = useStore.getState().org?.id ?? 'global';
      channel = supabase
        .channel(`locale_overrides:${orgId}`)
        .on('postgres_changes', {
          event: '*', schema: 'public', table: 'locale_overrides', filter: 'published=eq.true',
        }, (payload) => {
          const row = (payload.new ?? payload.old) as OverrideRow | undefined;
          if (row?.lng && row?.ns) void applyOverrides(i18n, row.lng, row.ns);
        })
        .subscribe();

      return () => {
        i18n.off('loaded', onLoaded);
        if (channel) void supabase.removeChannel(channel);
        channel = null;
      };
    }
    ```

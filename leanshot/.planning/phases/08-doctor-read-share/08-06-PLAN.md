    ```typescript
    function bundleGuardPlugin(): Plugin {
      return {
        name: 'bundle-size-guard',
        closeBundle() {
          const distAssets = path.join(__dirname, 'dist', 'assets');
          if (!fs.existsSync(distAssets)) return;
          const files = fs.readdirSync(distAssets);

          const limits: Record<string, number> = {
            'index-': 22 * 1024,        // 22 kB gz (current 21.49 kB; small headroom)
            'share-': 18 * 1024,        // 18 kB gz (UI-SPEC bundle contract)
            'vendor-react-': 110 * 1024,
          };

          for (const [prefix, limitBytes] of Object.entries(limits)) {
            const match = files.find((f) => f.startsWith(prefix) && f.endsWith('.js'));
            if (!match) continue;
            const gzPath = path.join(distAssets, `${match}.gz`);
            if (!fs.existsSync(gzPath)) continue;
            const sizeGz = fs.statSync(gzPath).size;
            if (sizeGz > limitBytes) {
              throw new Error(`Bundle guard: ${match}.gz is ${sizeGz} bytes, exceeds ${limitBytes}`);
            }
          }
        },
      };
    }
    ```

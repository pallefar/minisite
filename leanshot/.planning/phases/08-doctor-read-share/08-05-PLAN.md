    import { createClient } from '@supabase/supabase-js';
    import type { BrowserContext } from '@playwright/test';

    const SUPABASE_URL = process.env.SUPABASE_URL!;
    const SERVICE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY!;
    const FN_BASE = `${SUPABASE_URL}/functions/v1`;

    const admin = createClient(SUPABASE_URL, SERVICE_KEY, { auth: { autoRefreshToken: false, persistSession: false } });

    // HI-6 — track every share created during the run for afterAll cleanup
    const createdShareIds = new Set<string>();

    export interface TestShare {
      share_id: string;
      raw_token: string;
      raw_code: string;
      patient_user_id: string;
      label: string;
    }

    async function ensureTestUser(email: string): Promise<string> {
      // Look up existing
      const { data: existing } = await admin.auth.admin.listUsers();
      const found = existing?.users.find((u) => u.email === email);
      if (found) return found.id;
      const { data: created, error } = await admin.auth.admin.createUser({
        email,
        password: 'TestPass123!',
        email_confirm: true,
      });
      if (error || !created.user) throw new Error(`Cannot create test user: ${error?.message}`);
      return created.user.id;
    }

    async function getAuthedClient(email: string) {
      const userClient = createClient(SUPABASE_URL, process.env.SUPABASE_ANON_KEY!);
      await userClient.auth.signInWithPassword({ email, password: 'TestPass123!' });
      return userClient;
    }

    export async function createTestShare(opts: {
      patientEmail: string;
      label?: string;
      expiresInHours?: number;
    }): Promise<TestShare> {
      const userId = await ensureTestUser(opts.patientEmail);
      const userClient = await getAuthedClient(opts.patientEmail);
      const expiresAt = new Date(Date.now() + (opts.expiresInHours ?? 24) * 3_600_000).toISOString();
      // HI-6 — always suffix with timestamp so concurrent runs do not collide
      const label = `${opts.label ?? 'test'}-${Date.now()}`;
      const { data, error } = await userClient.rpc('create_share', {
        p_label: label,
        p_expires_at: expiresAt,
      });
      if (error || !data) throw new Error(`create_share failed: ${error?.message}`);
      const row = Array.isArray(data) ? data[0] : data;
      createdShareIds.add(row.share_id);
      return {
        share_id: row.share_id,
        raw_token: row.raw_token,
        raw_code: row.raw_code,
        patient_user_id: userId,
        label,
      };
    }

    export async function revokeTestShare(shareId: string): Promise<void> {
      // service-role bypass; for fixture purposes we don't need to act as owner
      await admin.from('shares').update({ revoked_at: new Date().toISOString() }).eq('id', shareId);
    }

    export async function impersonateAsRecipient(
      context: BrowserContext,
      token: string,
      code: string,
    ): Promise<void> {
      // POST /share/redeem via fetch (Node), capture cookie, inject into Playwright context
      const resp = await fetch(`${FN_BASE}/share/redeem`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Origin': 'http://localhost:5173' },
        body: JSON.stringify({ token, code }),
      });
      if (!resp.ok) throw new Error(`redeem failed in fixture: ${resp.status}`);
      const setCookie = resp.headers.get('set-cookie') ?? '';
      const match = setCookie.match(/recipient_session=([^;]+)/);
      if (!match) throw new Error('No recipient_session in Set-Cookie');
      const value = match[1];
      const url = new URL(FN_BASE);
      await context.addCookies([{
        name: 'recipient_session',
        value,
        domain: url.hostname,
        path: '/',
        httpOnly: true,
        secure: true,
        sameSite: 'Strict',
      }]);
    }

    export async function assertAuditRowCount(shareId: string, expected: number): Promise<void> {
      const { count } = await admin.from('audit_logs').select('*', { count: 'exact', head: true })
        .eq('share_id', shareId)
        .eq('actor_type', 'share_recipient')
        .eq('action', 'share_view');
      if (count !== expected) throw new Error(`Expected ${expected} audit rows, got ${count}`);
    }

    /**
     * HI-6 — afterAll cleanup. Admin-DELETE every share row created during the run.
     * audit_logs.share_id is `on delete set null` (Plan 08-01) so audit rows are preserved,
     * just unlinked from the deleted share. Idempotent: safe to call multiple times.
     */
    export async function cleanupTestShares(): Promise<void> {
      if (createdShareIds.size === 0) return;
      const ids = Array.from(createdShareIds);
      const { error } = await admin.from('shares').delete().in('id', ids);
      if (error) {
        // Log but don't throw — cleanup must not mask test failures
        console.error(`cleanupTestShares: ${error.message}`);
      }
      createdShareIds.clear();
    }

    import { createTestShare, revokeTestShare, impersonateAsRecipient, assertAuditRowCount, cleanupTestShares } from './fixtures/shares';

    const FN_BASE = `${process.env.SUPABASE_URL}/functions/v1`;

    test.describe('Share revocation drill — 4 failure modes', () => {

      test.afterAll(async () => {
        await cleanupTestShares();  // HI-6
      });

      test('(a) token cache / DB-row check — revoke beats poll within 10s (HI-4)', async ({ browser }) => {
        const share = await createTestShare({ patientEmail: 'alice@test.com', label: 'drill-a', expiresInHours: 24 });
        const ctx = await browser.newContext();
        await impersonateAsRecipient(ctx, share.raw_token, share.raw_code);
        const page = await ctx.newPage();
        await page.goto(`/#/share/${share.raw_token}`);
        await expect(page.getByRole('heading', { name: /LeanShot record/ })).toBeVisible({ timeout: 10_000 });

        // Owner revokes
        await revokeTestShare(share.share_id);

        // Doctor's polling re-fetch picks up within 10s
        // (5s polling interval per HI-4 + 5s buffer; or 1 polling interval + 1s — both yield 10s)
        await expect(page.getByRole('heading', { name: 'This share has been revoked' })).toBeVisible({ timeout: 10_000 });

        await ctx.close();
      });

      test('(b) HTTP cache — Cache-Control: private, no-store on 200 + 401 + 410', async ({ request }) => {
        const share = await createTestShare({ patientEmail: 'alice@test.com', label: 'drill-b' });

        // 200 path

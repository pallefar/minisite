    ```typescript
    import { supabase } from '@/lib/supabase';
    import type { Share, CreateShareRequest, CreateShareResponse } from '@/types/share';

    export interface ShareWithStats extends Share {
      view_count: number;
      last_viewed_at: string | null;
      recipient_ua_families: string[];
      recipient_ip_families: string[];
    }

    export async function createShare(req: CreateShareRequest): Promise<CreateShareResponse> {
      const { data, error } = await supabase.rpc('create_share', {
        p_label: req.label,
        p_expires_at: req.expires_at,
      });
      if (error) throw new Error(`create_share failed: ${error.message}`);
      if (!data || data.length === 0) throw new Error('create_share returned no rows');
      const row = Array.isArray(data) ? data[0] : data;
      return { share_id: row.share_id, raw_token: row.raw_token, raw_code: row.raw_code };
    }

    export async function listActiveShares(): Promise<ShareWithStats[]> {
      // SELECT shares (RLS-scoped to auth.uid()) WHERE revoked_at IS NULL AND expires_at > now()
      const { data: shares, error: e1 } = await supabase
        .from('shares')
        .select('id, user_id, label, expires_at, revoked_at, code_consumed_at, created_at')
        .is('revoked_at', null)
        .gt('expires_at', new Date().toISOString())
        .order('created_at', { ascending: false });
      if (e1) throw new Error(`list shares failed: ${e1.message}`);
      if (!shares || shares.length === 0) return [];

      const ids = shares.map((s) => s.id);
      // Aggregate audit log per share — RLS scopes patient to own audit_logs rows (user_id=auth.uid())
      const { data: audit, error: e2 } = await supabase
        .from('audit_logs')
        .select('share_id, timestamp, recipient_ua_family, recipient_ip_family')
        .eq('actor_type', 'share_recipient')
        .in('share_id', ids);
      if (e2) throw new Error(`audit aggregate failed: ${e2.message}`);

      // Fold audit rows into per-share stats
      const stats = new Map<string, { count: number; last: string | null; uaSet: Set<string>; ipSet: Set<string> }>();
      for (const a of audit ?? []) {
        if (!a.share_id) continue;
        const s = stats.get(a.share_id) ?? { count: 0, last: null, uaSet: new Set(), ipSet: new Set() };
        s.count += 1;
        if (!s.last || a.timestamp > s.last) s.last = a.timestamp;
        if (a.recipient_ua_family) s.uaSet.add(a.recipient_ua_family);
        if (a.recipient_ip_family) s.ipSet.add(a.recipient_ip_family);
        stats.set(a.share_id, s);
      }

      return shares.map((s) => {
        const st = stats.get(s.id) ?? { count: 0, last: null, uaSet: new Set<string>(), ipSet: new Set<string>() };
        return {
          ...s,
          view_count: st.count,
          last_viewed_at: st.last,
          recipient_ua_families: [...st.uaSet],
          recipient_ip_families: [...st.ipSet],
        };
      });
    }

    export async function revokeShare(shareId: string): Promise<void> {
      const { error } = await supabase.rpc('revoke_share', { p_share_id: shareId });
      if (error) throw new Error(`revoke_share failed: ${error.message}`);
    }

    export function shareLinkFor(rawToken: string): string {
      return `${window.location.origin}/#/share/${rawToken}`;
    }
    ```

    interface PostCreationData {

    const [step, setStep] = useState<ModalStep>('form');
    const [label, setLabel] = useState('');
    const [expirySec, setExpirySec] = useState(7 * 24 * 60 * 60); // default 7 days
    const [submitErr, setSubmitErr] = useState<string | null>(null);
    const [postData, setPostData] = useState<PostCreationData | null>(null);

    async function handleSubmit() {
      setStep('submitting');
      try {
        const expires_at = new Date(Date.now() + expirySec * 1000).toISOString();
        const resp = await createShare({ label, expires_at });
        setPostData({ raw_token: resp.raw_token, raw_code: resp.raw_code });

        onCreated?.(resp.share_id);  // notify parent to prepend to list
      } catch (err) {
        setSubmitErr(err instanceof Error ? err.message : 'Failed');
        setStep('form');

      setLabel('');
      setExpirySec(7 * 24 * 60 * 60);
      setStep('form');
      setSubmitErr(null);
      onClose();

    **`src/components/share/share-client.ts`** — implement per `<interfaces>` block.

    ```typescript
    import type { SnapshotResponse, SnapshotError, RedeemError } from '@/types/share';

    const FN_BASE = (import.meta.env.VITE_SUPABASE_FUNCTIONS_URL as string | undefined)
      ?? `${import.meta.env.VITE_SUPABASE_URL}/functions/v1`;

    export type SnapshotResult =
      | { ok: true; data: SnapshotResponse }
      | { ok: false; error: SnapshotError };

    export async function fetchSnapshot(token: string): Promise<SnapshotResult> {
      const url = `${FN_BASE}/share/snapshot?token=${encodeURIComponent(token)}`;
      let resp: Response;
      try {
        resp = await fetch(url, { credentials: 'include', method: 'GET' });
      } catch {
        return { ok: false, error: 'not-found' }; // network error — fall to load-error state
      }
      if (resp.ok) {
        const data = await resp.json();
        return { ok: true, data };  // data includes data.share_id per BL-1 (Plan 08-02)
      }
      let body: { error?: SnapshotError } = {};
      try { body = await resp.json(); } catch { /* noop */ }
      const errorCode = (body.error ?? 'not-found') as SnapshotError;
      return { ok: false, error: errorCode };
    }

    export type RedeemResult =
      | { ok: true }
      | { ok: false; error: RedeemError; retryAfterSec?: number };

    export async function redeemShare(token: string, code: string): Promise<RedeemResult> {
      const url = `${FN_BASE}/share/redeem`;
      let resp: Response;
      try {
        resp = await fetch(url, {
          method: 'POST',
          credentials: 'include',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ token, code }),
        });
      } catch {
        return { ok: false, error: 'invalid-code' }; // network error → generic
      }
      if (resp.ok) return { ok: true };
      let body: { error?: RedeemError; retry_after_sec?: number } = {};
      try { body = await resp.json(); } catch { /* noop */ }
      const errorCode = (body.error ?? 'invalid-code') as RedeemError;
      return { ok: false, error: errorCode, retryAfterSec: body.retry_after_sec };
    }
    ```

    function CodeEntryScreen({ token, onSuccess }: Props) {
      const [code, setCode] = useState('');
      const [submitting, setSubmitting] = useState(false);
      const [error, setError] = useState<RedeemError | null>(null);
      const inputRef = useRef<HTMLInputElement>(null);
      const reducedMotion = useReducedMotion();

      useEffect(() => { inputRef.current?.focus(); }, []);

      // Auto-submit when 6 digits entered
      useEffect(() => {

          handleSubmit();
        }
      }, [code]);

      async function handleSubmit() {
        if (submitting) return;
        setSubmitting(true);
        setError(null);
        const result = await redeemShare(token, code);
        setSubmitting(false);
        if (result.ok) {
          onSuccess();
        } else {
          setError(result.error);
          setCode('');
          if (!reducedMotion) {
            // Trigger shake — toggle a CSS class via state
          }
        }

    ```typescript
    import type { SnapshotResponse, SnapshotError, RedeemError } from '@/types/share';

    const FN_BASE = (import.meta.env.VITE_SUPABASE_FUNCTIONS_URL as string | undefined)
      ?? `${import.meta.env.VITE_SUPABASE_URL}/functions/v1`;

    export type SnapshotResult =
      | { ok: true; data: SnapshotResponse }
      | { ok: false; error: SnapshotError };

    export async function fetchSnapshot(token: string): Promise<SnapshotResult> {
      const url = `${FN_BASE}/share/snapshot?token=${encodeURIComponent(token)}`;
      let resp: Response;
      try {
        resp = await fetch(url, { credentials: 'include', method: 'GET' });
      } catch {
        return { ok: false, error: 'not-found' }; // network error — fall to load-error state
      }
      if (resp.ok) {
        const data = await resp.json();
        return { ok: true, data };
      }
      let body: { error?: SnapshotError } = {};
      try { body = await resp.json(); } catch { /* noop */ }
      const errorCode = (body.error ?? 'not-found') as SnapshotError;
      return { ok: false, error: errorCode };
    }

    export type RedeemResult =
      | { ok: true }
      | { ok: false; error: RedeemError; retryAfterSec?: number };

    export async function redeemShare(token: string, code: string): Promise<RedeemResult> {
      const url = `${FN_BASE}/share/redeem`;
      let resp: Response;
      try {
        resp = await fetch(url, {
          method: 'POST',
          credentials: 'include',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ token, code }),
        });
      } catch {
        return { ok: false, error: 'invalid-code' }; // network error → generic
      }
      if (resp.ok) return { ok: true };
      let body: { error?: RedeemError; retry_after_sec?: number } = {};
      try { body = await resp.json(); } catch { /* noop */ }
      const errorCode = (body.error ?? 'invalid-code') as RedeemError;
      return { ok: false, error: errorCode, retryAfterSec: body.retry_after_sec };
    }
    ```

    ```typescript
    import { useState, useEffect, useCallback, lazy, Suspense } from 'react';
    import { fetchSnapshot } from './share-client';
    import { CodeEntryScreen } from './CodeEntryScreen';
    import { ShareRevokedScreen } from './ShareRevokedScreen';
    import type { SnapshotResponse } from '@/types/share';
    import { Card, Button, Pill, Skeleton } from '@/components/ui';
    import { Printer } from 'lucide-react';

    // Lazy-load chart + doctor report so they ride existing chunks
    const MedLevelChart = lazy(() => import('@/components/dashboard/charts/MedLevelChart').then((m) => ({ default: m.MedLevelChart })));
    const DoctorReport = lazy(() => import('@/components/dashboard/modals/DoctorReport').then((m) => ({ default: m.DoctorReport })));

    type State =
      | { kind: 'loading' }
      | { kind: 'needs-code' }
      | { kind: 'rendering'; snapshot: SnapshotResponse['snapshot']; expiresAt: string }
      | { kind: 'revoked' }
      | { kind: 'expired' }
      | { kind: 'error' };

    interface Props { token: string }

    export function SharePage({ token }: Props) {
      const [state, setState] = useState<State>({ kind: 'loading' });

      const load = useCallback(async () => {
        const result = await fetchSnapshot(token);
        if (result.ok) {
          setState({ kind: 'rendering', snapshot: result.data.snapshot, expiresAt: result.data.expires_at });
          return;
        }
        switch (result.error) {
          case 'requires-code':
          case 'invalid-session':
            setState({ kind: 'needs-code' });
            return;
          case 'revoked':
            setState({ kind: 'revoked' });
            return;
          case 'expired':
            setState({ kind: 'expired' });
            return;
          case 'not-found':
          default:
            setState({ kind: 'error' });
            return;
        }
      }, [token]);

      useEffect(() => { load(); }, [load]);

      // Poll while rendering — picks up revocation within 30s
    ```

    ```typescript
    import Stripe from 'https://esm.sh/stripe@19?target=denonext';
    import { _createServiceRoleClientUnsafe } from '../_shared/supabase-server.ts';
    import * as Sentry from '../_shared/sentry.ts';

    const EVENT_NAME = 'active_patient_month';

    // Startup health check per [[reference_vendor_gated_send_health_check]]
    if (!Deno.env.get('STRIPE_SECRET_KEY')) {
      console.warn('[org-metered-billing-cron] STRIPE_SECRET_KEY missing — cron will no-op');
    }

    function yyyymm(d: Date = new Date()): string {
      const y = d.getUTCFullYear();
      const m = String(d.getUTCMonth() + 1).padStart(2, '0');
      return `${y}${m}`;
    }

    export function buildMeterEventPayload(clinicId: string, count: number, stripeCustomerId: string, now = new Date()) {
      return {
        event_name: EVENT_NAME,
        payload: {
          value: String(count),          // Pitfall 3 — MUST be string
          stripe_customer_id: stripeCustomerId,
        },
        identifier: `org_${clinicId}_${yyyymm(now)}`,  // D-03 idempotency
      } as const;
    }

    export async function runForOrgs(opts: {
      stripe: Stripe;
      admin: ReturnType<typeof _createServiceRoleClientUnsafe>;
      orgs: Array<{ clinic_id: string; stripe_customer_id: string }>;
      now?: Date;
    }) {
      const results: Array<{ clinic_id: string; ok: boolean; error?: string }> = [];
      for (const org of opts.orgs) {
        try {
          const { data: count, error } = await opts.admin.rpc('count_active_patients', { p_org_id: org.clinic_id });
          if (error) throw error;
          const payload = buildMeterEventPayload(org.clinic_id, Number(count ?? 0), org.stripe_customer_id, opts.now);
          await opts.stripe.billing.meterEvents.create(payload);
          results.push({ clinic_id: org.clinic_id, ok: true });
        } catch (err) {
          Sentry.captureException(err, { tags: { cron: 'org-metered-billing', clinic_id: org.clinic_id } });
          results.push({ clinic_id: org.clinic_id, ok: false, error: err instanceof Error ? err.message : String(err) });
          // continue loop — per-org isolation
        }
      }
      return results;
    }

    Deno.serve(async (_req) => {
      const stripeKey = Deno.env.get('STRIPE_SECRET_KEY');
      if (!stripeKey) {
        return new Response(JSON.stringify({ ok: false, error: 'stripe_key_missing' }), { status: 503 });
      }
      const stripe = new Stripe(stripeKey, { apiVersion: '2026-04-22.dahlia' as Stripe.LatestApiVersion });
      const admin = _createServiceRoleClientUnsafe();
      const { data: orgs, error } = await admin
        .from('clinic_stripe_customers')
        .select('clinic_id, stripe_customer_id')
        .not('stripe_customer_id', 'is', null);
      if (error) {
        Sentry.captureException(error, { tags: { cron: 'org-metered-billing', stage: 'orgs_query' } });
        return new Response(JSON.stringify({ ok: false, error: 'orgs_query_failed' }), { status: 500 });
      }
      const results = await runForOrgs({ stripe, admin, orgs: orgs ?? [] });
      const okCount = results.filter(r => r.ok).length;
      return new Response(JSON.stringify({ ok: true, processed: results.length, succeeded: okCount, failed: results.length - okCount }), { status: 200 });
    });
    ```

    ```typescript
    import { assertEquals, assertMatch, assertExists } from 'jsr:@std/assert';
    import { buildMeterEventPayload, runForOrgs } from './index.ts';

    Deno.test('value is a string (Pitfall 3)', () => {
      const p = buildMeterEventPayload('abc-1234', 42, 'cus_xyz');
      assertEquals(typeof p.payload.value, 'string');
      assertEquals(p.payload.value, '42');

    Deno.test('identifier format = org_${uuid}_YYYYMM', () => {
      const p = buildMeterEventPayload('11111111-2222-3333-4444-555555555555', 10, 'cus_x', new Date(Date.UTC(2026, 4, 17)));
      assertEquals(p.identifier, 'org_11111111-2222-3333-4444-555555555555_202605');
      assertMatch(p.identifier, /^org_[0-9a-f-]{36}_[0-9]{6}$/);
    });

    Deno.test('payload contains ONLY value + stripe_customer_id (D-11 PHI lint)', () => {
      const p = buildMeterEventPayload('abc', 1, 'cus_a');
      assertEquals(Object.keys(p.payload).sort(), ['stripe_customer_id', 'value']);
    });

    Deno.test('per-org error isolation: one failure does not stop the loop', async () => {
      const calls: string[] = [];
      const stripe = { billing: { meterEvents: { create: async (p: any) => {
        calls.push(p.identifier);
        if (p.identifier.includes('aaaa')) throw new Error('stripe boom');
        return { id: 'me_x' };
      }}}} as any;
      const admin = { rpc: async (_fn: string, { p_org_id }: any) => ({ data: 3, error: null }) } as any;
      const results = await runForOrgs({
        stripe, admin,
        orgs: [
          { clinic_id: 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa', stripe_customer_id: 'cus_a' },
          { clinic_id: 'bbbbbbbb-bbbb-bbbb-bbbb-bbbbbbbbbbbb', stripe_customer_id: 'cus_b' },
        ],
      });
      assertEquals(results.length, 2);
      assertEquals(results[0].ok, false);
      assertEquals(results[1].ok, true);
      assertEquals(calls.length, 2); // BOTH attempted

      const created: any[] = [];

      await runForOrgs({ stripe, admin, orgs: [{ clinic_id: 'org-x', stripe_customer_id: 'cus_x' }] });
      assertEquals(created.length, 1);
      assertEquals(created[0].payload.value, '0');
    ```

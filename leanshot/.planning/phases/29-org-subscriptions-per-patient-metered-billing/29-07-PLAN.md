    ```typescript
    #!/usr/bin/env tsx
    // Phase 25 D-09 + Phase 29 D-11 — block PHI keywords in Stripe call sites.
    import { readFileSync, readdirSync, statSync } from 'node:fs';
    import { join } from 'node:path';

    const STRIPE_PATHS = [
      'supabase/functions/stripe-webhook',
      'supabase/functions/stripe-checkout',
      'supabase/functions/admin-stripe-action',
      'supabase/functions/org-metered-billing-cron', // P29 D-11
    ];
    const KEYWORDS = JSON.parse(readFileSync('scripts/stripe-phi-keywords.json', 'utf8'));

    function* walk(dir: string): Generator<string> {
      for (const entry of readdirSync(dir, { withFileTypes: true })) {
        const p = join(dir, entry.name);
        if (entry.isDirectory()) yield* walk(p);
        else if (entry.isFile() && (p.endsWith('.ts') || p.endsWith('.js'))) yield p;
      }
    }

    let violations = 0;
    for (const root of STRIPE_PATHS) {
      try { statSync(root); } catch { continue; }
      for (const file of walk(root)) {
        const content = readFileSync(file, 'utf8');
        // Strip single-line comments per [[reference_grep_gate_comment_strip]]
        const stripped = content.split('\n').filter(l => !l.trim().startsWith('//')).join('\n');
        for (const kw of KEYWORDS.keywords) {
          // Match whole-word keyword
          const rx = new RegExp(`\\b${kw}\\b`, 'gi');
          const matches = stripped.match(rx);
          if (matches) {
            console.error(`PHI keyword "${kw}" found in ${file} (${matches.length}x)`);
            violations += matches.length;
          }
        }
      }
    }
    if (violations > 0) {
      console.error(`\nFAIL: ${violations} PHI keyword violations in Stripe call sites.`);
      process.exit(1);
    }
    console.log(`OK: no PHI keywords in ${STRIPE_PATHS.length} Stripe call site directories.`);
    ```

Use the `allowlist_inline_marker` convention (`// stripe-phi-lint:allow reason='X'`) to suppress legitimate occurrences (per Phase 25 D-09 design).

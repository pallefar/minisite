    ```sql
    CREATE OR REPLACE FUNCTION public.rank_org_patients(
      p_org_id uuid,
      p_sort_column text DEFAULT 'score',
      p_sort_direction text DEFAULT 'desc',
      p_offset int DEFAULT 0,
      p_limit int DEFAULT 50
    )
    RETURNS TABLE (
      user_id uuid,
      display_name text,
      score smallint,
      breakdown jsonb,
      last_injection_at timestamptz,
      weight_trend_arrow text,
      recent_symptom_severity smallint,
      days_since_injection int,
      missed_dose_flag bool
    )
    LANGUAGE plpgsql
    SECURITY DEFINER
    SET search_path = public, extensions, pg_catalog
    AS $$
    DECLARE
      v_caller_uid uuid := auth.uid();
      v_patient_count int;
      v_threshold_payload jsonb;
    BEGIN
      -- 1. Permission gate (D-01)
      IF NOT public.has_permission(v_caller_uid, p_org_id, 'patient_data.read') THEN
        RAISE EXCEPTION 'access_denied';
      END IF;

      -- 2. Validate sort params (whitelist)
      IF p_sort_column NOT IN ('score','name','last_dose','weight_trend','symptom_severity','days_since_injection','missed_dose_flag') THEN
        RAISE EXCEPTION 'invalid_sort_column';
      END IF;
      IF p_sort_direction NOT IN ('asc','desc') THEN
        RAISE EXCEPTION 'invalid_sort_direction';
      END IF;
      IF p_limit < 1 OR p_limit > 200 THEN
        RAISE EXCEPTION 'invalid_limit';
      END IF;

      -- 3. Build per-patient signal CTE (mirror Phase 3 weight architecture)
      RETURN QUERY EXECUTE format(
        $sql$
        WITH patients AS (
          SELECT m.user_id, u.raw_user_meta_data->>'display_name' AS display_name, m.consent_scope
          FROM public.memberships m
          JOIN auth.users u ON u.id = m.user_id
          WHERE m.org_id = %L
            AND m.revoked_at IS NULL
        ),
        signals AS (
          SELECT
            p.user_id,
            COALESCE(p.display_name, 'Patient') AS display_name,
            -- last_injection_at: NULL if consent_scope.injections=false
            CASE WHEN (p.consent_scope->>'injections')::bool THEN
              (SELECT MAX(i.created_at) FROM public.injections i WHERE i.user_id = p.user_id)
            ELSE NULL END AS last_injection_at,
            -- weight_trend_arrow: latest 3 weights, simple delta
            CASE WHEN (p.consent_scope->>'weights')::bool THEN
              public.compute_weight_trend(p.user_id)
            ELSE 'flat'::text END AS weight_trend_arrow,
            -- recent_symptom_severity: max severity in last 7d
            CASE WHEN (p.consent_scope->>'symptoms')::bool THEN
              COALESCE((SELECT MAX(s.severity)::smallint FROM public.symptoms s WHERE s.user_id = p.user_id AND s.recorded_at > now() - interval '7 days'), 0::smallint)
            ELSE 0::smallint END AS recent_symptom_severity,
            -- days_since_injection: 999 sentinel when no injection or no consent
            CASE WHEN (p.consent_scope->>'injections')::bool THEN
              COALESCE(EXTRACT(DAY FROM (now() - (SELECT MAX(i.created_at) FROM public.injections i WHERE i.user_id = p.user_id)))::int, 999)
            ELSE 999 END AS days_since_injection,
            -- missed_dose_flag: days_since_injection > 14 for an active patient
            CASE WHEN (p.consent_scope->>'injections')::bool THEN
              COALESCE(EXTRACT(DAY FROM (now() - (SELECT MAX(i.created_at) FROM public.injections i WHERE i.user_id = p.user_id)))::int, 0) > 14
            ELSE false END AS missed_dose_flag
          FROM patients p
        ),
        scored AS (
          SELECT
            s.*,
            -- Score weights mirror Phase 3 pickFocus/generateInsights priority order:
            -- missed_dose (28) > symptom_severity (severity * 5, max 25) > weight_trend reversal (15) > days_since_injection (linear * 1, capped 20) > streak break (12)
            LEAST(100, GREATEST(0,
              (CASE WHEN s.missed_dose_flag THEN 28 ELSE 0 END)
              + (s.recent_symptom_severity * 5)
              + (CASE WHEN s.weight_trend_arrow = 'down' THEN 15 ELSE 0 END)
              + LEAST(20, s.days_since_injection)
              + (CASE WHEN s.days_since_injection > 7 THEN 12 ELSE 0 END)
            ))::smallint AS score,
            jsonb_build_object(
              'missed_dose', CASE WHEN s.missed_dose_flag THEN 28 ELSE 0 END,
              'symptom_severity', s.recent_symptom_severity * 5,
              'weight_trend_reversal', CASE WHEN s.weight_trend_arrow = 'down' THEN 15 ELSE 0 END,
              'days_since_injection', LEAST(20, s.days_since_injection),
              'streak_break', CASE WHEN s.days_since_injection > 7 THEN 12 ELSE 0 END
            ) AS breakdown
          FROM signals s
        )
        SELECT user_id, display_name, score, breakdown, last_injection_at, weight_trend_arrow, recent_symptom_severity, days_since_injection, missed_dose_flag
        FROM scored
        ORDER BY %I %s NULLS LAST, user_id ASC
        OFFSET %L LIMIT %L
        $sql$,
        p_org_id,
        CASE p_sort_column
          WHEN 'score' THEN 'score'
          WHEN 'name' THEN 'display_name'
          WHEN 'last_dose' THEN 'last_injection_at'
          WHEN 'weight_trend' THEN 'weight_trend_arrow'
          WHEN 'symptom_severity' THEN 'recent_symptom_severity'
          WHEN 'days_since_injection' THEN 'days_since_injection'
          WHEN 'missed_dose_flag' THEN 'missed_dose_flag'
        END,
        p_sort_direction,
        p_offset,
        p_limit
      );

      GET DIAGNOSTICS v_patient_count = ROW_COUNT;

      -- 4. Audit row for the rank computation (D-18)
      INSERT INTO public.audit_logs (action, actor_id, actor_type, org_id, metadata, created_at)
      VALUES (
        'rank_computed', v_caller_uid, 'org_system', p_org_id,
        jsonb_build_object(
          'patient_count', v_patient_count,
          'weights_snapshot', jsonb_build_object('missed_dose', 28, 'symptom_severity_per_unit', 5, 'weight_trend_reversal', 15, 'days_since_injection_max', 20, 'streak_break', 12)
        ),
        now()
      );

      -- 5. Threshold-crossing detection — compare against most recent prior rank_computed call's per-patient breakdown
      -- For each patient, look up their last `rank_threshold_crossed` event for this org; if score now ≥70 and last event direction='down' (or no event ever and current ≥70), insert 'up' event. Reverse for ≤69 after 'up'.
      -- (executor: implement this as a CTE that joins audit_logs to current rank result; insert 1 row per crossing into audit_logs with action='rank_threshold_crossed', target_user_id=user_id, metadata={score, breakdown_snapshot, threshold_crossed: 70, direction}.)
      -- NOTE: this block is the most complex SQL in the migration; it MUST tolerate the case where this is the first call ever (no prior 'rank_threshold_crossed' rows — emit 'up' for any current ≥70).
      -- For implementation reference: query `SELECT DISTINCT ON (target_user_id) target_user_id, metadata->>'direction' AS last_direction FROM audit_logs WHERE org_id = p_org_id AND action = 'rank_threshold_crossed' ORDER BY target_user_id, created_at DESC` and join against the just-returned rank result.

    END;
    $$;

    -- Helper function: compute_weight_trend(user_id) — returns 'up' | 'down' | 'flat' from last 3 weight rows
    CREATE OR REPLACE FUNCTION public.compute_weight_trend(p_user_id uuid)
    RETURNS text
    LANGUAGE plpgsql
    SECURITY DEFINER
    SET search_path = public, extensions, pg_catalog
    AS $$
    DECLARE
      v_recent numeric;
      v_older numeric;
    BEGIN
      SELECT AVG(weight_kg) INTO v_recent FROM (SELECT weight_kg FROM public.weights WHERE user_id = p_user_id ORDER BY recorded_at DESC LIMIT 1) sub;
      SELECT AVG(weight_kg) INTO v_older FROM (SELECT weight_kg FROM public.weights WHERE user_id = p_user_id ORDER BY recorded_at DESC OFFSET 1 LIMIT 2) sub;
      IF v_recent IS NULL OR v_older IS NULL THEN RETURN 'flat'; END IF;
      IF v_recent > v_older + 0.3 THEN RETURN 'up';
      ELSIF v_recent < v_older - 0.3 THEN RETURN 'down';
      ELSE RETURN 'flat'; END IF;
    END;
    $$;

    REVOKE ALL ON FUNCTION public.rank_org_patients(uuid, text, text, int, int) FROM public;
    GRANT EXECUTE ON FUNCTION public.rank_org_patients(uuid, text, text, int, int) TO authenticated;
    REVOKE ALL ON FUNCTION public.compute_weight_trend(uuid) FROM public;
    GRANT EXECUTE ON FUNCTION public.compute_weight_trend(uuid) TO authenticated;
    ```

    **Critical implementation notes:**
    - The `EXECUTE format(...)` pattern is needed because the `ORDER BY` column is dynamic. Use `%I` for the column identifier (auto-quotes) and `%s` for the direction (whitelisted above so safe). Do NOT use raw string concatenation.
    - The `compute_weight_trend` helper is SECURITY DEFINER because the RPC needs to read patient weights without exposing them to the operator's RLS context — the operator only sees the trend arrow.
    - Threshold-crossing is the most subtle logic. Executor: implement the comparison via a temp table or CTE inside the function body, joining the just-computed scored CTE with the most recent prior `rank_threshold_crossed` audit row per patient; insert one row per state transition.

       ```tsx
       import { useEffect, useRef } from 'react';
       import type { ReadOnlyPatientViewProps } from '@/types/snapshot';
       import { InjectionsSection } from './sections/InjectionsSection';
       import { WeightsSection } from './sections/WeightsSection';
       import { SymptomsSection } from './sections/SymptomsSection';
       import { PhotosSection } from './sections/PhotosSection';
       import { DoctorReportSection } from './sections/DoctorReportSection';
       import { ChartSection } from './sections/ChartSection';

       const SECTION_ORDER = ['chart','injections','weights','symptoms','photos','doctor_report'] as const;

       export function ReadOnlyPatientView({ snapshot, viewerMode, permissionMap, onSectionMount }: ReadOnlyPatientViewProps) {
         // Default: every section visible. Clinic mode applies permissionMap gating per D-12.
         const visible = (key: string) => {
           if (viewerMode === 'share') return true;
           if (!permissionMap) return true;
           switch (key) {
             case 'injections': return permissionMap.canViewInjections;
             case 'weights': return permissionMap.canViewWeights;
             case 'symptoms': return permissionMap.canViewSymptoms;
             case 'photos': return permissionMap.canViewPhotos;
             case 'doctor_report': return permissionMap.canViewDoctorReport;
             case 'chart': return permissionMap.canViewInjections; // chart depends on injections
             default: return true;
           }
         };

         return (
           <div className="space-y-8">
             {visible('chart') && <ChartSection data={snapshot.injections} viewerMode={viewerMode} onMount={onSectionMount} />}
             {visible('injections') && <InjectionsSection data={snapshot.injections} viewerMode={viewerMode} onMount={onSectionMount} />}
             {visible('weights') && <WeightsSection data={snapshot.weights} viewerMode={viewerMode} onMount={onSectionMount} />}
             {visible('symptoms') && <SymptomsSection data={snapshot.symptoms} viewerMode={viewerMode} onMount={onSectionMount} />}
             {visible('photos') && <PhotosSection data={snapshot.photos} viewerMode={viewerMode} orgId={snapshot.org_id} onMount={onSectionMount} />}
             {visible('doctor_report') && <DoctorReportSection data={snapshot.doctor_report} viewerMode={viewerMode} onMount={onSectionMount} />}
           </div>
         );
       }
       export default ReadOnlyPatientView;
       ```
